   346→   346→function LogoUpload({
   347→   347→  logo,
   348→   348→  etablissementId,
   349→   349→  accent,
   350→   350→  onLogoUpdate,
   351→   351→}: {
   352→   352→  logo: string | null
   353→   353→  etablissementId: string
   354→   354→  accent: ReturnType<typeof getAccent>
   355→   355→  onLogoUpdate: (logo: string | null) => void
   356→   356→}) {
   357→   357→  const fileInputRef = useRef<HTMLInputElement>(null)
   358→   358→  const [uploading, setUploading] = useState(false)
   359→   359→  const [dragOver, setDragOver] = useState(false)
   360→   360→
   361→   361→  const handleFileSelect = async (file: File) => {
   362→   362→    const allowedTypes = ['image/png', 'image/jpeg', 'image/webp', 'image/svg+xml']
   363→   363→    if (!allowedTypes.includes(file.type)) {
   364→   364→      toast.error('Format non supporté', { description: 'Utilisez PNG, JPG, WEBP ou SVG.' })
   365→   365→      return
   366→   366→    }
   367→   367→    if (file.size > 2 * 1024 * 1024) {
   368→   368→      toast.error('Fichier trop volumineux', { description: 'La taille maximale est de 2 Mo.' })
   369→   369→      return
   370→   370→    }
   371→   371→
   372→   372→    setUploading(true)
   373→   373→    try {
   374→   374→      const formData = new FormData()
   375→   375→      formData.append('file', file)
   376→   376→      formData.append('etablissementId', etablissementId)
   377→   377→      const res = await fetch('/api/etablissements/upload-logo', {
   378→   378→        method: 'POST',
   379→   379→        body: formData,
   380→   380→      })
   381→   381→      if (!res.ok) {
   382→   382→        const err = await res.json().catch(() => ({}))
   383→   383→        throw new Error(err.error || 'Erreur lors de l\'upload')
   384→   384→      }
   385→   385→      const data = await res.json()
   386→   386→      onLogoUpdate(data.logo || data.etablissement?.logo || null)
   387→   387→      toast.success('Logo mis à jour', { description: 'Le logo a été téléchargé avec succès.' })
   388→   388→    } catch (err) {
   389→   389→      toast.error('Erreur', {
   390→   390→        description: err instanceof Error ? err.message : 'Impossible de télécharger le logo.',
   391→   391→      })
   392→   392→    } finally {
   393→   393→      setUploading(false)
   394→   394→    }
   395→   395→  }
   396→   396→
   397→   397→  const handleDrop = (e: React.DragEvent) => {
   398→   398→    e.preventDefault()
   399→   399→    setDragOver(false)
   400→   400→    const file = e.dataTransfer.files[0]
   401→   401→    if (file) handleFileSelect(file)
   402→   402→  }
   403→   403→
   404→   404→  const handleDelete = async () => {
   405→   405→    setUploading(true)
   406→   406→    try {
   407→   407→      // PARAMETRES-FIX-P3 : endpoint dédié DELETE /api/etablissements/{id}/logo.
   408→   408→      // PARAMETRES-FIX-P3b : le backend utilise désormais ClearLogo (SET logo = NULL)
   409→   409→      // au lieu d'UpdateLogo(...,"") qui était rejeté par la validation.
   410→   410→      const res = await fetch(`/api/etablissements/${etablissementId}/logo`, {
   411→   411→        method: 'DELETE',
   412→   412→      })
   413→   413→      if (!res.ok) {
   414→   414→        const err = await res.json().catch(() => ({}))
   415→   415→        throw new Error(err.error || err.message || `Erreur ${res.status}`)
   416→   416→      }
   417→   417→      onLogoUpdate(null)
   418→   418→      toast.success('Logo supprimé', { description: 'Le logo a été retiré.' })
   419→   419→    } catch (err) {
   420→   420→      toast.error('Erreur', {
   421→   421→        description: err instanceof Error ? err.message : 'Impossible de supprimer le logo.',
   422→   422→      })
   423→   423→    } finally {
   424→   424→      setUploading(false)
   425→   425→    }
   426→   426→  }
   427→   427→
   428→   428→  return (
   429→   429→    <div className="space-y-3">
   430→   430→      <Label className="flex items-center gap-1.5">
   431→   431→        <ImageIcon className="h-3.5 w-3.5 text-muted-foreground" />
   432→   432→        Logo de l&apos;établissement
   433→   433→      </Label>
   434→   434→      <div className="flex items-start gap-4">
   435→   435→        {/* Preview */}
   436→   436→        <div
   437→   437→          className={`relative flex h-20 w-20 shrink-0 items-center justify-center rounded-xl border-2 border-dashed transition-colors cursor-pointer overflow-hidden ${
   438→   438→            dragOver ? `${accent.activeBorder} ${accent.activeBg}` : 'border-muted-foreground/25 hover:border-muted-foreground/50'
   439→   439→          }`}
   440→   440→          onClick={() => fileInputRef.current?.click()}
   441→   441→          onDragOver={(e) => { e.preventDefault(); setDragOver(true) }}
   442→   442→          onDragLeave={() => setDragOver(false)}
   443→   443→          onDrop={handleDrop}
   444→   444→        >
   445→   445→          {logo ? (
   446→   446→            <img src={logo} alt="Logo" className="h-full w-full object-cover rounded-xl" />
   447→   447→          ) : (
   448→   448→            <div className="flex flex-col items-center gap-1 text-muted-foreground/50">
   449→   449→              <Upload className="h-5 w-5" />
   450→   450→              <span className="text-[10px]">Clic/Glisser</span>
   451→   451→            </div>
   452→   452→          )}
   453→   453→          {uploading && (
   454→   454→            <div className="absolute inset-0 flex items-center justify-center bg-background/80 rounded-xl">
   455→   455→              <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   456→   456→            </div>
   457→   457→          )}
   458→   458→        </div>
   459→   459→        <div className="flex-1 space-y-2">
   460→   460→          <p className="text-xs text-muted-foreground">
   461→   461→            Cliquez ou glissez-déposez une image. Formats acceptés : PNG, JPG, WEBP, SVG (max 2 Mo).
   462→   462→          </p>
   463→   463→          <div className="flex gap-2">
   464→   464→            <Button
   465→   465→              variant="outline"
   466→   466→              size="sm"
   467→   467→              onClick={() => fileInputRef.current?.click()}
   468→   468→              disabled={uploading}
   469→   469→            >
   470→   470→              <Upload className="h-3.5 w-3.5 mr-1.5" />
   471→   471→              Parcourir
   472→   472→            </Button>
   473→   473→            {logo && (
   474→   474→              <Button
   475→   475→                variant="outline"
   476→   476→                size="sm"
   477→   477→                onClick={handleDelete}
   478→   478→                disabled={uploading}
   479→   479→                className="text-destructive hover:text-destructive hover:bg-destructive/10"
   480→   480→              >
   481→   481→                <Trash2 className="h-3.5 w-3.5 mr-1.5" />
   482→   482→                Supprimer
   483→   483→              </Button>
   484→   484→            )}
   485→   485→          </div>
   486→   486→        </div>
   487→   487→      </div>
   488→   488→      <input
   489→   489→        ref={fileInputRef}
