    42→/* ─── Magnetic Button ─── */
    43→function MagneticButton({
    44→  children,
    45→  className = '',
    46→  ...props
    47→}: React.ComponentProps<typeof Button> & { children: React.ReactNode }) {
    48→  const ref = useRef<HTMLButtonElement>(null)
    49→  const xTo = useRef<gsap.QuickToFunc | null>(null)
    50→  const yTo = useRef<gsap.QuickToFunc | null>(null)
    51→
    52→  useEffect(() => {
    53→    if (!ref.current) return
    54→    xTo.current = gsap.quickTo(ref.current, 'x', { duration: 0.3, ease: 'power2.out' })
    55→    yTo.current = gsap.quickTo(ref.current, 'y', { duration: 0.3, ease: 'power2.out' })
    56→    return () => {
    57→      xTo.current = null
    58→      yTo.current = null
    59→    }
    60→  }, [])
    61→
    62→  const handleMouseMove = useCallback(
    63→    (e: React.MouseEvent<HTMLButtonElement>) => {
    64→      if (!ref.current || !xTo.current || !yTo.current) return
    65→      const rect = ref.current.getBoundingClientRect()
    66→      const x = e.clientX - rect.left - rect.width / 2
    67→      const y = e.clientY - rect.top - rect.height / 2
    68→      xTo.current(x * 0.3)
    69→      yTo.current(y * 0.3)
    70→    },
    71→    []
    72→  )
    73→
    74→  const handleMouseLeave = useCallback(() => {
    75→    xTo.current?.(0)
    76→    yTo.current?.(0)
    77→  }, [])
    78→
    79→  return (
    80→    <Button
    81→      ref={ref}
    82→      className={className}
    83→      onMouseMove={handleMouseMove}
    84→      onMouseLeave={handleMouseLeave}
    85→      {...props}
    86→    >
    87→      {children}
    88→    </Button>
    89→  )
    90→}

   267→/* ─── Hero Section ─── */
   268→function HeroSection({ onDemo }: { onDemo: () => void }) {
   269→  const sectionRef = useRef<HTMLDivElement>(null)
   270→  const headlineRef = useRef<HTMLHeadingElement>(null)
   271→  const subRef = useRef<HTMLParagraphElement>(null)
   272→  const ctaRef = useRef<HTMLDivElement>(null)
   273→  const imageRef = useRef<HTMLDivElement>(null)
   274→  const typingRef = useRef<HTMLSpanElement>(null)
   275→
   276→  useEffect(() => {
   277→    const ctx = gsap.context(() => {
   278→      // Split headline into words and animate
   279→      if (headlineRef.current) {
   280→        const words = headlineRef.current.querySelectorAll('.hero-word')
   281→        gsap.fromTo(
   282→          words,
   283→          { y: 80, opacity: 0, rotateX: -40 },
   284→          {
   285→            y: 0,
   286→            opacity: 1,
   287→            rotateX: 0,
   288→            duration: 1.2,
   289→            stagger: 0.08,
   290→            ease: 'power3.out',
   291→            delay: 0.3,
   292→          }
   293→        )
   294→      }
   295→
   296→      // Subtitle fade in
   297→      if (subRef.current) {
   298→        gsap.fromTo(
   299→          subRef.current,
   300→          { opacity: 0, y: 20 },
   301→          { opacity: 1, y: 0, duration: 1, ease: 'power2.out', delay: 1.2 }
   302→        )
   303→      }
   304→
   305→      // CTA buttons
   306→      if (ctaRef.current) {
   307→        gsap.fromTo(
   308→          ctaRef.current.children,
   309→          { opacity: 0, y: 20 },
   310→          { opacity: 1, y: 0, duration: 0.8, stagger: 0.15, ease: 'power2.out', delay: 1.5 }
   311→        )
   312→      }
   313→
   314→      // Dashboard image parallax
   315→      if (imageRef.current) {
   316→        gsap.fromTo(
   317→          imageRef.current,
   318→          { y: 60, opacity: 0, scale: 0.95 },
   319→          { y: 0, opacity: 1, scale: 1, duration: 1.4, ease: 'power3.out', delay: 1.8 }
   320→        )
   321→
   322→        // Parallax on scroll
   323→        gsap.to(imageRef.current, {
   324→          y: -40,
   325→          ease: 'none',
   326→          scrollTrigger: {
   327→            trigger: sectionRef.current,
   328→            start: 'top top',
   329→            end: 'bottom top',
   330→            scrub: 1,
   331→          },
   332→        })
   333→      }
   334→    }, sectionRef)
   335→
   336→    // 3D tilt on mouse move
   337→    const handleMouseMove = (e: MouseEvent) => {
   338→      if (!imageRef.current) return
   339→      const rect = imageRef.current.getBoundingClientRect()
   340→      const centerX = rect.left + rect.width / 2
   341→      const centerY = rect.top + rect.height / 2
   342→      const rotateY = ((e.clientX - centerX) / (rect.width / 2)) * 5
   343→      const rotateX = -((e.clientY - centerY) / (rect.height / 2)) * 5
   344→      gsap.to(imageRef.current, {
   345→        rotateX,
   346→        rotateY,
   347→        duration: 0.5,
   348→        ease: 'power2.out',
   349→        transformPerspective: 1200,
   350→      })
   351→    }
   352→
   353→    const handleMouseLeave = () => {
   354→      if (!imageRef.current) return
   355→      gsap.to(imageRef.current, {
   356→        rotateX: 0,
   357→        rotateY: 0,
   358→        duration: 0.8,
   359→        ease: 'power2.out',
   360→      })
   361→    }
   362→
   363→    window.addEventListener('mousemove', handleMouseMove)
   364→    imageRef.current?.addEventListener('mouseleave', handleMouseLeave)
   365→
   366→    return () => {
   367→      ctx.revert()
   368→      window.removeEventListener('mousemove', handleMouseMove)
   369→    }
   370→  }, [])
   371→
   372→  // Typing effect
   373→  useEffect(() => {
   374→    if (!typingRef.current) return
   375→    const phrases = [
   376→      "Corriger 100 copies en 5 minutes",
   377→      "Generer des examens avec l'IA",
   378→      "Analyser les resultats en temps reel",
   379→      "Securiser chaque epreuve",
   380→    ]
   381→    let phraseIndex = 0
   382→    let charIndex = 0
   383→    let isDeleting = false
   384→    let timeout: ReturnType<typeof setTimeout>
   385→
   386→    const type = () => {
   387→      const currentPhrase = phrases[phraseIndex]
   388→      if (!isDeleting) {
   389→        charIndex++
   390→        if (typingRef.current) {
   391→          typingRef.current.textContent = currentPhrase.slice(0, charIndex)
   392→        }
   393→        if (charIndex === currentPhrase.length) {
   394→          isDeleting = true
   395→          timeout = setTimeout(type, 2000)
   396→          return
   397→        }
   398→        timeout = setTimeout(type, 60)
   399→      } else {
   400→        charIndex--
   401→        if (typingRef.current) {
   402→          typingRef.current.textContent = currentPhrase.slice(0, charIndex)
   403→        }
   404→        if (charIndex === 0) {
   405→          isDeleting = false
   406→          phraseIndex = (phraseIndex + 1) % phrases.length
   407→          timeout = setTimeout(type, 500)
   408→          return
   409→        }
   410→        timeout = setTimeout(type, 30)
   411→      }
   412→    }
   413→
   414→    timeout = setTimeout(type, 2000)
   415→    return () => clearTimeout(timeout)
   416→  }, [])
   417→
   418→  const headlineWords = ["L'examen", 'reinvente', 'par', "l'Intelligence", 'Artificielle']
   419→
   420→  return (
   421→    <section ref={sectionRef} className="relative min-h-screen flex flex-col items-center justify-center overflow-hidden bg-[#09090b]">
   422→      <DotGrid />
   423→      <GlowOrb x="50%" y="30%" color="emerald" />
   424→      <GlowOrb x="30%" y="60%" color="cyan" />
   425→
   426→      <div className="relative z-10 max-w-6xl mx-auto px-6 pt-32 pb-20 text-center">
   427→        {/* Badge */}
   428→        <motion.div
   429→          initial={{ opacity: 0, y: -10 }}
   430→          animate={{ opacity: 1, y: 0 }}
   431→          transition={{ delay: 0.1, duration: 0.6 }}
   432→          className="inline-flex items-center gap-2 px-4 py-1.5 mb-8 rounded-full border border-white/[0.08] bg-white/[0.03] backdrop-blur-sm"
   433→        >
   434→          <Sparkles className="h-3.5 w-3.5 text-emerald-400" />
   435→          <span className="text-xs text-zinc-400 font-medium tracking-wide">
   436→            Nouveau : IA Generative integree
   437→          </span>
   438→        </motion.div>
   439→
   440→        {/* Headline */}
   441→        <h1
   442→          ref={headlineRef}
   443→          className="text-5xl sm:text-6xl md:text-7xl lg:text-8xl font-bold tracking-tight leading-[0.95] mb-6"
   444→          style={{ perspective: '600px' }}

   461→        {/* Typing line */}
   462→        <p ref={subRef} className="text-lg sm:text-xl text-zinc-400 mb-10 min-h-[32px]">
