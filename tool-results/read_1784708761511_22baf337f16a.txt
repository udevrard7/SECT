   170→   170→function SliderRow({
   171→   171→  id,
   172→   172→  label,
   173→   173→  description,
   174→   174→  value,
   175→   175→  onValueChange,
   176→   176→  min,
   177→   177→  max,
   178→   178→  step = 1,
   179→   179→  formatValue,
   180→   180→  icon: Icon,
   181→   181→}: {
   182→   182→  id: string
   183→   183→  label: string
   184→   184→  description: string
   185→   185→  value: number
   186→   186→  onValueChange: (value: number) => void
   187→   187→  min: number
   188→   188→  max: number
   189→   189→  step?: number
   190→   190→  formatValue?: (value: number) => string
   191→   191→  icon: React.ComponentType<{ className?: string }>
   192→   192→}) {
   193→   193→  const displayValue = formatValue ? formatValue(value) : String(value)
   194→   194→
   195→   195→  return (
   196→   196→    <div className="rounded-lg border p-4 space-y-3">
   197→   197→      <div className="flex items-center gap-3">
   198→   198→        <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-md bg-info/10 text-info">
   199→   199→          <Icon className="h-4 w-4" />

   201→   201→        <div className="flex-1 min-w-0">
   202→   202→          <div className="flex items-center justify-between gap-2">
   203→   203→            <Label htmlFor={id} className="text-sm font-medium">
   204→   204→              {label}
   205→   205→            </Label>
   206→   206→            <Badge variant="secondary" className="bg-info/10 text-info font-mono tabular-nums text-xs">
   207→   207→              {displayValue}
   208→   208→            </Badge>
   209→   209→          </div>
   210→   210→          <p className="text-xs text-muted-foreground mt-0.5">
   211→   211→            {description}
   212→   212→          </p>
   213→   213→        </div>
   214→   214→      </div>
   215→   215→      <Slider
   216→   216→        id={id}
   217→   217→        value={[value]}
   218→   218→        onValueChange={([val]) => onValueChange(val)}
   219→   219→        min={min}
   220→   220→        max={max}
   221→   221→        step={step}
   222→   222→        className="w-full"
   223→   223→      />
   224→   224→      <div className="flex justify-between text-xs text-muted-foreground font-mono tabular-nums">
   225→   225→        <span>{formatValue ? formatValue(min) : min}</span>
   226→   226→        <span>{formatValue ? formatValue(max) : max}</span>
   227→   227→      </div>
   228→   228→    </div>
   229→   229→  )
   230→   230→}

   251→   251→  // useQuery. Le loadSettings réactif à selectedEtablissementId reste en
   252→   252→  // useEffect (pattern différent : pas un mount-time fetch).

   254→   254→    queryKey: ['etablissements'],
