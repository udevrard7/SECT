  1004→func (s *Server) corbeilleRestore(w http.ResponseWriter, r *http.Request) {

  1010→
  1011→        var body struct {
  1012→                Items []struct {
  1013→                        ID   string `json:"id"`
  1014→                        Type string `json:"type"` // document | question | epreuve | devoir
  1015→                } `json:"items"`
  1016→        }
  1017→        if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
  1018→                writeJSONError(w, http.StatusBadRequest, "JSON invalide")
  1019→                return
  1020→        }
  1021→        if len(body.Items) == 0 {
  1022→                writeJSONError(w, http.StatusBadRequest, "items requis")
  1023→                return
  1024→        }
  1025→
  1026→        restored := 0
  1027→        var errors []string
  1028→        txErr := appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
  1029→                for _, item := range body.Items {
  1030→                        var tableName string

  1032→                        case "epreuve":

  1034→                        case "document":
  1035→                                tableName = "Document"

  1043→                        }

  1052→                                restored++
  1053→                        }
  1054→                }
  1055→                return nil
  1056→        })

  1058→                writeJSONError(w, http.StatusInternalServerError, "erreur interne")
  1059→                return
  1060→        }
  1061→
  1062→        // C4 : code HTTP selon le résultat.
  1063→        // - 200 si tout OK (restored == len(items))
  1064→        // - 207 Multi-Status si partiel (0 < restored < len(items))

  1068→                statusCode = http.StatusConflict
  1069→        } else if restored < len(body.Items) {
  1070→                statusCode = http.StatusMultiStatus
  1071→        }
  1072→
  1073→        w.Header().Set("Content-Type", "application/json")
  1074→        w.WriteHeader(statusCode)

  1093→func (s *Server) corbeillePurge(w http.ResponseWriter, r *http.Request) {
  1094→        claims, ok := middleware.ClaimsFromContext(r.Context())
  1095→        if !ok || claims.UserID == "" {
  1096→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
  1097→                return
  1098→        }
  1099→
  1100→        var body struct {
  1101→                Items []struct {
  1102→                        ID   string `json:"id"`
  1103→                        Type string `json:"type"` // document | question | epreuve | devoir
  1104→                } `json:"items"`
  1105→        }
  1106→        if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
  1107→                writeJSONError(w, http.StatusBadRequest, "JSON invalide")
  1108→                return
  1109→        }
  1110→        if len(body.Items) == 0 {
  1111→                writeJSONError(w, http.StatusBadRequest, "items requis")
  1112→                return
  1113→        }
  1114→
  1115→        purged := 0
  1116→        var errors []string
  1117→        // C5 : collecter les chemins R2 des documents à purger pour suppression post-commit.
  1118→        var r2KeysToDelete []string
  1119→        var r2DocIDs []string
  1120→
  1121→        txErr := appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
  1122→                for _, item := range body.Items {
  1123→                        var tableName string
  1124→                        switch item.Type {
  1125→                        case "epreuve":
  1126→                                tableName = "Epreuve"
  1127→                        case "document":
  1128→                                tableName = "Document"
  1129→                        case "question":
  1130→                                tableName = "Question"
  1131→                        case "devoir":
  1132→                                tableName = "Devoir"
  1133→                        default:
  1134→                                errors = append(errors, "type inconnu: "+item.Type)
  1135→                                continue
  1136→                        }
  1137→
  1138→                        // C3 : vérifier FK RESTRICT avant DELETE pour Epreuve et Question.
  1139→                        if item.Type == "epreuve" {
  1140→                                var sessionCount int
  1141→                                err := tx.QueryRow(r.Context(), `SELECT count(*) FROM "SessionPassation" WHERE "epreuveId" = $1`, item.ID).Scan(&sessionCount)
  1142→                                if err != nil {
  1143→                                        errors = append(errors, fmt.Sprintf("purge epreuve %s: vérif sessions: %v", item.ID, err))
  1144→                                        continue
  1145→                                }
  1146→                                if sessionCount > 0 {
  1147→                                        errors = append(errors, fmt.Sprintf("purge epreuve %s impossible: %d session(s) de passation liée(s) — supprimez d'abord les sessions", item.ID, sessionCount))
  1148→                                        continue
  1149→                                }
  1150→                        }
  1151→                        if item.Type == "question" {
  1152→                                var eqCount int
  1153→                                err := tx.QueryRow(r.Context(), `SELECT count(*) FROM "EpreuveQuestion" WHERE "questionId" = $1`, item.ID).Scan(&eqCount)
  1154→                                if err != nil {
  1155→                                        errors = append(errors, fmt.Sprintf("purge question %s: vérif epreuveQuestion: %v", item.ID, err))
  1156→                                        continue
  1157→                                }
  1158→                                if eqCount > 0 {
  1159→                                        errors = append(errors, fmt.Sprintf("purge question %s impossible: utilisée dans %d épreuve(s) — retirez-la des épreuves d'abord", item.ID, eqCount))
  1160→                                        continue
  1161→                                }
  1162→                        }
  1163→
  1164→                        // C5 : pour Document, récupérer le chemin R2 avant DELETE.
  1165→                        if item.Type == "document" {
  1166→                                var cheminStockage string
  1167→                                err := tx.QueryRow(r.Context(), `SELECT "cheminStockage" FROM "Document" WHERE "id" = $1 AND "deletedAt" IS NOT NULL`, item.ID).Scan(&cheminStockage)
  1168→                                if err == nil && cheminStockage != "" {
  1169→                                        r2KeysToDelete = append(r2KeysToDelete, cheminStockage)
  1170→                                        r2DocIDs = append(r2DocIDs, item.ID)
  1171→                                }
  1172→                        }
  1173→
  1174→                        // Hard delete. SAFETY: tableName issu d'un switch contrôlé.
  1175→                        query := fmt.Sprintf(`DELETE FROM "%s" WHERE "id" = $1 AND "deletedAt" IS NOT NULL`, tableName)
  1176→                        tag, err := tx.Exec(r.Context(), query, item.ID)
  1177→                        if err != nil {
  1178→                                errors = append(errors, fmt.Sprintf("purge %s %s: %v", item.Type, item.ID, err))
  1179→                                continue
  1180→                        }
  1181→                        if tag.RowsAffected() > 0 {
  1182→                                purged++
  1183→                        }
  1184→                }
  1185→                return nil
  1186→        })
  1187→        if txErr != nil {
  1188→                writeJSONError(w, http.StatusInternalServerError, "erreur interne")
  1189→                return
  1190→        }
  1191→
  1192→        // C5 : suppression best-effort des objets R2 (post-commit, hors transaction).
  1193→        // On utilise le storage du DocumentUseCase si disponible.
  1194→        if len(r2KeysToDelete) > 0 && s.documentUC != nil {
  1195→                ctx := r.Context()
  1196→                for i, key := range r2KeysToDelete {
  1197→                        _ = s.documentUC.PurgeR2Object(ctx, key) // best-effort, ignoré si échec
  1198→                        _ = r2DocIDs[i]                          // docID conservé pour logging futur
  1199→                }
  1200→        }
  1201→
  1202→        // C4 : code HTTP selon le résultat.
  1203→        statusCode := http.StatusOK
  1204→        if purged == 0 {
  1205→                statusCode = http.StatusConflict
  1206→        } else if purged < len(body.Items) {
  1207→                statusCode = http.StatusMultiStatus
  1208→        }
  1209→
  1210→        w.Header().Set("Content-Type", "application/json")
  1211→        w.WriteHeader(statusCode)
  1212→        json.NewEncoder(w).Encode(map[string]any{
  1213→                "message": fmt.Sprintf("%d élément(s) supprimé(s) définitivement", purged),
  1214→                "purged":  purged,
  1215→                "errors":  errors,
  1216→        })
  1217→}

  1421→                        d.CreatedAt = createdAt.UTC().Format(time.RFC3339)

  1423→                        if ueNiveau != "" {
  1424→                                d.UniteEnseignement.Niveau = ueNiveau
  1425→                        }
  1426→                        if grilleID != nil && grilleCriteres != nil {

  1430→                        // Soumission de l'étudiant (si applicable)

  1466→// 9. GET /api/devoirs/stats — Devoir stats
  1467→// ──────────────────────────────────────────────────────────────────────────
  1468→
  1469→// BUGFIX (DEVOIRS-ANALYSE-1) : l'ancien endpoint retournait {total, enCours,
  1470→// corriges} (flat) mais le frontend AnalysisView attend {kpis:{...}, byType:[],
  1471→// soumissionsByStatut:[], timeline:[], moyenneNotes} → stats.kpis.total →
  1472→// TypeError → crash de l'onglet Analyses.
  1473→// Maintenant : retourne le contrat structuré attendu par DevoirStats (TS).
  1474→func (s *Server) devoirsStatsReal(w http.ResponseWriter, r *http.Request) {
  1475→        claims, ok := middleware.ClaimsFromContext(r.Context())
  1476→        if !ok || claims.UserID == "" {
  1477→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
  1478→                return
  1479→        }
  1480→
  1481→        type kpisT struct {
  1482→                Total                int `json:"total"`
  1483→                Brouillons           int `json:"brouillons"`
  1484→                Publies              int `json:"publies"`
  1485→                Fermes               int `json:"fermes"`
  1486→                Archives             int `json:"archives"`
  1487→                TotalSoumissions     int `json:"totalSoumissions"`
  1488→                SoumissionsEnAttente int `json:"soumissionsEnAttente"`
  1489→                SoumissionsCorrigees int `json:"soumissionsCorrigees"`
  1490→                EnRetard             int `json:"enRetard"`
  1491→        }
  1492→        type byTypeT struct {
  1493→                Type  string `json:"type"`
  1494→                Count int    `json:"count"`
  1495→                Label string `json:"label"`
  1496→        }
  1497→        type soumByStatutT struct {
  1498→                Statut string `json:"statut"`
  1499→                Count  int    `json:"count"`
  1500→                Label  string `json:"label"`
  1501→        }
  1502→        type timelineT struct {
  1503→                Date         string `json:"date"`
  1504→                Soumissions  int    `json:"soumissions"`
  1505→        }
  1506→
  1507→        stats := struct {
  1508→                Kpis                kpisT          `json:"kpis"`
  1509→                ByType              []byTypeT      `json:"byType"`
  1510→                SoumissionsByStatut []soumByStatutT `json:"soumissionsByStatut"`
  1511→                Timeline            []timelineT    `json:"timeline"`
  1512→                MoyenneNotes        *float64       `json:"moyenneNotes"`
  1513→        }{
  1514→                ByType:              []byTypeT{},
  1515→                SoumissionsByStatut: []soumByStatutT{},
  1516→                Timeline:            []timelineT{},
  1517→                MoyenneNotes:        nil,
  1518→        }
  1519→
  1520→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
  1521→                // KPIs : compteurs par statut
  1522→                _ = tx.QueryRow(r.Context(), `SELECT count(*) FROM "Devoir" WHERE "deletedAt" IS NULL`).Scan(&stats.Kpis.Total)
  1523→                _ = tx.QueryRow(r.Context(), `SELECT count(*) FROM "Devoir" WHERE "deletedAt" IS NULL AND "statut"::text = 'BROUILLON'`).Scan(&stats.Kpis.Brouillons)
  1524→                _ = tx.QueryRow(r.Context(), `SELECT count(*) FROM "Devoir" WHERE "deletedAt" IS NULL AND "statut"::text = 'PUBLIE'`).Scan(&stats.Kpis.Publies)
  1525→                _ = tx.QueryRow(r.Context(), `SELECT count(*) FROM "Devoir" WHERE "deletedAt" IS NULL AND "statut"::text = 'FERME'`).Scan(&stats.Kpis.Fermes)
  1526→                _ = tx.QueryRow(r.Context(), `SELECT count(*) FROM "Devoir" WHERE "deletedAt" IS NULL AND "statut"::text = 'ARCHIVE'`).Scan(&stats.Kpis.Archives)
  1527→
  1528→                // Soumissions
  1529→                // BUGFIX (DEVOIRS-STATS-FIX-1) : exclure les soumissions des devoirs soft-supprimés
  1530→                _ = tx.QueryRow(r.Context(), `SELECT count(*) FROM "Soumission" s JOIN "Devoir" d ON d."id" = s."devoirId" WHERE d."deletedAt" IS NULL`).Scan(&stats.Kpis.TotalSoumissions)
  1531→                _ = tx.QueryRow(r.Context(), `SELECT count(*) FROM "Soumission" s JOIN "Devoir" d ON d."id" = s."devoirId" WHERE d."deletedAt" IS NULL AND s."statut"::text = 'SOUMIS'`).Scan(&stats.Kpis.SoumissionsEnAttente)
  1532→                _ = tx.QueryRow(r.Context(), `SELECT count(*) FROM "Soumission" s JOIN "Devoir" d ON d."id" = s."devoirId" WHERE d."deletedAt" IS NULL AND s."statut"::text IN ('CORRIGE','RETOURNE')`).Scan(&stats.Kpis.SoumissionsCorrigees)
  1533→                _ = tx.QueryRow(r.Context(), `SELECT count(*) FROM "Soumission" s JOIN "Devoir" d ON d."id" = s."devoirId" WHERE d."deletedAt" IS NULL AND s."renduAt" IS NULL AND d."dateLimite" < CURRENT_TIMESTAMP`).Scan(&stats.Kpis.EnRetard)
  1534→
  1535→                // byType : répartition par type de séance
  1536→                rows, err := tx.Query(r.Context(), `SELECT "typeSeance"::text, count(*) FROM "Devoir" WHERE "deletedAt" IS NULL GROUP BY "typeSeance" ORDER BY count(*) DESC`)
  1537→                if err == nil {
  1538→                        defer rows.Close()
  1539→                        typeLabels := map[string]string{"CM": "Cours magistral", "TD": "Travail dirigé", "TP": "Travaux pratiques"}
  1540→                        for rows.Next() {
  1541→                                var t string
  1542→                                var c int
  1543→                                if err := rows.Scan(&t, &c); err == nil {
  1544→                                        stats.ByType = append(stats.ByType, byTypeT{Type: t, Count: c, Label: typeLabels[t]})
  1545→                                }
  1546→                        }
  1547→                }
  1548→
  1549→                // soumissionsByStatut
  1550→                rows2, err := tx.Query(r.Context(), `SELECT s."statut"::text, count(*) FROM "Soumission" s JOIN "Devoir" d ON d."id" = s."devoirId" WHERE d."deletedAt" IS NULL GROUP BY s."statut" ORDER BY count(*) DESC`)
  1551→                if err == nil {
  1552→                        defer rows2.Close()
  1553→                statutLabels := map[string]string{"BROUILLON": "Brouillon", "SOUMIS": "Soumis", "CORRIGE": "Corrigé", "RETOURNE": "Rendu"}
  1554→                        for rows2.Next() {
  1555→                                var st string
  1556→                                var c int
  1557→                                if err := rows2.Scan(&st, &c); err == nil {
  1558→                                        stats.SoumissionsByStatut = append(stats.SoumissionsByStatut, soumByStatutT{Statut: st, Count: c, Label: statutLabels[st]})
  1559→                                }
  1560→                        }
  1561→                }
  1562→
  1563→                // timeline : 7 derniers jours
  1564→                rows3, err := tx.Query(r.Context(), `
  1565→                        SELECT to_char(d::date, 'YYYY-MM-DD') as date,
  1566→                        (SELECT count(*) FROM "Soumission" s JOIN "Devoir" dv ON dv."id" = s."devoirId" WHERE dv."deletedAt" IS NULL AND date_trunc('day', s."createdAt") = d) as soumissions
  1567→                        FROM generate_series(CURRENT_DATE - INTERVAL '6 days', CURRENT_DATE, INTERVAL '1 day') AS d
  1568→                        ORDER BY d ASC
  1569→                `)
  1570→                if err == nil {
  1571→                        defer rows3.Close()
  1572→                        for rows3.Next() {
  1573→                                var dt string
  1574→                                var sc int
  1575→                                if err := rows3.Scan(&dt, &sc); err == nil {
  1576→                                        stats.Timeline = append(stats.Timeline, timelineT{Date: dt, Soumissions: sc})
  1577→                                }
  1578→                        }
  1579→                }
  1580→
  1581→                // moyenneNotes
  1582→                var moy *float64
  1583→                var moyVal float64
  1584→                err = tx.QueryRow(r.Context(), `SELECT AVG(s."note") FROM "Soumission" s JOIN "Devoir" d ON d."id" = s."devoirId" WHERE d."deletedAt" IS NULL AND s."note" IS NOT NULL`).Scan(&moyVal)
  1585→                if err == nil {
  1586→                        moy = &moyVal
  1587→                        stats.MoyenneNotes = moy
  1588→                }
  1589→                return nil
  1590→        })
  1591→
  1592→        w.Header().Set("Content-Type", "application/json")
  1593→        json.NewEncoder(w).Encode(stats)
  1594→}
