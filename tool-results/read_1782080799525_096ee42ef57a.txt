     1→'use client'
     2→
     3→import { useState, useEffect, useCallback } from 'react'
     4→import { motion, AnimatePresence } from 'framer-motion'
     5→import {
     6→  Download,
     7→  Award,
     8→  Shield,
     9→  Star,
    10→  ExternalLink,
    11→  FileText,
    12→  CheckCircle2,

    17→  AlertCircle,
    18→} from 'lucide-react'
    19→import { useAuthStore } from '@/stores/auth-store'
    20→import {
    21→  Card,
    22→  CardContent,
    23→  CardHeader,
    24→  CardTitle,
    25→} from '@/components/ui/card'
    26→import { Button } from '@/components/ui/button'
    27→import { Badge } from '@/components/ui/badge'
    28→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    29→import { Progress } from '@/components/ui/progress'
    30→import { toast } from 'sonner'
    31→
    32→// ─── Types ───
    33→
    34→type CertificatType = 'STANDARD' | 'AVANCE' | 'EXPERT'
    35→
    36→interface Certificat {
    37→  id: string
    38→  type: CertificatType
    39→  ueCode: string
    40→  ueNom: string
    41→  note: number
    42→  mention: string
    43→  dateEmission: string
    44→  verificationUrl?: string
    45→}
    46→
    47→type StatutUE = 'EN_COURS' | 'VALIDEE' | 'NON_VALIDEE'
    48→
    49→interface ValidationUE {
    50→  id: string
    51→  ueCode: string
    52→  ueNom: string
    53→  creditsECTS: number
    54→  epreuvesCompletees: number
    55→  epreuvesTotal: number
    56→  note: number | null
    57→  statut: StatutUE
    58→  certificatId: string | null
    59→}
    60→
    61→// ─── Constants ───
    62→

    64→  CertificatType,
    65→  { label: string; color: string; bgClass: string; borderClass: string; badgeClass: string; icon: typeof Award }
    66→> = {
    67→  STANDARD: {
    68→    label: 'Standard',
    69→    color: 'text-emerald-600',
    70→    bgClass: 'bg-emerald-50',
    71→    borderClass: 'border-emerald-200',
    72→    badgeClass: 'bg-emerald-100 text-emerald-700',
    73→    icon: FileText,
    74→  },
    75→  AVANCE: {
    76→    label: 'Avancé',
    77→    color: 'text-blue-600',
    78→    bgClass: 'bg-blue-50',
    79→    borderClass: 'border-blue-200',
    80→    badgeClass: 'bg-blue-100 text-blue-700',
    81→    icon: Award,
    82→  },
    83→  EXPERT: {
    84→    label: 'Expert',
    85→    color: 'text-amber-600',
    86→    bgClass: 'bg-amber-50',
    87→    borderClass: 'border-amber-200',
    88→    badgeClass: 'bg-amber-100 text-amber-700',
    89→    icon: Star,
    90→  },
    91→}
    92→
    93→const STATUT_CONFIG: Record<
    94→  StatutUE,
    95→  { label: string; icon: typeof Clock; badgeClass: string }
    96→> = {
    97→  EN_COURS: {
    98→    label: 'En cours',
    99→    icon: Clock,
   100→    badgeClass: 'bg-yellow-100 text-yellow-800',
   101→  },
   102→  VALIDEE: {
   103→    label: 'Validée',
   104→    icon: CheckCircle2,
   105→    badgeClass: 'bg-green-100 text-green-800',
   106→  },
   107→  NON_VALIDEE: {
   108→    label: 'Non validée',
   109→    icon: XCircle,
   110→    badgeClass: 'bg-red-100 text-red-800',
   111→  },
   112→}
   113→
   114→// ─── Animation variants ───
   115→
   116→const containerVariants = {
   117→  hidden: { opacity: 0 },
   118→  visible: {
   119→    opacity: 1,
   120→    transition: { staggerChildren: 0.06 },
   121→  },
   122→}
   123→
   124→const itemVariants = {
   125→  hidden: { opacity: 0, y: 16 },
   126→  visible: { opacity: 1, y: 0, transition: { duration: 0.35, ease: 'easeOut' } },
   127→}

   129→// ─── Component ───
   130→
   131→export function MesCertificatsPage() {
   132→  const { user } = useAuthStore()
   133→
   134→  const [certificats, setCertificats] = useState<Certificat[]>([])
   135→  const [validations, setValidations] = useState<ValidationUE[]>([])
   136→  const [loading, setLoading] = useState(true)
   137→  const [error, setError] = useState<string | null>(null)
   138→  const [downloadingId, setDownloadingId] = useState<string | null>(null)
   139→  const [pdfOrientation, setPdfOrientation] = useState<'landscape' | 'portrait'>('landscape')
   140→  const [activeTab, setActiveTab] = useState('certificats')
   141→
   142→  // ─── Data fetching ───
   143→
   144→  const fetchData = useCallback(async () => {
   145→    if (!user?.id) return
   146→    setLoading(true)
   147→    setError(null)
   148→
   149→    try {
   150→      // 1. Trigger recalculation (computes validations + generates certificates)
   151→      await fetch('/api/validations-ue', {
   152→        method: 'POST',
   153→        headers: { 'Content-Type': 'application/json' },
   154→        body: JSON.stringify({}),
   155→      }).catch(() => {
   156→        // Silently continue if compute endpoint is not available
   157→      })
   158→
   159→      // 2. Fetch certificates
   160→      const certRes = await fetch('/api/certificats')
   161→      let certs: Certificat[] = []
   162→      if (certRes.ok) {
   163→        const certData = await certRes.json()
   164→        const rawCerts: Record<string, unknown>[] = Array.isArray(certData)
   165→          ? certData
   166→          : certData.certificats ?? []
   167→        // Map API (DB fields) -> frontend interface.
   168→        // API returns noteFinale/mention/ueCode/ueNom/codeVerification directly on the Certificat row.
   169→        const validTypes: CertificatType[] = ['STANDARD', 'AVANCE', 'EXPERT']
   170→        certs = rawCerts
   171→          .filter((c) => c && typeof c === 'object')
   172→          .map((c) => {
   173→            const type = validTypes.includes(c.type as CertificatType)
   174→              ? (c.type as CertificatType)
   175→              : 'STANDARD'
   176→            const codeVerification = (c.codeVerification as string) || ''
   177→            return {
   178→              id: String(c.id ?? ''),
   179→              type,
   180→              ueCode: String(c.ueCode ?? '—'),
   181→              ueNom: String(c.ueNom ?? '—'),
   182→              note: typeof c.noteFinale === 'number' ? c.noteFinale : 0,
   183→              mention: typeof c.mention === 'string' ? c.mention : '',
   184→              dateEmission: c.dateEmission ? String(c.dateEmission) : new Date().toISOString(),
   185→              verificationUrl: codeVerification
   186→                ? `${window.location.origin}/verify/${codeVerification}`
   187→                : undefined,
   188→            }
   189→          })
   190→      }
   191→
   192→      // 3. Fetch UE validations
   193→      const valRes = await fetch('/api/validations-ue')
   194→      let vals: ValidationUE[] = []
   195→      if (valRes.ok) {
   196→        const valData = await valRes.json()
   197→        const rawVals: Record<string, unknown>[] = Array.isArray(valData)
   198→          ? valData
   199→          : valData.validations ?? []
   200→        // Map API (DB fields + nested uniteEnseignement/certificats) -> frontend interface.
   201→        const validStatuts: StatutUE[] = ['EN_COURS', 'VALIDEE', 'NON_VALIDEE']
   202→        vals = rawVals
   203→          .filter((v) => v && typeof v === 'object')
   204→          .map((v) => {
   205→            const ue = (v.uniteEnseignement as Record<string, unknown> | null) ?? null
   206→            const certificats = Array.isArray(v.certificats) ? (v.certificats as Record<string, unknown>[]) : []
   207→            const firstCertId = certificats.length > 0 ? String(certificats[0].id ?? '') : null
   208→            const statut = validStatuts.includes(v.statut as StatutUE)
   209→              ? (v.statut as StatutUE)
   210→              : 'EN_COURS'
   211→            const noteFinale = typeof v.noteFinale === 'number' ? v.noteFinale : null
   212→            // noteFinale defaults to 0 in DB even when EN_COURS (no sessions yet);
   213→            // expose null to the UI in that case so it shows "—" instead of "0.0".
   214→            const note = statut === 'EN_COURS' ? null : noteFinale
   215→            return {
   216→              id: String(v.id ?? ''),
   217→              ueCode: String(ue?.code ?? '—'),
   218→              ueNom: String(ue?.nom ?? '—'),
   219→              creditsECTS: typeof ue?.creditsECTS === 'number' ? ue.creditsECTS : 0,
   220→              epreuvesCompletees: typeof v.nbEpreuvesCompletees === 'number' ? v.nbEpreuvesCompletees : 0,
   221→              epreuvesTotal: typeof v.nbEpreuvesTotal === 'number' ? v.nbEpreuvesTotal : 0,
   222→              note,
   223→              statut,
   224→              certificatId: firstCertId,
   225→            }
   226→          })
   227→      }
   228→
   229→      setCertificats(certs)
   230→      setValidations(vals)
   231→    } catch (err) {
   232→      console.error('Error fetching certificate data:', err)
   233→      setError('Impossible de charger vos certificats. Veuillez réessayer.')
   234→    } finally {
   235→      setLoading(false)
   236→    }
   237→  }, [user?.id])
   238→
   239→  useEffect(() => {
   240→    fetchData()
   241→  }, [fetchData])
   242→
   243→  // ─── Summary stats ───
   244→
   245→  const stats = {
   246→    total: certificats.length,
   247→    accomplissements: certificats.filter((c) => c.note >= 10).length,
   248→    excellence: certificats.filter((c) => c.note >= 16).length,
   249→    participations: certificats.filter((c) => c.note < 10).length,
   250→  }
   251→
   252→  // ─── PDF download ───
   253→
   254→  const handleDownloadPDF = async (certificatId: string, orientation?: 'landscape' | 'portrait') => {
   255→    const orient = orientation || pdfOrientation
   256→    setDownloadingId(certificatId)
   257→    try {
   258→      const res = await fetch(`/api/certificats/${certificatId}/pdf?orientation=${orient}`)
   259→      if (!res.ok) throw new Error('Erreur lors du téléchargement')
   260→      const blob = await res.blob()
   261→      const url = URL.createObjectURL(blob)
   262→      window.open(url)
   263→      toast.success(`Certificat téléchargé (${orient === 'landscape' ? 'Paysage' : 'Portrait'})`)
   264→    } catch (err) {
   265→      console.error('PDF download error:', err)
   266→      toast.error('Impossible de télécharger le certificat')
   267→    } finally {
   268→      setDownloadingId(null)
   269→    }
   270→  }
   271→
   272→  // ─── Loading state ───
   273→
   274→  if (loading) {
   275→    return (
   276→      <div className="flex items-center justify-center min-h-[60vh]">
   277→        <div className="flex flex-col items-center gap-3">
   278→          <Loader2 className="h-8 w-8 animate-spin text-emerald-600" />
   279→          <p className="text-sm text-muted-foreground">Chargement de vos certificats...</p>
   280→        </div>
   281→      </div>
   282→    )
   283→  }
   284→
   285→  // ─── Error state ───
   286→
   287→  if (error) {
   288→    return (
   289→      <div className="flex items-center justify-center min-h-[60vh]">
   290→        <Card className="w-full max-w-md">
   291→          <CardHeader className="text-center">
   292→            <div className="mx-auto mb-2 flex h-12 w-12 items-center justify-center rounded-full bg-red-100">
   293→              <AlertCircle className="h-6 w-6 text-red-600" />
   294→            </div>
   295→            <CardTitle className="text-lg">Erreur de chargement</CardTitle>
   296→          </CardHeader>
   297→          <CardContent className="text-center">
   298→            <p className="text-sm text-muted-foreground mb-4">{error}</p>
   299→            <Button onClick={fetchData} variant="outline" className="gap-2">
   300→              <Clock className="h-4 w-4" />
   301→              Réessayer
   302→            </Button>
   303→          </CardContent>
   304→        </Card>
   305→      </div>
   306→    )
   307→  }
   308→
   309→  // ─── Render ───
   310→
   311→  return (
   312→    <motion.div
   313→      variants={containerVariants}
   314→      initial="hidden"
   315→      animate="visible"
   316→      className="space-y-6"
   317→    >
   318→      {/* ─── Header ─── */}
   319→      <motion.div variants={itemVariants} className="flex items-center justify-between">
   320→        <div className="flex items-center gap-3">
   321→          <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-emerald-100">
   322→            <ScrollText className="h-5 w-5 text-emerald-600" />
   323→          </div>
   324→          <div>
   325→            <h1 className="text-2xl font-bold tracking-tight">Mes Certificats</h1>
   326→            <p className="text-sm text-muted-foreground">
   327→              Consultez et téléchargez vos certificats
   328→            </p>
   329→          </div>
   330→        </div>
   331→        <Badge variant="secondary" className="text-sm px-3 py-1">
   332→          {stats.total} certificat{stats.total !== 1 ? 's' : ''}
   333→        </Badge>
   334→      </motion.div>
   335→
   336→      {/* ─── Summary stats row ─── */}
   337→      <motion.div variants={itemVariants} className="grid grid-cols-2 md:grid-cols-4 gap-4">
   338→        <Card>
   339→          <CardContent className="p-4">
   340→            <div className="flex items-center gap-3">
   341→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-slate-100">
   342→                <FileText className="h-4 w-4 text-slate-600" />
   343→              </div>
   344→              <div>
   345→                <p className="text-2xl font-bold">{stats.total}</p>
   346→                <p className="text-xs text-muted-foreground">Certificats émis</p>
   347→              </div>
   348→            </div>
   349→          </CardContent>
   350→        </Card>
   351→        <Card>
   352→          <CardContent className="p-4">
   353→            <div className="flex items-center gap-3">
   354→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-emerald-100">
   355→                <Award className="h-4 w-4 text-emerald-600" />
   356→              </div>
   357→              <div>
   358→                <p className="text-2xl font-bold text-emerald-600">{stats.accomplissements}</p>
   359→                <p className="text-xs text-muted-foreground">Accomplissements</p>
   360→              </div>
   361→            </div>
   362→          </CardContent>
   363→        </Card>
   364→        <Card>
   365→          <CardContent className="p-4">
   366→            <div className="flex items-center gap-3">
   367→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-amber-100">
   368→                <Star className="h-4 w-4 text-amber-600" />
   369→              </div>
   370→              <div>
   371→                <p className="text-2xl font-bold text-amber-600">{stats.excellence}</p>
   372→                <p className="text-xs text-muted-foreground">Excellence</p>
   373→              </div>
   374→            </div>
   375→          </CardContent>
   376→        </Card>
   377→        <Card>
   378→          <CardContent className="p-4">
   379→            <div className="flex items-center gap-3">
   380→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-slate-100">
   381→                <Shield className="h-4 w-4 text-slate-500" />
   382→              </div>
   383→              <div>
   384→                <p className="text-2xl font-bold text-slate-500">{stats.participations}</p>
   385→                <p className="text-xs text-muted-foreground">Participations</p>
   386→              </div>
   387→            </div>
   388→          </CardContent>
   389→        </Card>
   390→      </motion.div>
   391→
   392→      {/* ─── Tabs ─── */}
   393→      <motion.div variants={itemVariants}>
   394→        <Tabs value={activeTab} onValueChange={setActiveTab}>
   395→          <TabsList>
   396→            <TabsTrigger value="certificats" className="gap-1.5">
   397→              <Award className="h-4 w-4" />
   398→              Certificats
   399→            </TabsTrigger>
   400→            <TabsTrigger value="progression" className="gap-1.5">
   401→              <BarChart3Icon className="h-4 w-4" />
   402→              Progression par UE
   403→            </TabsTrigger>
   404→          </TabsList>
   405→
   406→          {/* ─── Certificats tab ─── */}
   407→          <TabsContent value="certificats">
   408→            {certificats.length === 0 ? (
   409→              <Card>
   410→                <CardContent className="flex flex-col items-center justify-center py-12">
   411→                  <div className="flex h-14 w-14 items-center justify-center rounded-full bg-muted mb-4">
   412→                    <FileText className="h-7 w-7 text-muted-foreground" />
   413→                  </div>
   414→                  <p className="text-lg font-medium mb-1">Aucun certificat</p>
   415→                  <p className="text-sm text-muted-foreground text-center max-w-sm">
   416→                    Vos certificats apparaîtront ici une fois que vous aurez complété des épreuves et obtenu des résultats.
   417→                  </p>
   418→                </CardContent>
   419→              </Card>
   420→            ) : (
   421→              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   422→                <AnimatePresence mode="popLayout">
   423→                  {certificats.map((cert, index) => {
   424→                    const config = CERTIFICAT_CONFIG[cert.type]
   425→                    const IconComponent = config.icon
   426→
   427→                    return (
   428→                      <motion.div
   429→                        key={cert.id}
   430→                        variants={itemVariants}
   431→                        initial="hidden"
   432→                        animate="visible"
   433→                        transition={{ delay: index * 0.05 }}
   434→                      >
   435→                        <Card className={`border ${config.borderClass} hover:shadow-md transition-shadow`}>
   436→                          <CardHeader className="pb-3">
   437→                            <div className="flex items-start justify-between">
   438→                              <div className={`flex h-10 w-10 items-center justify-center rounded-lg ${config.bgClass}`}>
   439→                                <IconComponent className={`h-5 w-5 ${config.color}`} />
   440→                              </div>
   441→                              <Badge className={config.badgeClass} variant="secondary">
   442→                                {config.label}
   443→                              </Badge>
   444→                            </div>
   445→                            <CardTitle className="text-base mt-2">
   446→                              <span className="font-mono text-sm text-muted-foreground mr-2">
   447→                                {cert.ueCode}
   448→                              </span>
   449→                              {cert.ueNom}
   450→                            </CardTitle>
   451→                          </CardHeader>
   452→                          <CardContent className="space-y-4">
   453→                            {/* Note & mention */}
   454→                            <div className="flex items-center justify-between">
   455→                              <div className="flex items-center gap-2">
   456→                                <span className="text-2xl font-bold">{cert.note.toFixed(1)}</span>
   457→                                <span className="text-sm text-muted-foreground">/20</span>
   458→                              </div>
   459→                              <Badge variant="outline" className="text-xs">
   460→                                {cert.mention}
   461→                              </Badge>
   462→                            </div>
   463→
   464→                            {/* Progress bar */}
   465→                            <Progress value={(cert.note / 20) * 100} className="h-2" />
   466→
   467→                            {/* Date */}
   468→                            <p className="text-xs text-muted-foreground">
   469→                              Émis le{' '}
   470→                              {new Date(cert.dateEmission).toLocaleDateString('fr-FR', {
   471→                                day: 'numeric',
   472→                                month: 'long',
   473→                                year: 'numeric',
   474→                              })}
   475→                            </p>
   476→
   477→                            {/* Format selector + Download */}
   478→                            <div className="flex flex-col gap-2 pt-1">
   479→                              {/* Toggle Paysage/Portrait */}
   480→                              <div className="flex gap-1 bg-muted/50 rounded-md p-0.5">
   481→                                <button
   482→                                  onClick={() => setPdfOrientation('landscape')}
   483→                                  className={`flex-1 px-2 py-1 rounded text-xs font-medium transition-all ${
   484→                                    pdfOrientation === 'landscape'
   485→                                      ? 'bg-background shadow-sm text-emerald-600'
   486→                                      : 'text-muted-foreground hover:text-foreground'
   487→                                  }`}
   488→                                >
   489→                                  📐 Paysage
   490→                                </button>
   491→                                <button
   492→                                  onClick={() => setPdfOrientation('portrait')}
   493→                                  className={`flex-1 px-2 py-1 rounded text-xs font-medium transition-all ${
   494→                                    pdfOrientation === 'portrait'
   495→                                      ? 'bg-background shadow-sm text-emerald-600'
   496→                                      : 'text-muted-foreground hover:text-foreground'
   497→                                  }`}
   498→                                >
   499→                                  📄 Portrait
   500→                                </button>
   501→                              </div>
   502→                              <Button
   503→                                size="sm"
   504→                                className="flex-1 gap-1.5"
   505→                                onClick={() => handleDownloadPDF(cert.id)}
   506→                                disabled={downloadingId === cert.id}
   507→                              >
   508→                                {downloadingId === cert.id ? (
   509→                                  <Loader2 className="h-3.5 w-3.5 animate-spin" />
   510→                                ) : (
   511→                                  <Download className="h-3.5 w-3.5" />
   512→                                )}
   513→                                Télécharger PDF ({pdfOrientation === 'landscape' ? 'Paysage' : 'Portrait'})
   514→                              </Button>
   515→                              {cert.verificationUrl && (
   516→                                <Button
   517→                                  size="sm"
   518→                                  variant="outline"
   519→                                  className="gap-1.5"
   520→                                  asChild
   521→                                >
   522→                                  <a
   523→                                    href={cert.verificationUrl}
   524→                                    target="_blank"
   525→                                    rel="noopener noreferrer"
   526→                                  >
   527→                                    <Shield className="h-3.5 w-3.5" />
   528→                                    Vérifier
   529→                                  </a>
   530→                                </Button>
   531→                              )}
   532→                            </div>
   533→                          </CardContent>
   534→                        </Card>
   535→                      </motion.div>
   536→                    )
   537→                  })}
   538→                </AnimatePresence>
   539→              </div>
   540→            )}
   541→          </TabsContent>
   542→
   543→          {/* ─── Progression par UE tab ─── */}
   544→          <TabsContent value="progression">
   545→            {validations.length === 0 ? (
   546→              <Card>
