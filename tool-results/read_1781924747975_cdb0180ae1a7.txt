   131→   131→function getScoreColor(score: number, maxScore: number = 20): string {
   132→   132→  const halfMax = maxScore / 2
   133→   133→  if (score >= halfMax) return '#10b981'
   134→   134→  if (score >= halfMax * 0.8) return '#f59e0b'
   135→   135→  return '#ef4444'
   136→   136→}
   137→   137→
   138→   138→function getTauxColor(taux: number): string {
   139→   139→  if (taux >= 70) return '#10b981'
   140→   140→  if (taux >= 50) return '#f59e0b'
   141→   141→  return '#ef4444'
   142→   142→}

   144→   144→function formatDateFR(dateStr: string): string {
   145→   145→  try {
   146→   146→    const date = new Date(dateStr)
   147→   147→    return date.toLocaleDateString('fr-FR', {
   148→   148→      day: '2-digit',
   149→   149→      month: 'long',
   150→   150→      year: 'numeric',
   151→   151→    })
   152→   152→  } catch {
   153→   153→    return dateStr
   154→   154→  }
   155→   155→}
   156→   156→
   157→   157→function formatMonth(mois: string): string {
   158→   158→  try {
   159→   159→    const [year, month] = mois.split('-')
   160→   160→    const date = new Date(parseInt(year), parseInt(month) - 1)
   161→   161→    return date.toLocaleDateString('fr-FR', { month: 'short', year: '2-digit' })
   162→   162→  } catch {
   163→   163→    return mois
   164→   164→  }
   165→   165→}
