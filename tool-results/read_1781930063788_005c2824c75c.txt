   213→// ─── Typewriter Hook ───
   214→function useTypewriter(words: string[], typingSpeed = 80, deletingSpeed = 40, pauseTime = 2000) {
   215→  const [text, setText] = useState('')
   216→  const [wordIndex, setWordIndex] = useState(0)
   217→  const [isDeleting, setIsDeleting] = useState(false)
   218→
   219→  useEffect(() => {
   220→    const currentWord = words[wordIndex]
   221→
   222→    const timeout = setTimeout(() => {
   223→      if (!isDeleting) {
   224→        setText(currentWord.substring(0, text.length + 1))
   225→        if (text.length === currentWord.length) {
   226→          setTimeout(() => setIsDeleting(true), pauseTime)
   227→        }
   228→      } else {
   229→        setText(currentWord.substring(0, text.length - 1))
   230→        if (text.length === 0) {
   231→          setIsDeleting(false)
   232→          setWordIndex((prev) => (prev + 1) % words.length)
   233→        }
   234→      }
   235→    }, isDeleting ? deletingSpeed : typingSpeed)
   236→
   237→    return () => clearTimeout(timeout)
   238→  }, [text, isDeleting, wordIndex, words, typingSpeed, deletingSpeed, pauseTime])
   239→
   240→  return text
   241→}

   243→// ─── Glowing Feature Card ───
   244→function FeatureCard({ feature, index }: { feature: typeof features[0]; index: number }) {
   245→  const [hovered, setHovered] = useState(false)
   246→  const x = useMotionValue(0)
   247→  const y = useMotionValue(0)
   248→  const rotateX = useTransform(y, [-50, 50], [5, -5])
   249→  const rotateY = useTransform(x, [-50, 50], [-5, 5])
   250→
   251→  const handleMouse = useCallback((e: React.MouseEvent<HTMLDivElement>) => {
   252→    const rect = e.currentTarget.getBoundingClientRect()
   253→    x.set(e.clientX - rect.left - rect.width / 2)
   254→    y.set(e.clientY - rect.top - rect.height / 2)
   255→  }, [x, y])
   256→
   257→  return (
   258→    <motion.div
   259→      key={feature.label}
   260→      variants={itemVariants}
   261→      style={{ rotateX, rotateY, transformPerspective: 600 }}
   262→      onMouseMove={handleMouse}
   263→      onMouseEnter={() => setHovered(true)}
   264→      onMouseLeave={() => { setHovered(false); x.set(0); y.set(0) }}
   265→      className={`flex items-center gap-3 rounded-2xl px-4 py-3.5 border transition-all duration-300 cursor-default ${
   266→        hovered
   267→          ? 'bg-white/20 backdrop-blur-md border-white/30 shadow-lg shadow-emerald-500/10'
   268→          : 'bg-white/8 backdrop-blur-sm border-white/12 hover:bg-white/12'
   269→      }`}
   270→    >
   271→      <motion.div
   272→        className="w-10 h-10 rounded-xl bg-gradient-to-br from-white/20 to-white/5 flex items-center justify-center shrink-0 border border-white/10"
   273→        animate={hovered ? { scale: [1, 1.1, 1], rotate: [0, 5, -5, 0] } : {}}
   274→        transition={{ duration: 0.4 }}
   275→      >
   276→        <feature.icon className="w-4.5 h-4.5 text-white" />
   277→      </motion.div>
   278→      <div className="flex-1 min-w-0">
   279→        <div className="flex items-center justify-between">
   280→          <p className="text-sm font-semibold text-white">{feature.label}</p>
   281→          <motion.span
   282→            className="text-[10px] font-bold text-emerald-300 bg-emerald-400/20 px-1.5 py-0.5 rounded-full"
   283→            animate={hovered ? { scale: [1, 1.2, 1] } : {}}
   284→            transition={{ duration: 0.3 }}
   285→          >
   286→            {feature.stat}

   401→export function LoginForm({ onBack }: LoginFormProps) {
   402→  const [loginMode, setLoginMode] = useState<LoginMode>('personnel')
   403→  const [showPassword, setShowPassword] = useState(false)
   404→  const [loginError, setLoginError] = useState<string | null>(null)
   405→  const login = useAuthStore((state) => state.login)
   406→  const loginStudent = useAuthStore((state) => state.loginStudent)
   407→  const isLoading = useAuthStore((state) => state.isLoading)
   408→
   409→  // Typewriter
   410→  const typedText = useTypewriter(typewriterWords)
   411→
   412→  // Password Reset state
   413→  const [resetDialogOpen, setResetDialogOpen] = useState(false)
   414→  const [resetEmail, setResetEmail] = useState('')
   415→  const [resetSending, setResetSending] = useState(false)
   416→  const [resetSent, setResetSent] = useState(false)
   417→  const [resetToken, setResetToken] = useState<string | null>(null)
   418→
   419→  // Password Reset Confirm state
   420→  const [confirmDialogOpen, setConfirmDialogOpen] = useState(false)
   421→  const [confirmToken, setConfirmToken] = useState('')
   422→  const [confirmPassword, setConfirmPassword] = useState('')
   423→  const [confirmSubmitting, setConfirmSubmitting] = useState(false)
   424→  const [confirmSuccess, setConfirmSuccess] = useState(false)
   425→
   426→  const form = useForm({
   427→    resolver: zodResolver(loginMode === 'personnel' ? personnelSchema : etudiantSchema),
   428→    defaultValues: { identifier: '', password: '' },
   429→  })
   430→
   431→  // Reset form when mode changes
   432→  useEffect(() => {
   433→    form.reset({ identifier: '', password: '' })
   434→    setLoginError(null)
   435→  }, [loginMode, form])
   436→
   437→  const onSubmit = useCallback(async (data: { identifier: string; password: string }) => {
   438→    setLoginError(null)
   439→
   440→    try {
   441→      let success: boolean
   442→      if (loginMode === 'etudiant') {
   443→        success = await loginStudent(data.identifier, data.password)
   444→      } else {
   445→        success = await login(data.identifier, data.password)
   446→      }
   447→
   448→      if (!success) {
   449→        // This shouldn't happen anymore since we throw errors, but just in case
   450→        setLoginError('Identifiants incorrects. Veuillez réessayer.')
   451→      }
   452→    } catch (err: unknown) {
   453→      const loginErr = err as LoginError
   454→      if (loginErr?.status === 500) {
   455→        // Server error (DB connection issue, etc.) — NOT a wrong password
   456→        setLoginError('Erreur serveur. Veuillez réessayer plus tard.')
   457→        toast.error('Erreur serveur', {
   458→          description: 'Une erreur technique est survenue. Veuillez réessayer dans quelques instants.',
   459→        })
   460→      } else if (loginErr?.status === 403) {
   461→        // Account disabled
   462→        setLoginError(loginErr.message || 'Votre compte a été désactivé.')
   463→        toast.error('Accès refusé', {
   464→          description: loginErr.message,
   465→        })
   466→      } else if (loginErr?.status === 0) {
   467→        // Network error
   468→        setLoginError('Erreur de connexion. Vérifiez votre réseau.')
   469→        toast.error('Erreur réseau', {
   470→          description: 'Impossible de contacter le serveur. Vérifiez votre connexion internet.',
   471→        })
   472→      } else {
   473→        // 401 or other — wrong credentials
   474→        const errorMsg = loginMode === 'etudiant'
   475→          ? 'Matricule, email ou mot de passe incorrect.'
   476→          : 'Identifiants incorrects. Veuillez réessayer.'
   477→        setLoginError(errorMsg)
   478→        toast.error('Échec de la connexion', {
   479→          description: errorMsg,
   480→        })
   481→      }
   482→    }
   483→  }, [loginMode, login, loginStudent])
   484→
   485→  const handleModeChange = useCallback((newMode: LoginMode) => {
   486→    setLoginMode(newMode)
   487→  }, [])
   488→
   489→  const handleResetRequest = async () => {
   490→    if (!resetEmail.trim()) {
   491→      toast.error('Champ requis', { description: 'Veuillez entrer votre adresse email.' })
   492→      return
   493→    }
   494→    setResetSending(true)
   495→    try {
   496→      const res = await fetch('/api/auth/password-reset', {
   497→        method: 'POST',
   498→        headers: { 'Content-Type': 'application/json' },
   499→        body: JSON.stringify({ email: resetEmail.trim() }),
   500→      })
   501→      const data = await res.json()
   502→      setResetSent(true)
   503→      if (data.token) setResetToken(data.token)
   504→      toast.success('Demande envoyée', {
   505→        description: data.message || 'Si un compte existe avec cet email, un lien de réinitialisation a été envoyé.',
   506→      })
   507→    } catch {
   508→      toast.error('Erreur', { description: 'Impossible d\'envoyer la demande. Veuillez réessayer.' })
   509→    } finally {
   510→      setResetSending(false)
   511→    }
   512→  }
   513→
   514→  const handleResetConfirm = async () => {
   515→    if (!confirmToken.trim()) {
   516→      toast.error('Champ requis', { description: 'Veuillez entrer le token de réinitialisation.' })
   517→      return
   518→    }
   519→    if (!confirmPassword || confirmPassword.length < 6) {
   520→      toast.error('Mot de passe invalide', { description: 'Le mot de passe doit contenir au moins 6 caractères.' })
   521→      return
   522→    }
   523→    setConfirmSubmitting(true)
   524→    try {
   525→      const res = await fetch('/api/auth/password-reset/confirm', {
   526→        method: 'POST',
   527→        headers: { 'Content-Type': 'application/json' },
   528→        body: JSON.stringify({ token: confirmToken.trim(), password: confirmPassword }),
   529→      })
   530→      const data = await res.json()
   531→      if (!res.ok) throw new Error(data.error || 'Erreur lors de la réinitialisation')
   532→      setConfirmSuccess(true)
   533→      toast.success('Mot de passe réinitialisé', {
   534→        description: 'Vous pouvez maintenant vous connecter avec votre nouveau mot de passe.',
   535→      })
   536→    } catch (err) {
   537→      toast.error('Erreur', {
   538→        description: err instanceof Error ? err.message : 'Impossible de réinitialiser le mot de passe.',
   539→      })
   540→    } finally {
   541→      setConfirmSubmitting(false)
