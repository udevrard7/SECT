   137→   137→function getScoreColor(score: number, maxScore: number = 20): string {
   138→   138→  const halfMax = maxScore / 2
   139→   139→  if (score >= halfMax) return '#10b981'

   141→   141→  return '#ef4444'

   148→   148→      day: '2-digit',

   151→   151→    })
