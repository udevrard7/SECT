    32→// ──────────────────────────────────────────────────────────────────────────
    33→// B2 : PUT /api/sessions/{id} — bulk save OU log alerte
    34→// ──────────────────────────────────────────────────────────────────────────
    35→//
    36→// Le frontend passation-page.tsx envoie 2 formes de body sur cette route :
    37→//   1. { reponses: { questionId: contenu, ... } }  → bulk save (auto-save 30s)
    38→//   2. { alerte: { type, details, penalite } }      → log anti-fraude
    39→//
    40→// On dispatche selon la présence du champ "reponses" ou "alerte".
    41→
    42→func (s *Server) updateSessionBulk(w http.ResponseWriter, r *http.Request) {
    43→        claims, ok := middleware.ClaimsFromContext(r.Context())
    44→        if !ok || claims.UserID == "" {
    45→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
    46→                return
    47→        }
    48→        sessionID := chi.URLParam(r, "id")
    49→        if sessionID == "" {
    50→                writeJSONError(w, http.StatusBadRequest, "id session requis")
    51→                return
    52→        }
    53→
    54→        // Decoder le body de façon permissive (champs optionnels)
    55→        var body map[string]json.RawMessage
    56→        if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
    57→                writeJSONError(w, http.StatusBadRequest, "corps de requête invalide")
    58→                return
    59→        }
    60→
    61→        // Cas 1 : bulk save reponses
    62→        if rawReponses, ok := body["reponses"]; ok {
    63→                var reponses map[string]string
    64→                if err := json.Unmarshal(rawReponses, &reponses); err != nil {
    65→                        writeJSONError(w, http.StatusBadRequest, "format reponses invalide")
    66→                        return
    67→                }
    68→
    69→                // OPT-1 + OPT-2 : écriture via cache RAM + ownership cache.
    70→                //
    71→                // AVANT : chaque auto-save (30s) faisait 1 WithTx ownership + 1 BulkSaveReponses
    72→                // = 2 transactions DB par étudiant par save. Pour 5000 étudiants = 333 tx/s.
    73→                //
    74→                // APRÈS : l'auto-save écrit en RAM (< 1ms, 0 tx DB). Le worker goroutine
    75→                // synchronise vers Neon toutes les 30s (1 tx par session). L'ownership est
    76→                // vérifiée en cache après la première vérification DB.
    77→                //
    78→                // Pour 5000 étudiants : 5000 × 1 tx / 30s = 167 tx/s au lieu de 333 tx/s (-50%).
    79→                // Avec OPT-3 (batch flush) : ~10 tx/s au lieu de 167 tx/s (-95%).
    80→
    81→                // OWNERSHIP-CACHE-2 : vérifier ownership via le cache en mémoire d'abord,
    82→                // puis via DB si pas encore vérifié (même pattern que saveReponse).
    83→                if s.sessionCache != nil && s.sessionCache.IsOwnershipVerified(sessionID, claims.UserID) {
    84→                        // Ownership déjà vérifiée en cache → skip DB query (-1 tx par save)
    85→                } else {
    86→                        // Première vérification : requêter la DB
    87→                        var etudiantID string
    88→                        found := false
    89→                        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
    90→                                err := tx.QueryRow(r.Context(),
    91→                                        `SELECT "etudiantId" FROM "SessionPassation" WHERE "id" = $1`,
    92→                                        sessionID).Scan(&etudiantID)
    93→                                if err == nil {
    94→                                        found = true
    95→                                }
    96→                                return err
    97→                        })
    98→                        if !found || etudiantID != claims.UserID {
    99→                                writeJSONError(w, http.StatusNotFound, "session introuvable ou accès refusé")
   100→                                return
   101→                        }
   102→                        // Cacher l'ownership vérifiée pour les prochains saves
   103→                        if s.sessionCache != nil {
   104→                                s.sessionCache.MarkOwnershipVerified(sessionID, claims.UserID)
   105→                        }
   106→                }
   107→
   108→                // OPT-1 : écrire en RAM au lieu de DB directe.
   109→                // Le cache merge les réponses (map[string]string) et marque la session dirty.
   110→                // Le worker goroutine (FlushSessionToNeon toutes les 30s) synchronisera vers Neon.
   111→                // Le submit force un flush immédiat avant la soumission finale.
   112→                if s.sessionCache != nil {
   113→                        s.sessionCache.SaveAnswers(sessionID, "", claims.UserID, reponses)
   114→                }
   115→
   116→                w.Header().Set("Content-Type", "application/json")
   117→                json.NewEncoder(w).Encode(map[string]any{
   118→                        "saved": true,
   119→                        "count": len(reponses),
   120→                })
   121→                return
   122→        }
   123→
   124→        // Cas 2 : log alerte
   125→        if rawAlerte, ok := body["alerte"]; ok {
   126→                var alerte struct {
   127→                        Type      string  `json:"type"`
   128→                        Details   string  `json:"details"`
   129→                        Penalite  float64 `json:"penalite"`
   130→                }
   131→                if err := json.Unmarshal(rawAlerte, &alerte); err != nil {
   132→                        writeJSONError(w, http.StatusBadRequest, "format alerte invalide")
   133→                        return
   134→                }
   135→                if alerte.Type == "" {
   136→                        writeJSONError(w, http.StatusBadRequest, "type alerte requis")
   137→                        return
   138→                }
   139→
   140→                // OWNERSHIP-CACHE-2 : vérifier ownership via le cache d'abord (comme reponses).
   141→                // Les alertes sont moins fréquentes mais pendant un examen avec proctoring,
   142→                // elles peuvent être nombreuses (changement d'onglet, copier-coller...).
   143→                if s.sessionCache != nil && s.sessionCache.IsOwnershipVerified(sessionID, claims.UserID) {
   144→                        // Ownership déjà vérifiée → skip DB
   145→                } else {
   146→                        var etudiantID string
   147→                        found := false
   148→                        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   149→                                err := tx.QueryRow(r.Context(),
   150→                                        `SELECT "etudiantId" FROM "SessionPassation" WHERE "id" = $1`,
   151→                                        sessionID).Scan(&etudiantID)
   152→                                if err == nil {
   153→                                        found = true
   154→                                }
   155→                                return err
   156→                        })
   157→                        if !found || etudiantID != claims.UserID {
   158→                                writeJSONError(w, http.StatusNotFound, "session introuvable ou accès refusé")
   159→                                return
   160→                        }
   161→                        if s.sessionCache != nil {
   162→                                s.sessionCache.MarkOwnershipVerified(sessionID, claims.UserID)
   163→                        }
   164→                }
   165→
   166→                // AddAlerte (RLS off dans le repo — conforme au pattern existant)
   167→                alerteInput := domain.AlerteInput{
   168→                        Type:     alerte.Type,
   169→                        Details:  alerte.Details,
   170→                }
   171→                _ = s.sessionUC.AddAlerte(r.Context(), claims, sessionID, alerte.Penalite, alerteInput)
   172→
   173→                // OPT-7 : push WebSocket aux surveillants — alerte déclenchée.
   174→                // On récupère l'epreuveId de la session pour le broadcast ciblé.
   175→                var alerteEpreuveID string
   176→                _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   177→                        _ = tx.QueryRow(r.Context(),
   178→                                `SELECT "epreuveId" FROM "SessionPassation" WHERE "id" = $1`,
   179→                                sessionID).Scan(&alerteEpreuveID)
   180→                        return nil
   181→                })
   182→                s.surveillanceHub.BroadcastEvent("ALERT_TRIGGERED", alerteEpreuveID, map[string]any{
   183→                        "sessionId":  sessionID,
   184→                        "epreuveId":  alerteEpreuveID,
   185→                        "etudiantId": claims.UserID,
   186→                        "alertType":  alerte.Type,
   187→                        "penalite":   alerte.Penalite,
   188→                })
   189→
   190→                w.Header().Set("Content-Type", "application/json")
   191→                json.NewEncoder(w).Encode(map[string]any{
   192→                        "logged": true,
   193→                })
   194→                return
   195→        }
   196→
   197→        writeJSONError(w, http.StatusBadRequest, "body doit contenir 'reponses' ou 'alerte'")
   198→}

   200→// ──────────────────────────────────────────────────────────────────────────
   201→// B3 : POST /api/sessions/{id}/capture — persiste capture écran anti-fraude
   202→// ──────────────────────────────────────────────────────────────────────────
   203→//
   204→// Le frontend envoie { image: "data:image/jpeg;base64,..." }.
   205→// On decode le base64, on upload l'image vers R2, et on stocke les
   206→// métadonnées dans la table SessionCapture. On conserve aussi l'event
   207→// CAPTURE dans logEvents pour compatibilité ascendante.
   208→//
   209→// Si R2 n'est pas disponible (mode dev), on stocke juste les métadonnées
   210→// avec r2Key vide et l'image comme miniature base64.
   211→
   212→func (s *Server) captureSession(w http.ResponseWriter, r *http.Request) {
   213→        claims, ok := middleware.ClaimsFromContext(r.Context())
   214→        if !ok || claims.UserID == "" {
   215→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
   216→                return
   217→        }
   218→        sessionID := chi.URLParam(r, "id")
   219→        if sessionID == "" {
   220→                writeJSONError(w, http.StatusBadRequest, "id session requis")
   221→                return
   222→        }
   223→
   224→        var input struct {
   225→                Image string `json:"image"`
   226→        }
   227→        if err := json.NewDecoder(r.Body).Decode(&input); err != nil {
   228→                writeJSONError(w, http.StatusBadRequest, "corps de requête invalide")
   229→                return
   230→        }
   231→        if input.Image == "" {
   232→                writeJSONError(w, http.StatusBadRequest, "image requise")
   233→                return
   234→        }
   235→
   236→        // Vérifier ownership + récupérer epreuveId
   237→        var etudiantID, epreuveID string
   238→        found := false
   239→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   240→                err := tx.QueryRow(r.Context(),
   241→                        `SELECT "etudiantId", "epreuveId" FROM "SessionPassation" WHERE "id" = $1`,
   242→                        sessionID).Scan(&etudiantID, &epreuveID)
   243→                if err == nil {
   244→                        found = true
   245→                }
   246→                return err
   247→        })
   248→        if !found || etudiantID != claims.UserID {
   249→                writeJSONError(w, http.StatusNotFound, "session introuvable ou accès refusé")
   250→                return
   251→        }
   252→
   253→        // Décoder le base64 de l'image (accepte data:image/jpeg;base64,... ou base64 brut)
   254→        imageData := input.Image
   255→        if idx := strings.Index(imageData, ";base64,"); idx >= 0 {
   256→                imageData = imageData[idx+8:]
   257→        }
   258→        imageBytes, err := base64.StdEncoding.DecodeString(imageData)
   259→        if err != nil {
   260→                // Essayer RawStdEncoding au cas où
   261→                imageBytes, err = base64.RawStdEncoding.DecodeString(imageData)
   262→                if err != nil {
   263→                        writeJSONError(w, http.StatusBadRequest, "image base64 invalide")
   264→                        return
   265→                }
   266→        }
   267→
   268→        // Calculer SHA-256 pour intégrité
   269→        hash := sha256.Sum256(imageBytes)
   270→        imageHash := fmt.Sprintf("%x", hash)
   271→
   272→        // Déterminer le prochain captureIndex pour cette session
   273→        var captureIndex int
   274→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   275→                err := tx.QueryRow(r.Context(),
   276→                        `SELECT COALESCE(MAX("captureIndex"), 0) + 1 FROM "SessionCapture" WHERE "sessionId" = $1`,
   277→                        sessionID).Scan(&captureIndex)
   278→                return err
   279→        })
   280→        if captureIndex == 0 {
   281→                captureIndex = 1
   282→        }
   283→
   284→        // Upload vers R2 si disponible
   285→        captureID := uuid.New().String()
   286→        r2Key := ""
   287→        var fileSize int
   288→
   289→        if s.storage != nil {
   290→                // Générer la clé R2 : captures/{sessionID}/{timestamp}.jpg
   291→                ts := time.Now().UTC().Format("20060102-150405.000")
   292→                r2Key = fmt.Sprintf("captures/%s/%s.jpg", sessionID, ts)
   293→                fileSize = len(imageBytes)
   294→
   295→                _, err := s.storage.Upload(r.Context(), domain.StorageObject{
   296→                        Key:           r2Key,
   297→                        Content:       imageBytes,
   298→                        ContentType:   "image/jpeg",
   299→                        ContentLength: int64(fileSize),
   300→                })
   301→                if err != nil {
   302→                        // Log l'erreur mais ne pas bloquer la capture — on stocke juste les métadonnées
   303→                        fmt.Printf("[captureSession] R2 upload failed: %v\n", err)
   304→                        r2Key = ""
   305→                        fileSize = 0
   306→                }
   307→        }
   308→
   309→        // Insérer les métadonnées dans SessionCapture
   310→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   311→                _, err := tx.Exec(r.Context(),
   312→                        `INSERT INTO "SessionCapture" ("id", "sessionId", "etudiantId", "epreuveId", "r2Key", "imageHash", "fileSize", "captureIndex", "createdAt")
   313→                         VALUES ($1, $2, $3, $4, $5, $6, $7, $8, NOW())`,
   314→                        captureID, sessionID, etudiantID, epreuveID, r2Key, imageHash, fileSize, captureIndex)
   315→                return err
   316→        })
   317→
   318→        // Logger la capture dans logEvents (append un event CAPTURE)
   319→        // Compatibilité ascendante : le logEvent CAPTURE reste présent.
   320→        alerteInput := domain.AlerteInput{
   321→                Type:    "CAPTURE",
   322→                Details: fmt.Sprintf("Capture écran #%d prise (hash: %s...)", captureIndex, imageHash[:16]),
   323→        }
   324→        _ = s.sessionUC.AddAlerte(r.Context(), claims, sessionID, 0, alerteInput)
   325→
   326→        w.Header().Set("Content-Type", "application/json")
   327→        json.NewEncoder(w).Encode(map[string]any{
   328→                "captured":     true,
   329→                "captureId":    captureID,
   330→                "captureIndex": captureIndex,
   331→        })
   332→}

   334→// ──────────────────────────────────────────────────────────────────────────
   335→// GET /api/sessions/{id}/captures — liste des captures d'une session
   336→// ──────────────────────────────────────────────────────────────────────────
   337→//
   338→// Retourne la liste des captures avec URL présignée R2 (valide 5 minutes).
   339→// Accès : ENSEIGNANT/ADMIN/RESPONSABLE ou l'étudiant propriétaire.
   340→
   341→func (s *Server) listSessionCaptures(w http.ResponseWriter, r *http.Request) {
   342→        claims, ok := middleware.ClaimsFromContext(r.Context())
   343→        if !ok || claims.UserID == "" {
   344→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
   345→                return
   346→        }
   347→        sessionID := chi.URLParam(r, "id")
   348→        if sessionID == "" {
   349→                writeJSONError(w, http.StatusBadRequest, "id session requis")
   350→                return
   351→        }
   352→
   353→        // Vérifier l'accès : l'étudiant propriétaire OU enseignant/admin/responsable
   354→        role := domain.Role(claims.Role)
   355→        isStaff := role == domain.RoleEnseignant || role == domain.RoleAdmin || role == domain.RoleResponsable
   356→
   357→        var etudiantID string
   358→        found := false
   359→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   360→                err := tx.QueryRow(r.Context(),
   361→                        `SELECT "etudiantId" FROM "SessionPassation" WHERE "id" = $1`,
   362→                        sessionID).Scan(&etudiantID)
   363→                if err == nil {
   364→                        found = true
   365→                }
   366→                return err
   367→        })
   368→        if !found {
   369→                writeJSONError(w, http.StatusNotFound, "session introuvable")
   370→                return
   371→        }
   372→        if !isStaff && etudiantID != claims.UserID {
   373→                writeJSONError(w, http.StatusForbidden, "accès refusé")
   374→                return
   375→        }
   376→
   377→        // Récupérer les captures
   378→        type captureRow struct {
   379→                ID           string     `json:"id"`
   380→                R2Key        string     `json:"r2Key"`
   381→                ImageHash    *string    `json:"imageHash,omitempty"`
   382→                FileSize     *int       `json:"fileSize,omitempty"`
   383→                CaptureIndex int        `json:"captureIndex"`
   384→                CreatedAt    time.Time  `json:"createdAt"`
   385→        }
   386→
   387→        var captures []captureRow
   388→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   389→                rows, err := tx.Query(r.Context(),
   390→                        `SELECT "id", "r2Key", "imageHash", "fileSize", "captureIndex", "createdAt"
   391→                         FROM "SessionCapture"
   392→                         WHERE "sessionId" = $1
   393→                         ORDER BY "captureIndex" ASC`,
   394→                        sessionID)
   395→                if err != nil {
   396→                        return err
   397→                }
   398→                defer rows.Close()
   399→                for rows.Next() {
   400→                        var c captureRow
   401→                        if err := rows.Scan(&c.ID, &c.R2Key, &c.ImageHash, &c.FileSize, &c.CaptureIndex, &c.CreatedAt); err != nil {
   402→                                return err
   403→                        }
   404→                        captures = append(captures, c)
   405→                }
   406→                return rows.Err()
   407→        })
   408→
   409→        // Enrichir avec URL présignées si R2 est disponible
   410→        type captureResponse struct {
   411→                ID           string  `json:"id"`
   412→                URL          string  `json:"url,omitempty"`
   413→                R2Key        string  `json:"r2Key,omitempty"`
   414→                ImageHash    *string `json:"imageHash,omitempty"`
   415→                FileSize     *int    `json:"fileSize,omitempty"`
   416→                CaptureIndex int     `json:"captureIndex"`
   417→                CreatedAt    string  `json:"createdAt"`
   418→        }
   419→
   420→        result := make([]captureResponse, 0, len(captures))
   421→        for _, c := range captures {
   422→                resp := captureResponse{
   423→                        ID:           c.ID,
   424→                        R2Key:        c.R2Key,
   425→                        ImageHash:    c.ImageHash,
   426→                        FileSize:     c.FileSize,
   427→                        CaptureIndex: c.CaptureIndex,
   428→                        CreatedAt:    c.CreatedAt.UTC().Format(time.RFC3339),
   429→                }
   430→                // Générer URL présignée (5 min = 300s) si R2 disponible et clé présente
   431→                if s.storage != nil && c.R2Key != "" {
   432→                        url, err := s.storage.PresignURL(r.Context(), c.R2Key, 300)
   433→                        if err == nil {
   434→                                resp.URL = url
   435→                        }
   436→                }
   437→                result = append(result, resp)
   438→        }
   439→
   440→        w.Header().Set("Content-Type", "application/json")
   441→        json.NewEncoder(w).Encode(map[string]any{
   442→                "captures": result,
   443→        })
   444→}

   446→// ──────────────────────────────────────────────────────────────────────────
   447→// B4 : GET /api/epreuves/auto-close?epreuveId=X — détection clôture auto
   448→// ──────────────────────────────────────────────────────────────────────────
   449→//
   450→// Le frontend poll toutes les 15s pendant l'examen pour détecter :
   451→//   - si l'épreuve a été clôturée manuellement (clotureeAt non null)
   452→//   - si la dateFin est dépassée (avec période de grâce)
   453→//
   454→// Response: { isClosed, raisonCloture, inGracePeriod, gracePeriodEndsAt }
   455→
   456→func (s *Server) epreuveAutoClose(w http.ResponseWriter, r *http.Request) {
   457→        claims, ok := middleware.ClaimsFromContext(r.Context())
   458→        if !ok || claims.UserID == "" {
   459→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
   460→                return
   461→        }
   462→        epreuveID := r.URL.Query().Get("epreuveId")
   463→        if epreuveID == "" {
   464→                writeJSONError(w, http.StatusBadRequest, "epreuveId requis")
   465→                return
   466→        }
   467→
   468→        var (
   469→                clotureeAt     *time.Time
   470→                raisonCloture  *string
   471→                dateFin        *time.Time
   472→                delaiGrace     int
   473→                statut         string
   474→        )
   475→        found := false
   476→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   477→                err := tx.QueryRow(r.Context(), `
   478→                        SELECT "clotureeAt", "raisonCloture", "dateFin", COALESCE("delaiGrace", 0), "statut"::text
   479→                        FROM "Epreuve"
   480→                        WHERE "id" = $1 AND "deletedAt" IS NULL
   481→                `, epreuveID).Scan(&clotureeAt, &raisonCloture, &dateFin, &delaiGrace, &statut)
   482→                if err == nil {
   483→                        found = true
   484→                }
   485→                return err
   486→        })
   487→        if !found {
   488→                writeJSONError(w, http.StatusNotFound, "épreuve introuvable")
   489→                return
   490→        }
   491→
   492→        now := time.Now().UTC()
   493→        isClosed := false
   494→        inGracePeriod := false
   495→        var gracePeriodEndsAt *time.Time
   496→
   497→        if clotureeAt != nil {
   498→                // Clôturée manuellement
   499→                isClosed = true
   500→        } else if statut == "CLOTUREE" {
   501→                isClosed = true
   502→        } else if dateFin != nil && now.After(*dateFin) {
   503→                // Date fin dépassée — vérifier période de grâce
   504→                graceEnd := dateFin.Add(time.Duration(delaiGrace) * time.Minute)
   505→                if now.Before(graceEnd) {
   506→                        inGracePeriod = true
   507→                        gracePeriodEndsAt = &graceEnd
   508→                } else {
   509→                        isClosed = true
