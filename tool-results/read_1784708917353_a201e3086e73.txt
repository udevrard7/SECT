  1528→-- ============================================================
  1529→-- Migration 000005 — Fonction + triggers updated_at
  1530→-- ============================================================
  1531→-- Prisma gère @updatedAt côté client (JS). Le backend Go n'utilisera
  1532→-- pas Prisma : on délègue donc la mise à jour de "updatedAt" à un
  1533→-- trigger PostgreSQL qui s'exécute automatiquement sur chaque UPDATE.
  1534→--
  1535→-- La fonction est idempotent (CREATE OR REPLACE) et safe à ré-exécuter.
  1536→-- Les triggers utilisent `FOR EACH STATEMENT` (et non ROW) pour rester
  1537→-- performants même sur les updates bulk ; la colonne updatedAt est
  1538→-- juste mise à CURRENT_TIMESTAMP.
  1539→-- ============================================================
  1540→
  1541→CREATE OR REPLACE FUNCTION public.set_updated_at()
  1542→RETURNS TRIGGER AS $$
  1543→BEGIN
  1544→  -- For statement-level trigger, update the NEW row's updatedAt
  1545→  NEW."updatedAt" = CURRENT_TIMESTAMP;
  1546→  RETURN NEW;
  1547→END;
  1548→$$ LANGUAGE plpgsql;
  1549→
  1550→-- ============================================================
  1551→-- Création des triggers FOR EACH ROW sur toutes les tables ayant
  1552→-- une colonne "updatedAt" (31 tables identifiées dans le schéma)
  1553→-- ============================================================
  1554→
  1555→DO $$
  1556→DECLARE
  1557→  t text;
  1558→BEGIN
  1559→  FOR t IN
  1560→    SELECT table_name
  1561→    FROM information_schema.columns
  1562→    WHERE table_schema = 'public'
  1563→      AND column_name = 'updatedAt'
  1564→    ORDER BY table_name
  1565→  LOOP
  1566→    EXECUTE format(
  1567→      'CREATE TRIGGER trg_set_updated_at BEFORE UPDATE ON public.%I FOR EACH ROW EXECUTE FUNCTION public.set_updated_at();',
  1568→      t
  1569→    );
  1570→    -- Drop & recreate pour idempotence (si le trigger existe déjà)
  1571→    EXECUTE format(
  1572→      'DROP TRIGGER IF EXISTS trg_set_updated_at ON public.%I; CREATE TRIGGER trg_set_updated_at BEFORE UPDATE ON public.%I FOR EACH ROW EXECUTE FUNCTION public.set_updated_at();',
  1573→      t, t
  1574→    );
  1575→  END LOOP;
  1576→END
  1577→$$;

  1579→-- Migration 000006 — Row Level Security (Option A : claims de session)
  1580→-- ============================================================
  1581→--
  1582→-- Stratégie : "claims-based RLS"
  1583→--
  1584→-- Le backend Go pose, au début de chaque transaction, des claims via :
  1585→--   SELECT set_config('app.claims.user_id', $1, true);
  1586→--   SELECT set_config('app.claims.role', $2, true);
  1587→--   SELECT set_config('app.claims.etablissement_id', $3, true);  -- NULL pour ADMIN
  1588→--
  1589→-- Le paramètre `true` (is_local) scope les claims à la transaction
  1590→-- courante — ils sont automatiquement nettoyés en fin de transaction.
  1591→--
  1592→-- Hiérarchie des rôles :
  1593→--   ADMIN       — Propriétaire PaaS. NON lié à un établissement.
  1594→--                 Accès aux tables plateforme uniquement (Plan, PlatformSettings,
  1595→--                 IpWhitelist, Facture, MonitoringEvent, AIProviderConfig, ...).
  1596→--                 Accès aux données d'un établissement UNIQUEMENT si une entrée
  1597→--                 existe dans "EtablissementAccess" (autorisation explicite).
  1598→--   RESPONSABLE — Gère SON établissement (via etablissementId).
  1599→--   ENSEIGNANT  — Ses propres données + données de ses étudiants/filières.
  1600→--   ETUDIANT    — Ses propres données uniquement.
  1601→--
  1602→-- Note technique : neondb_owner est le propriétaire des tables. En PostgreSQL,
  1603→-- le propriétaire bypass RLS sauf si FORCE ROW LEVEL SECURITY est activé.
  1604→-- On active donc FORCE sur toutes les tables pour que les policies s'appliquent
  1605→-- même via neondb_owner (qui n'est PAS superuser sur Neon).
  1606→-- ============================================================

  1718→    EXECUTE format('ALTER TABLE public.%I FORCE ROW LEVEL SECURITY;', t);
  1719→  END LOOP;
  1720→END
  1721→$$;
