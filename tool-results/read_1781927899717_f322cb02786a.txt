   455→/**
   456→ * Calcule la valeur métrique pour un badge donné à partir des métriques fournies.
   457→ */
   458→function getMetricValue(cle: string, metrics: BadgeMetrics): number {
   459→  switch (cle) {
   460→    // Étudiant
   461→    case 'bapteme_du_feu': return metrics.nbEpreuvesTerminees ?? 0
   462→    case 'bien_joue': return metrics.nbNotesSup12 ?? 0
   463→    case 'major_de_promo': return metrics.nbNotesSup16 ?? 0
   464→    case 'eclair_de_genie': return metrics.nbEpreuvesRapides ?? 0
   465→    case 'persévérant': return metrics.nbJoursConnexion ?? 0
   466→    case 'zero_faute': return metrics.nbScoresParfaits ?? 0
   467→
   468→    // Enseignant
   469→    case 'premiere_epreuve': return metrics.nbEpreuvesCreees ?? 0
   470→    case 'maitre_corrigeur': return metrics.nbCopiesCorrigees ?? 0
   471→    case 'createur_ia': return metrics.nbEpreuvesIA ?? 0
   472→    case 'excellence_pedagogique':
   473→      // Pour ce badge, la métrique est la moyenne (seuil = note minimale)
   474→      // mais on ne valide que si on a assez de copies
   475→      if ((metrics.nbCopiesPourMoyenne ?? 0) < 5) return 0
   476→      return metrics.moyenneEtudiants ?? 0
   477→    case 'banquier_questions': return metrics.nbQuestionsCreees ?? 0
   478→    case 'correcteur_rapide': return metrics.nbCorrectionsRapides ?? 0
   479→
   480→    // Responsable
   481→    case 'batisseur': return metrics.nbEvaluations ?? 0
   482→    case 'pilier_academique': return metrics.nbEtudiants ?? 0
   483→    case 'visionnaire': return metrics.tauxReussite ?? 0
   484→    case 'architecte_filiaire': return metrics.nbFilieres ?? 0
   485→
   486→    // Admin
   487→    case 'gardien_plateforme': return metrics.nbEtablissements ?? 0
   488→    case 'strategiste': return metrics.tauxReussiteGlobal ?? 0
   489→    case 'pilote_ia': return metrics.nbProvidersIA ?? 0
   490→    case 'sensei': return metrics.nbEnseignantsActifs ?? 0
   491→
   492→    default: return 0
   493→  }
   494→}

   496→/**
   497→ * Calcule la progression et le niveau pour un badge donné.
   498→ */
   499→export function computeBadgeProgress(
   500→  definition: BadgeDefinitionConfig,
   501→  metrics: BadgeMetrics
   502→): Omit<BadgeWithProgress, 'isNewlyUnlocked'> {
   503→  const valeurActuelle = getMetricValue(definition.cle, metrics)
   504→  const niveaux = definition.niveaux
   505→
   506→  // Trouver le niveau actuel et les paliers
   507→  let niveauActuel: NiveauBadge = 'BRONZE'
   508→  let valeurPalier = niveaux[0].seuil
   509→  let valeurProchain: number | null = niveaux.length > 1 ? niveaux[1].seuil : null
   510→  let debloque = false
   511→  let dateObtention: string | null = null
   512→
   513→  // Vérifier chaque niveau du plus haut au plus bas
   514→  for (let i = niveaux.length - 1; i >= 0; i--) {
   515→    if (valeurActuelle >= niveaux[i].seuil) {
   516→      niveauActuel = niveaux[i].niveau
   517→      valeurPalier = niveaux[i].seuil
   518→      valeurProchain = i < niveaux.length - 1 ? niveaux[i + 1].seuil : null
   519→      debloque = true
   520→      break
   521→    }
   522→  }
   523→
   524→  // Si pas encore débloquent, le palier est le premier niveau
   525→  if (!debloque) {
   526→    niveauActuel = 'BRONZE'
   527→    valeurPalier = niveaux[0].seuil
   528→    valeurProchain = niveaux.length > 1 ? niveaux[1].seuil : null
   529→  }
   530→
   531→  // Calculer la progression (0-100) vers le prochain niveau
   532→  let progression = 0
   533→  if (valeurProchain !== null && valeurProchain > valeurPalier) {
   534→    const range = valeurProchain - valeurPalier
   535→    const current = Math.max(0, valeurActuelle - valeurPalier)
   536→    progression = Math.min(100, Math.round((current / range) * 100))
   537→  } else if (debloque && valeurProchain === null) {
   538→    // Niveau maximum atteint
   539→    progression = 100
   540→  } else if (!debloque && valeurPalier > 0) {
   541→    // Pas encore débloquent, progression vers le premier niveau
   542→    progression = Math.min(100, Math.round((valeurActuelle / valeurPalier) * 100))
   543→  }
   544→
   545→  return {
   546→    cle: definition.cle,
   547→    titre: definition.titre,
   548→    description: definition.description,
   549→    icone: definition.icone,
   550→    categorie: definition.categorie,
   551→    niveauActuel,
   552→    niveaux,
   553→    valeurActuelle,
   554→    valeurPalier,
   555→    valeurProchain,
   556→    debloque,
   557→    progression,
   558→    dateObtention,
   559→  }
   560→}

   564→/**
   565→ * Collecte les métriques pour un étudiant
   566→ */
   567→export async function collectEtudiantMetrics(userId: string): Promise<BadgeMetrics> {
   568→  const sessions = await withRetry(() =>
   569→    db.sessionPassation.findMany({
   570→      where: {
   571→        etudiantId: userId,
   572→        statut: { in: ['SOUMISE', 'CORRIGEE', 'RETOURNEE'] },
   573→        score: { not: null },
   574→      },
   575→      include: {
   576→        epreuve: { select: { duree: true, noteTotal: true } },
   577→        resultat: { select: { totalPossible: true, detailParQuestion: true } },
   578→      },
   579→    })
   580→  )
   581→
   582→  const nbEpreuvesTerminees = sessions.length
   583→
   584→  let nbNotesSup12 = 0
   585→  let nbNotesSup16 = 0
   586→  let nbScoresParfaits = 0
   587→  let nbEpreuvesRapides = 0
   588→
   589→  for (const s of sessions) {
   590→    const totalPossible = s.resultat?.totalPossible || s.epreuve.noteTotal || 20
   591→    const normalizedScore = totalPossible > 0 ? ((s.score || 0) / totalPossible) * 20 : 0
   592→
   593→    if (normalizedScore >= 12) nbNotesSup12++
   594→    if (normalizedScore >= 16) nbNotesSup16++
   595→    if (normalizedScore >= 19.5) nbScoresParfaits++
   596→
   597→    // Vérifier la rapidité
   598→    if (s.dateDebut && s.dateFin && s.epreuve.duree) {
   599→      const timeTaken = (s.dateFin.getTime() - s.dateDebut.getTime()) / (1000 * 60)
   600→      if (timeTaken < s.epreuve.duree / 2) nbEpreuvesRapides++
   601→    }
   602→  }
   603→
   604→  // Jours de connexion (approximation basée sur derniereConnexion)
   605→  const user = await withRetry(() =>
   606→    db.user.findUnique({
   607→      where: { id: userId },
   608→      select: { derniereConnexion: true, createdAt: true },
   609→    })
   610→  )
   611→  const nbJoursConnexion = user?.derniereConnexion
   612→    ? Math.max(1, Math.floor((Date.now() - user.createdAt.getTime()) / (1000 * 60 * 60 * 24)))
   613→    : 0
   614→
   615→  return {
   616→    nbEpreuvesTerminees,
   617→    nbNotesSup12,
   618→    nbNotesSup16,
   619→    nbEpreuvesRapides,
   620→    nbScoresParfaits,
   621→    nbJoursConnexion,
   622→  }
   623→}

   625→/**
   626→ * Collecte les métriques pour un enseignant
   627→ */
   628→export async function collectEnseignantMetrics(userId: string): Promise<BadgeMetrics> {
   629→  const [nbEpreuvesCreees, nbQuestionsCreees, nbEpreuvesIA] = await Promise.all([
   630→    withRetry(() => db.epreuve.count({ where: { enseignantId: userId, deletedAt: null } })),
   631→    withRetry(() => db.question.count({ where: { OR: [{ auteurId: userId }, { document: { ownerId: userId } }], deletedAt: null } })),
   632→    withRetry(() => db.epreuve.count({ where: { enseignantId: userId, generationMode: 'IA_ASSISTEE', deletedAt: null } })),
   633→  ])
   634→
   635→  // Copies corrigées
   636→  const nbCopiesCorrigees = await withRetry(() =>
   637→    db.sessionPassation.count({
   638→      where: {
   639→        epreuve: { enseignantId: userId },
   640→        statut: { in: ['CORRIGEE', 'RETOURNEE'] },
   641→      },
   642→    })
   643→  )
   644→
   645→  // Moyenne étudiants
   646→  const scoredSessions = await withRetry(() =>
   647→    db.sessionPassation.findMany({
   648→      where: {
   649→        epreuve: { enseignantId: userId, deletedAt: null },
   650→        score: { not: null },
   651→      },
   652→      select: { score: true, epreuve: { select: { noteTotal: true } } },
   653→    })
   654→  )
   655→  const nbCopiesPourMoyenne = scoredSessions.length
   656→  const moyenneEtudiants = nbCopiesPourMoyenne > 0
   657→    ? scoredSessions.reduce((sum, s) => {
   658→        const noteTotal = s.epreuve?.noteTotal || 20
   659→        return sum + (s.score! / noteTotal) * 20
   660→      }, 0) / nbCopiesPourMoyenne
   661→    : 0
   662→
   663→  // Corrections rapides (< 24h)
   664→  const correctedSessions = await withRetry(() =>
   665→    db.sessionPassation.findMany({
   666→      where: {
   667→        epreuve: { enseignantId: userId },
   668→        statut: { in: ['CORRIGEE', 'RETOURNEE'] },
   669→        dateFin: { not: null },
   670→      },
   671→      select: { dateFin: true, updatedAt: true },
   672→    })
   673→  )
   674→  const nbCorrectionsRapides = correctedSessions.filter(s => {
   675→    if (!s.dateFin) return false
   676→    const diffHours = (s.updatedAt.getTime() - s.dateFin.getTime()) / (1000 * 60 * 60)
   677→    return diffHours <= 24
   678→  }).length
   679→
   680→  return {
   681→    nbEpreuvesCreees,
   682→    nbCopiesCorrigees,
   683→    nbEpreuvesIA,
   684→    moyenneEtudiants: Math.round(moyenneEtudiants * 10) / 10,
   685→    nbCopiesPourMoyenne,
   686→    nbQuestionsCreees,
   687→    nbCorrectionsRapides,
   688→  }
   689→}

   761→/**
   762→ * Calcule tous les badges pour un utilisateur donné avec leur progression.
   763→ * Persiste les progressions en base et détecte les nouveaux déblocages.
   764→ */
   765→export async function computeAllBadges(
   766→  userId: string,
   767→  role: string,
   768→  etablissementId: string | null = null
   769→): Promise<BadgeWithProgress[]> {
   770→  // 1. Collecter les métriques
   771→  let metrics: BadgeMetrics
   772→  switch (role) {
   773→    case 'ETUDIANT':
   774→      metrics = await collectEtudiantMetrics(userId)
   775→      break
   776→    case 'ENSEIGNANT':
   777→      metrics = await collectEnseignantMetrics(userId)
   778→      break
   779→    case 'RESPONSABLE':
   780→      metrics = await collectResponsableMetrics(userId, etablissementId)
   781→      break
   782→    case 'ADMIN':
   783→      metrics = await collectAdminMetrics()
   784→      break
   785→    default:
   786→      metrics = {}
   787→  }
   788→
   789→  // 2. Filtrer les badges pertinents pour ce rôle
   790→  const relevantBadges = BADGE_DEFINITIONS.filter(b => b.rolesCibles.includes(role as RoleCible))
   791→
   792→  // 3. Calculer la progression pour chaque badge
   793→  const badgesWithProgress: BadgeWithProgress[] = relevantBadges.map(def => {
   794→    const progress = computeBadgeProgress(def, metrics)
   795→
   796→    // Vérifier si c'est un nouveau déblocage en comparant avec la DB
   797→    return {
   798→      ...progress,
   799→      isNewlyUnlocked: false, // Sera mis à jour après comparaison DB
   800→    }
   801→  })
   802→
   803→  // 4. Synchroniser avec la base de données
   804→  try {
   805→    // Récupérer les progressions existantes
   806→    const existingProgressions = await withRetry(() =>
   807→      db.badgeProgression.findMany({
   808→        where: { userId },
   809→        include: { badgeDefinition: true },
   810→      })
   811→    )
   812→
   813→    // S'assurer que toutes les définitions de badges existent en base
   814→    for (const def of relevantBadges) {
   815→      const existing = existingProgressions.find(p => p.badgeDefinition.cle === def.cle)
   816→      const computed = badgesWithProgress.find(b => b.cle === def.cle)!
   817→
   818→      if (!existing) {
   819→        // Créer la définition en base si elle n'existe pas
   820→        let badgeDef = await withRetry(() =>
   821→          db.badgeDefinition.findUnique({ where: { cle: def.cle } })
   822→        )
   823→        if (!badgeDef) {
   824→          badgeDef = await withRetry(() =>
   825→            db.badgeDefinition.create({
   826→              data: {
   827→                cle: def.cle,
   828→                titre: def.titre,
   829→                description: def.description,
   830→                icone: def.icone,
   831→                categorie: def.categorie,
   832→                roleCible: def.rolesCibles.length === 1 ? def.rolesCibles[0] : null,
   833→                niveaux: def.niveaux.map(n => n.niveau),
   834→                ordre: def.ordre,
   835→              },
   836→            })
   837→          )
   838→        }
   839→
   840→        // Créer la progression
   841→        const wasLocked = !computed.debloque
   842→        await withRetry(() =>
   843→          db.badgeProgression.create({
   844→            data: {
   845→              userId,
   846→              badgeDefinitionId: badgeDef!.id,
   847→              niveauActuel: computed.niveauActuel,
   848→              valeurActuelle: computed.valeurActuelle,
   849→              valeurPalier: computed.valeurPalier,
   850→              valeurProchain: computed.valeurProchain,
   851→              debloque: computed.debloque,
   852→              dateObtention: computed.debloque ? new Date() : null,
   853→            },
   854→          })
   855→        )
   856→
   857→        // Marquer comme nouvellement débloquent si applicable
   858→        if (computed.debloque && wasLocked) {
   859→          computed.isNewlyUnlocked = true
   860→        }
   861→      } else {
   862→        // Mettre à jour la progression existante
   863→        const wasUnlocked = existing.debloque
   864→        const currentNiveau = existing.niveauActuel
   865→        const hasNewLevel = computed.debloque && computed.niveauActuel !== currentNiveau
   866→        const hasNewUnlock = computed.debloque && !wasUnlocked
   867→
   868→        if (
   869→          existing.valeurActuelle !== computed.valeurActuelle ||
   870→          existing.niveauActuel !== computed.niveauActuel ||
   871→          existing.debloque !== computed.debloque
   872→        ) {
   873→          await withRetry(() =>
   874→            db.badgeProgression.update({
   875→              where: { id: existing.id },
   876→              data: {
   877→                niveauActuel: computed.niveauActuel,
   878→                valeurActuelle: computed.valeurActuelle,
   879→                valeurPalier: computed.valeurPalier,
   880→                valeurProchain: computed.valeurProchain,
   881→                debloque: computed.debloque,
   882→                dateObtention: hasNewLevel || hasNewUnlock ? new Date() : existing.dateObtention,
   883→              },
   884→            })
   885→          )
   886→        }
   887→
   888→        // Marquer comme nouvellement débloquent si nouveau niveau ou nouveau déblocage
   889→        if (hasNewLevel || hasNewUnlock) {
   890→          computed.isNewlyUnlocked = true
   891→          computed.dateObtention = new Date().toISOString()
   892→        } else {
   893→          computed.dateObtention = existing.dateObtention?.toISOString() ?? null
   894→        }
   895→      }
   896→    }
   897→  } catch (error) {
   898→    console.error('[Badge Engine] Error syncing badge progressions:', error)
   899→    // Ne pas bloquer l'affichage si la synchronisation échoue
   900→  }
   901→
   902→  return badgesWithProgress
   903→}

   940→        progression: 0,

   955→    } else if (progression.valeurPalier > 0) {

   957→    }

   972→      dateObtention: progression.dateObtention?.toISOString() ?? null,
   973→      isNewlyUnlocked: false,
   974→    }
