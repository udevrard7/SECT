   136→/** isMobile — détection responsive pour décider du rendu timeline vs table. */
   137→function useIsMobile(): boolean {
   138→  const [isMobile, setIsMobile] = useState(false)
   139→  useEffect(() => {
   140→    const mq = window.matchMedia('(max-width: 639px)')
   141→    const onChange = (e: MediaQueryListEvent | MediaQueryList) => {
   142→      setIsMobile('matches' in e ? e.matches : false)
   143→    }
   144→    onChange(mq)
   145→    mq.addEventListener('change', onChange)
   146→    return () => mq.removeEventListener('change', onChange)
   147→  }, [])
   148→  return isMobile
   149→}
