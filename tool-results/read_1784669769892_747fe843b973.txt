   227→function mapConfigToApi(config: AppConfig): Record<string, unknown> {

   236→// ─── Main Component ───
   237→
   238→export function ConfigurationPage() {
   239→  const router = useRouter()
   240→  const queryClient = useQueryClient()
   241→  const [config, setConfig] = useState<AppConfig>(DEFAULT_CONFIG)
   242→  const [savingTab, setSavingTab] = useState<string | null>(null)
   243→
   244→  // ─── Fetch settings (TanStack Query) ───
   245→  // BUGFIX (QUERY-CACHE-2) : migration de useEffect+fetch vers TanStack Query.
   246→  // config est éditable localement (forms) → sync via useEffect query.data → setConfig
   247→  // (pattern approuvé : sync de données externes vers state local pour édition).
   248→  const settingsQuery = useQuery<{ settings: Record<string, unknown> }>({
   249→    queryKey: ['platform-settings'],
   250→    queryFn: async () => {
   251→      const res = await fetch('/api/platform-settings')
   252→      if (!res.ok) throw new Error('Erreur réseau')
   253→      const data = await res.json()
   254→      return { settings: data.settings ?? {} }
   255→    },
   256→    staleTime: 60 * 1000,
   257→    refetchOnWindowFocus: false,
   258→  })
   259→
   260→  // Sync query.data → config (pour édition de form)
   261→  useEffect(() => {
   262→    if (settingsQuery.data) {
   263→      setConfig(mapApiToConfig(settingsQuery.data.settings))
   264→    }
   265→  }, [settingsQuery.data])
   266→
   267→  // Toast sur erreur (préserve le comportement du catch original)
   268→  useEffect(() => {
   269→    if (settingsQuery.error) {
   270→      toast.error('Erreur de chargement', {
   271→        description: 'Impossible de récupérer les paramètres de la plateforme.',
   272→      })
   273→    }
   274→  }, [settingsQuery.error])
   275→
   276→  const isLoading = settingsQuery.isLoading
   277→  const loadError = settingsQuery.error ? 'Impossible de charger la configuration. Vérifiez votre connexion.' : null
   278→  const refreshSettings = () => queryClient.invalidateQueries({ queryKey: ['platform-settings'] })
   279→
   280→  // ─── Save handler (generic) ───
   281→
   282→  const handleSave = async (tab: string) => {
   283→    setSavingTab(tab)
   284→    try {
   285→      const flatSettings = mapConfigToApi(config)
   286→      const res = await fetch('/api/platform-settings', {
   287→        method: 'POST',
   288→        headers: { 'Content-Type': 'application/json' },
   289→        body: JSON.stringify(flatSettings),
   290→      })
   291→      if (!res.ok) {
   292→        const err = await res.json().catch(() => ({}))
   293→        throw new Error(err.error || 'Erreur lors de la sauvegarde')
   294→      }
   295→      toast.success('Configuration sauvegardée', {
   296→        description: `Les paramètres ${tab === 'general' ? 'généraux' : tab === 'security' ? 'de sécurité' : tab === 'notifications' ? 'de notifications' : 'IA'} ont été mis à jour.`,
   297→      })
   298→    } catch (err) {
   299→      toast.error('Erreur de sauvegarde', {
   300→        description: err instanceof Error ? err.message : 'Une erreur est survenue.',
   301→      })
   302→    } finally {
   303→      setSavingTab(null)
   304→    }
   305→  }
   306→
   307→  // ─── Loading state ───
   308→
   309→  if (isLoading) {
   310→    return (
   311→      <div className="space-y-6">
   312→        <div className="space-y-2">
   313→          <PulseSkeleton className="h-9 w-64" />
   314→          <PulseSkeleton className="h-5 w-96" />
   315→        </div>
   316→        <PulseSkeleton className="h-10 w-full" />
   317→        <Card>
   318→          <CardHeader>
   319→            <PulseSkeleton className="h-6 w-48" />
   320→            <PulseSkeleton className="h-4 w-64" />
   321→          </CardHeader>
   322→          <CardContent className="space-y-6">
   323→            {Array.from({ length: 4 }).map((_, i) => (
   324→              <div key={i} className="space-y-2">
   325→                <PulseSkeleton className="h-4 w-32" />
   326→                <PulseSkeleton className="h-10 w-full" />
   327→              </div>
   328→            ))}
   329→          </CardContent>
   330→        </Card>
   331→      </div>
   332→    )
   333→  }
   334→
   335→  // ─── Error state ───
   336→
   337→  if (loadError) {
   338→    return (
   339→      <div className="space-y-6">
   340→        <div className="ds-kente-pattern -mx-4 -mt-4 rounded-lg px-4 py-4 sm:-mx-6 sm:px-6">
   341→          <h1 className="text-2xl font-display font-bold tracking-tight md:text-3xl flex items-center gap-2">
   342→            <Settings className="h-7 w-7 text-destructive" />
   343→            Paramètres Plateforme
   344→          </h1>
   345→        </div>
   346→        <Card className="border-destructive/30 bg-destructive/5">
   347→          <CardContent className="flex flex-col items-center justify-center py-12">
   348→            <AlertCircle className="h-12 w-12 text-destructive mb-4" />
   349→            <p className="text-lg font-display font-semibold text-destructive mb-2">Erreur de chargement</p>
   350→            <p className="text-sm text-muted-foreground mb-4">{loadError}</p>
   351→            <Button variant="outline" onClick={refreshSettings}>
   352→              <Loader2 className="h-4 w-4 mr-2" />
   353→              Réessayer
   354→            </Button>
   355→          </CardContent>
   356→        </Card>
   357→      </div>
   358→    )
   359→  }
   360→
   361→  return (
   362→    <div className="space-y-6">
   363→      {/* ─── Header ─── */}
   364→      <div className="ds-kente-pattern -mx-4 -mt-4 rounded-lg px-4 py-4 sm:-mx-6 sm:px-6">
   365→        <h1 className="text-2xl font-display font-bold tracking-tight md:text-3xl flex items-center gap-2">
   366→          <Settings className="h-7 w-7 text-destructive" />
   367→          Paramètres Plateforme
   368→        </h1>
   369→        <p className="mt-1 text-sm text-muted-foreground">
   370→          Configuration globale de la plateforme SaaS — réservée à l&apos;administrateur
   371→        </p>
   372→      </div>
   373→
   374→      {/* ─── Tabs ─── */}
   375→      <Tabs defaultValue="general" className="space-y-6">
   376→        <TabsList className="grid w-full grid-cols-4">
   377→          <TabsTrigger value="general" className="gap-1.5">
   378→            <Globe className="h-4 w-4 hidden sm:block" />
   379→            Général
   380→          </TabsTrigger>
   381→          <TabsTrigger value="security" className="gap-1.5">
   382→            <Shield className="h-4 w-4 hidden sm:block" />
   383→            Sécurité
   384→          </TabsTrigger>
   385→          <TabsTrigger value="notifications" className="gap-1.5">
   386→            <Bell className="h-4 w-4 hidden sm:block" />
   387→            Notifications
   388→          </TabsTrigger>
   389→          <TabsTrigger value="ia" className="gap-1.5">
   390→            <Sparkles className="h-4 w-4 hidden sm:block" />
   391→            IA
   392→          </TabsTrigger>
   393→        </TabsList>
   394→
   395→        {/* ─── Tab: Général ─── */}
   396→        <TabsContent value="general">
   397→          <Card>
   398→            <CardHeader>
   399→              <CardTitle className="flex items-center gap-2 font-display tracking-tight">
   400→                <Globe className="h-5 w-5 text-success-text" />
   401→                Paramètres généraux
   402→              </CardTitle>
   403→              <CardDescription>
   404→                Configuration de base de la plateforme
   405→              </CardDescription>
   406→            </CardHeader>
   407→            <CardContent className="space-y-6">
   408→              <div className="space-y-2">
   409→                <Label htmlFor="cfg-site-name">Nom du site</Label>
   410→                <Input
   411→                  id="cfg-site-name"
   412→                  value={config.general.siteName}
   413→                  onChange={(e) => setConfig({
   414→                    ...config,
   415→                    general: { ...config.general, siteName: e.target.value },
   416→                  })}
   417→                />
   418→              </div>
   419→
   420→              <div className="space-y-2">
   421→                <Label htmlFor="cfg-description">Description</Label>
   422→                <Textarea
