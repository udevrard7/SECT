   253→// ─── Typewriter Hook ───
   254→function useTypewriter(words: string[], typingSpeed = 80, deletingSpeed = 40, pauseTime = 2000) {
   255→  const [text, setText] = useState('')
   256→  const [wordIndex, setWordIndex] = useState(0)
   257→  const [isDeleting, setIsDeleting] = useState(false)
   258→
   259→  useEffect(() => {
   260→    const currentWord = words[wordIndex]
   261→
   262→    const timeout = setTimeout(() => {
   263→      if (!isDeleting) {
   264→        setText(currentWord.substring(0, text.length + 1))
   265→        if (text.length === currentWord.length) {
   266→          setTimeout(() => setIsDeleting(true), pauseTime)
   267→        }
   268→      } else {
   269→        setText(currentWord.substring(0, text.length - 1))
   270→        if (text.length === 0) {
   271→          setIsDeleting(false)
   272→          setWordIndex((prev) => (prev + 1) % words.length)
   273→        }
   274→      }
   275→    }, isDeleting ? deletingSpeed : typingSpeed)
   276→
   277→    return () => clearTimeout(timeout)
   278→  }, [text, isDeleting, wordIndex, words, typingSpeed, deletingSpeed, pauseTime])
   279→
   280→  return text
   281→}

   283→// ─── GSAP 3D Tilt Feature Card ───
   284→function FeatureCard({ feature, index }: { feature: typeof features[0]; index: number }) {
   285→  const cardRef = useRef<HTMLDivElement>(null)
   286→  const iconRef = useRef<HTMLDivElement>(null)
   287→  const statRef = useRef<HTMLSpanElement>(null)
   288→  const [hovered, setHovered] = useState(false)
   289→
   290→  useEffect(() => {
   291→    if (!cardRef.current) return
   292→
   293→    const card = cardRef.current
   294→
   295→    const handleMouseMove = (e: MouseEvent) => {
   296→      const rect = card.getBoundingClientRect()
   297→      const x = e.clientX - rect.left - rect.width / 2
   298→      const y = e.clientY - rect.top - rect.height / 2
   299→      const rotateX = (y / rect.height) * -12
   300→      const rotateY = (x / rect.width) * 12
   301→
   302→      gsap.to(card, {
   303→        rotateX,
   304→        rotateY,
   305→        duration: 0.3,
   306→        ease: 'power2.out',
   307→        transformPerspective: 600,
   308→      })
   309→    }
   310→
   311→    const handleMouseLeave = () => {
   312→      gsap.to(card, {
   313→        rotateX: 0,
   314→        rotateY: 0,
   315→        duration: 0.6,
   316→        ease: 'elastic.out(1, 0.5)',
   317→        transformPerspective: 600,
   318→      })
   319→      setHovered(false)
   320→    }
   321→
   322→    const handleMouseEnter = () => {
   323→      setHovered(true)
   324→      if (iconRef.current) {
   325→        gsap.to(iconRef.current, {
   326→          scale: 1.15,
   327→          rotation: 8,
   328→          duration: 0.3,
   329→          ease: 'back.out(2)',
   330→        })
   331→      }
   332→      if (statRef.current) {
   333→        gsap.fromTo(statRef.current,
   334→          { scale: 1 },
   335→          { scale: 1.2, duration: 0.3, yoyo: true, repeat: 1, ease: 'power2.out' }
   336→        )
   337→      }
   338→    }
   339→
   340→    card.addEventListener('mousemove', handleMouseMove)
   341→    card.addEventListener('mouseleave', handleMouseLeave)
   342→    card.addEventListener('mouseenter', handleMouseEnter)
   343→
   344→    return () => {
   345→      card.removeEventListener('mousemove', handleMouseMove)
   346→      card.removeEventListener('mouseleave', handleMouseLeave)
   347→      card.removeEventListener('mouseenter', handleMouseEnter)
   348→    }
   349→  }, [index])
   350→
   351→  // Reset icon on unhover via GSAP
   352→  useEffect(() => {
   353→    if (!hovered && iconRef.current) {
   354→      gsap.to(iconRef.current, {
   355→        scale: 1,
   356→        rotation: 0,
   357→        duration: 0.4,
   358→        ease: 'power2.out',
   359→      })
   360→    }
   361→  }, [hovered])
   362→
   363→  return (
   364→    <div
   365→      ref={cardRef}
   366→      className={`flex items-center gap-3 rounded-2xl px-4 py-3.5 border transition-colors duration-300 cursor-default ${
   367→        hovered
   368→          ? 'bg-white/20 backdrop-blur-md border-white/30 shadow-lg shadow-emerald-500/10'
   369→          : 'bg-white/8 backdrop-blur-sm border-white/12 hover:bg-white/12'
   370→      }`}
   371→      style={{ opacity: 0, transform: 'translateY(20px)' }}
   372→    >
   373→      <div
   374→        ref={iconRef}
   375→        className="w-10 h-10 rounded-xl bg-gradient-to-br from-white/20 to-white/5 flex items-center justify-center shrink-0 border border-white/10"
   376→      >
   377→        <feature.icon className="w-4.5 h-4.5 text-white" />
   378→      </div>
   379→      <div className="flex-1 min-w-0">
   380→        <div className="flex items-center justify-between">
   381→          <p className="text-sm font-semibold text-white">{feature.label}</p>
   382→          <span
   383→            ref={statRef}
   384→            className="text-[10px] font-bold text-emerald-300 bg-emerald-400/20 px-1.5 py-0.5 rounded-full"

   389→        <p className="text-[11px] text-white/50 mt-0.5">{feature.desc}</p>

   391→    </div>
