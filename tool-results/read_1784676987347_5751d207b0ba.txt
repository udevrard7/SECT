    60→function formatDateTime(date: string | Date | null): string {
    61→  if (!date) return '—'
    62→  const d = typeof date === 'string' ? new Date(date) : date
    63→  return d.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit', second: '2-digit' })
    64→}
    65→
    66→function formatRelativeDate(dateStr: string): string {
    67→  const date = new Date(dateStr)
    68→  const now = new Date()
    69→  const diffMs = now.getTime() - date.getTime()
    70→  const diffMin = Math.floor(diffMs / 60000)
    71→  const diffH = Math.floor(diffMin / 60)
    72→  const diffD = Math.floor(diffH / 24)
    73→  if (diffMin < 1) return "À l'instant"
    74→  if (diffMin < 60) return `Il y a ${diffMin} min`
    75→  if (diffH < 24) return `Il y a ${diffH}h`
    76→  if (diffD < 7) return `Il y a ${diffD}j`
    77→  return date.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit' })
    78→}
    79→
    80→function formatTime(date: string): string {
    81→  return new Date(date).toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit', second: '2-digit' })
    82→}

   146→function useDebounce<T>(value: T, delay = 350): T {
   147→  const [debounced, setDebounced] = useState(value)
   148→  useEffect(() => {
   149→    const t = setTimeout(() => setDebounced(value), delay)
   150→    return () => clearTimeout(t)
   151→  }, [value, delay])
   152→  return debounced
   153→}

   812→        </div>
