   125→   125→function formatDate(dateStr: string | null) {
   126→   126→  if (!dateStr) return '—'
   127→   127→  return new Date(dateStr).toLocaleDateString('fr-FR', {
   128→   128→    day: '2-digit',
   129→   129→    month: '2-digit',
   130→   130→    year: 'numeric',
   131→   131→  })
   132→   132→}

   134→   134→function getStatutBadge(statut: string) {
   135→   135→  switch (statut) {
   136→   136→    case 'EN_ATTENTE':
   137→   137→      return (
   138→   138→        <Badge className="bg-warning/10 text-warning border-warning/30">
   139→   139→          <Clock className="h-3 w-3 mr-1" />
   140→   140→          En attente
   141→   141→        </Badge>
   142→   142→      )
   143→   143→    case 'APPROUVE':
   144→   144→      return (
   145→   145→        <Badge className="bg-success/10 text-success-text border-success/30">
   146→   146→          <CheckCircle2 className="h-3 w-3 mr-1" />
   147→   147→          Approuvé
   148→   148→        </Badge>
   149→   149→      )
   150→   150→    case 'REFUSE':
   151→   151→      return (
   152→   152→        <Badge className="bg-destructive/10 text-destructive border-destructive/30">
   153→   153→          <X className="h-3 w-3 mr-1" />
   154→   154→          Refusé
   155→   155→        </Badge>
   156→   156→      )
   157→   157→    case 'EXPIRE':
   158→   158→      return (
   159→   159→        <Badge className="bg-gray-100 text-gray-600 border-gray-200 dark:bg-gray-800 dark:text-gray-400 dark:border-gray-700">
   160→   160→          <AlertCircle className="h-3 w-3 mr-1" />
   161→   161→          Expiré
   162→   162→        </Badge>
   163→   163→      )
   164→   164→    default:
   165→   165→      return <Badge variant="outline">{statut}</Badge>
   166→   166→  }
   167→   167→}

   281→   281→  // BUGFIX B-7 : protection anti double-clic sur les boutons "Annuler" et

   292→   292→
   293→   293→  // ─── Form state ───
   294→   294→  const [formEtablissementId, setFormEtablissementId] = useState('')
   295→   295→  const [formMotif, setFormMotif] = useState('')
   296→   296→  const [formDateDebut, setFormDateDebut] = useState('')

   300→   300→  // ─── Stats ───

   421→   421→      router.push('/dashboard')
   422→   422→    } catch (err) {
   423→   423→      toast.error('Erreur', {
   424→   424→        description: err instanceof Error ? err.message : 'Une erreur est survenue.',
   425→   425→      })
   426→   426→    } finally {
   427→   427→      setAssistanceLoadingId(null)
   428→   428→    }
   429→   429→  }
   430→   430→
   431→   431→  // ─── Submit access request ───
   432→   432→  const handleSubmitRequest = async () => {
   433→   433→    if (!formEtablissementId || !formMotif || !user?.id) {
   434→   434→      toast.error('Champs manquants', {
   435→   435→        description: 'L\'établissement et le motif sont obligatoires.',
   436→   436→      })
   437→   437→      return
   438→   438→    }
   439→   439→
   440→   440→    setIsSubmitting(true)
   441→   441→    try {
   442→   442→      // ACCES-ETABLISSEMENTS-FIX : convertir les dates "YYYY-MM-DD" (input type=date)
