   213→   213→// ─── Typewriter Hook ───
   214→   214→function useTypewriter(words: string[], typingSpeed = 80, deletingSpeed = 40, pauseTime = 2000) {
   215→   215→  const [text, setText] = useState('')
   216→   216→  const [wordIndex, setWordIndex] = useState(0)
   217→   217→  const [isDeleting, setIsDeleting] = useState(false)
   218→   218→
   219→   219→  useEffect(() => {
   220→   220→    const currentWord = words[wordIndex]
   221→   221→
   222→   222→    const timeout = setTimeout(() => {
   223→   223→      if (!isDeleting) {
   224→   224→        setText(currentWord.substring(0, text.length + 1))
   225→   225→        if (text.length === currentWord.length) {
   226→   226→          setTimeout(() => setIsDeleting(true), pauseTime)
   227→   227→        }
   228→   228→      } else {
   229→   229→        setText(currentWord.substring(0, text.length - 1))
   230→   230→        if (text.length === 0) {
   231→   231→          setIsDeleting(false)
   232→   232→          setWordIndex((prev) => (prev + 1) % words.length)
   233→   233→        }
   234→   234→      }
   235→   235→    }, isDeleting ? deletingSpeed : typingSpeed)
   236→   236→
   237→   237→    return () => clearTimeout(timeout)
   238→   238→  }, [text, isDeleting, wordIndex, words, typingSpeed, deletingSpeed, pauseTime])
   239→   239→
   240→   240→  return text
   241→   241→}

   244→   244→function FeatureCard({ feature, index }: { feature: typeof features[0]; index: number }) {
   245→   245→  const [hovered, setHovered] = useState(false)
   246→   246→  const x = useMotionValue(0)
   247→   247→  const y = useMotionValue(0)
   248→   248→  const rotateX = useTransform(y, [-50, 50], [5, -5])
   249→   249→  const rotateY = useTransform(x, [-50, 50], [-5, 5])
   250→   250→
   251→   251→  const handleMouse = useCallback((e: React.MouseEvent<HTMLDivElement>) => {
   252→   252→    const rect = e.currentTarget.getBoundingClientRect()
   253→   253→    x.set(e.clientX - rect.left - rect.width / 2)
   254→   254→    y.set(e.clientY - rect.top - rect.height / 2)
   255→   255→  }, [x, y])
   256→   256→
   257→   257→  return (
   258→   258→    <motion.div
   259→   259→      key={feature.label}
   260→   260→      variants={itemVariants}
   261→   261→      style={{ rotateX, rotateY, transformPerspective: 600 }}
   262→   262→      onMouseMove={handleMouse}
   263→   263→      onMouseEnter={() => setHovered(true)}
   264→   264→      onMouseLeave={() => { setHovered(false); x.set(0); y.set(0) }}
   265→   265→      className={`flex items-center gap-3 rounded-2xl px-4 py-3.5 border transition-all duration-300 cursor-default ${
   266→   266→        hovered
   267→   267→          ? 'bg-white/20 backdrop-blur-md border-white/30 shadow-lg shadow-emerald-500/10'
