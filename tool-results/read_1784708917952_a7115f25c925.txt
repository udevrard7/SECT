   237→func (s *Server) setupRouter(corsOrigins []string, authMiddleware func(http.Handler) http.Handler) {
   238→        r := chi.NewRouter()
   239→
   240→        // Middlewares globaux
   241→        // NOTE: chimw.RealIP retiré — on utilise middleware.GetClientIP() qui lit
   242→        // X-Forwarded-For / X-Real-IP directement (Vercel injecte ces headers).
   243→        // chimw.RealIP modifie r.RemoteAddr ce qui peut causer des conflits.
   244→        r.Use(chimw.RequestID)
   245→        r.Use(chimw.Recoverer)
   246→        r.Use(chimw.Compress(5)) // OPT-8: gzip compression (level 5 = good ratio/speed balance)
   247→        // Monitoring middleware : capture erreurs 5xx + panics → MonitoringEvent
   248→        r.Use(monitoring.Middleware(s.monRecorder, nil))
   249→        r.Use(cors.Handler(cors.Options{
   250→                AllowedOrigins:   corsOrigins,
   251→                AllowedMethods:   []string{"GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"},
   252→                AllowedHeaders:   []string{"Accept", "Authorization", "Content-Type", "Cookie"},
   253→                ExposedHeaders:   []string{"Link", "Set-Cookie"},
   254→                AllowCredentials: true,
   255→                MaxAge:           300,
   256→        }))
   257→
   258→        // Health check (public)
   259→        // /health pour Render healthCheckPath (sans préfixe /api)
   260→        // /api/health pour le rewrite Vercel (préfixe /api/* → Go /api/*)
   261→        r.Get("/health", s.health)
   262→        r.Get("/api/health", s.health)
   263→
   264→        // Auth routes publiques
   265→        r.Group(func(r chi.Router) {
   266→                r.Post("/api/auth/login", s.login)
   267→                r.Post("/api/auth/refresh", s.refresh)
   268→                r.Post("/api/auth/logout", s.logout)
   269→                // SELF-SERVICE RESET (000054) : "mot de passe oublié" — 2 endpoints publics.
   270→                // requestPasswordReset : toujours 200 (anti-énumération).
   271→                // confirmPasswordReset : valide le token + définit le nouveau mot de passe.
   272→                r.Post("/api/auth/password-reset", s.requestPasswordReset)
   273→                r.Post("/api/auth/password-reset/confirm", s.confirmPasswordReset)
   274→                // SECT-B2C-SOUSCRIPTION-AUTO : souscription B2C publique (enseignant
   275→                // freelance). Crée auto étab personnel + user ENSEIGNANT + abonnement.
   276→                r.Post("/api/subscriptions/b2c", s.createB2CSubscription)
   277→                // SECT-B2B-FACTURATION : self-service B2B (inscription établissement).
   278→                r.Post("/api/subscriptions/b2b", s.createB2BSubscription)
   279→                // SECT-B2B-ANTI-ABUS : vérification email + validation admin + liste en attente
   280→                r.Get("/api/b2b/verify-email", s.verifyB2BEmail)
   281→                // SECT-B2B-FACTURATION : paiement Wave capitation B2B.
   282→                r.Post("/api/subscriptions/b2b/{id}/initiate-payment", s.initiateB2BPayment)
   283→                // SECT-B2C-PAIEMENT : confirmation de paiement (V1 simulation, V2 CinetPay).
   284→                // Active l'abonnement EN_ATTENTE_PAIEMENT → ACTIF.
   285→                r.Post("/api/subscriptions/b2c/{id}/confirm-payment", s.confirmB2CPayment)
   286→                // SECT-GENIUSPAY-WAVE : paiement Wave via GeniusPay (V2, remplace la V1 simulation).
   287→                // initiate-payment : crée le paiement Wave → retourne paymentUrl pour redirection.
   288→                // payment-status : polling frontend après retour Wave (active l'abo si completed).
   289→                // renew : renouvellement d'un abonnement ACTIF (prolonge dateFin de 30j).
   290→                r.Post("/api/subscriptions/b2c/{id}/initiate-payment", s.initiateB2CPayment)
   291→                r.Get("/api/subscriptions/b2c/{id}/payment-status", s.getB2CPaymentStatus)
   292→                r.Post("/api/subscriptions/b2c/{id}/renew", s.renewB2CPayment)
   293→                // SECT-B2C-EXPIRE : rétrograde un abonnement Premium EXPIRE vers Prof Solo gratuit.
   294→                r.Post("/api/subscriptions/b2c/{id}/downgrade", s.downgradeB2CToSolo)
   295→                // Webhook GeniusPay (public, signature HMAC vérifiée dans le handler).
   296→                // Reçoit payment.success/failed et active l'abonnement de façon idempotente.
   297→                r.Post("/api/webhooks/geniuspay", s.geniuspayWebhook)
   298→                // SECT-DEMO-REQUEST : demande de démo B2B depuis le landing page (public).
   299→                // Envoie un email à l'admin avec les infos du prospect.
   300→                r.Post("/api/demo-request", s.submitDemoRequest)
   301→                // ACCESS-ASSISTANCE : mode assistance ADMIN (accès temporaire aux pages RESPONSABLE).
   302→                r.With(authMiddleware, middleware.RequireAuth, middleware.RequireRole("ADMIN")).Post("/api/auth/assistance-mode", s.enterAssistanceMode)
   303→                r.With(authMiddleware, middleware.RequireAuth, middleware.RequireRole("ADMIN")).Post("/api/auth/exit-assistance-mode", s.exitAssistanceMode)
   304→        })
   305→
   306→        // E1-INVITATIONS — endpoints publics (pas de RequireAuth).
   307→        // Le token d'invitation sert d'authentification ; la RLS est
   308→        // désactivée pour ces deux opérations (cf. repository/invitation.go).
   309→        // Les routes /verify et /accept sont déclarées AVANT le r.Route
   310→        // authentifié "/api/invitations" pour que chi les traite comme des
   311→        // routes statiques publiques (la sous-route authentifiée
   312→        // "/api/invitations/{id}/..." ne les intercepte pas).
   313→        r.Get("/api/invitations/verify", s.verifyInvitation)
   314→        r.Post("/api/invitations/accept", s.acceptInvitation)
   315→
   316→        // SECT-REG-LINK-B2C-MVP-1 : inscription self-service étudiant via lien direct.
   317→        // Endpoints PUBLICS (pas de RequireAuth — le token du lien est l'auth).
   318→        // /verify retourne le contexte (étab, filière, créateur) pour pré-remplir
   319→        //   le formulaire public /inscription?token=xxx.
   320→        // POST /api/student-signup crée le User ETUDIANT + incrémente useCount
   321→        //   atomiquement via la fonction SQL accept_student_signup (SECURITY DEFINER).
   322→        // Phase 2 (SECT-REG-LINK-PHASE2-BACKEND-1) : rate-limit + Turnstile + audit.
   323→        r.Get("/api/student-signup/verify", s.verifyStudentSignupLink)
   324→        r.Post("/api/student-signup", s.acceptStudentSignup)
   325→
   326→        // SECT-REG-LINK-PHASE2-BACKEND-1 : site key publique Cloudflare Turnstile.
   327→        // Endpoint public (le frontend doit pouvoir récupérer la key avant de rendre
   328→        // le widget, sans auth). Retourne {"siteKey": "..."} ou {"siteKey": ""} si
   329→        // Turnstile n'est pas configuré (le frontend skip le widget dans ce cas).
   330→        r.Get("/api/turnstile/site-key", s.getTurnstileSiteKey)
   331→
   332→        // Certificats verify (public — no auth required for verification)
   333→        r.Get("/api/certificats/verify/{code}", s.verifyCertificat)
   334→
   335→        // Landing demo (public — pour la section "Démo interactive" du landing page).
   336→        // Génère un QCM via le provider IA actif en base. Rate-limité par IP.
   337→        r.Post("/api/landing-demo", s.landingDemo)
   338→
   339→        // Routes authentifiées
   340→        r.Group(func(r chi.Router) {
   341→                r.Use(authMiddleware)
   342→
   343→                // /api/me
   344→                r.With(middleware.RequireAuth).Get("/api/me", s.me)
   345→                r.With(middleware.RequireAuth).Post("/api/auth/change-password", s.changePassword)
   346→
   347→                // /api/users
   348→                // ETUDIANTS-FIX-E2 : route /import déclarée AVANT /{id} pour éviter
   349→                // que chi matche "import" comme un paramètre id. E6 : RequireRole
   350→                // sur mutations (POST/PATCH/DELETE) pour defense-in-depth.
   351→                //
   352→                // SECT-USER-CLEANUP-INFRA-1 : routes /orphans + /cleanup-orphans déclarées
   353→                // AVANT /{id} pour éviter que chi matche "orphans" comme un id. Les 3
   354→                // nouvelles routes (orphans, soft-delete, cleanup-orphans) sont ADMIN-only
   355→                // via RequireRole("ADMIN").
   356→                r.Route("/api/users", func(r chi.Router) {
   357→                        r.Use(middleware.RequireAuth)
   358→                        r.Get("/", s.listUsers)
   359→
   360→                        // SECT-USER-CLEANUP-INFRA-1 : routes admin pour la gestion des orphelins.
   361→                        // Déclarées AVANT /{id} pour éviter le conflit de routing chi.
   362→                        // - GET /orphans : liste les users orphelins (actif=false, sans étab)
   363→                        // - POST /cleanup-orphans : purge manuelle des soft-deleted > 90j
   364→                        // - DELETE /{id}/soft : soft-delete un user (corbeille 90j)
   365→                        r.Group(func(r chi.Router) {
   366→                                r.Use(middleware.RequireRole("ADMIN"))
   367→                                r.Get("/orphans", s.listOrphanUsers)
   368→                                r.Post("/cleanup-orphans", s.hardDeleteOrphans)
   369→                                r.Delete("/{id}/soft", s.softDeleteUser)
   370→                        })
   371→
   372→                        r.Get("/{id}", s.getUser)
   373→                        r.Group(func(r chi.Router) {
   374→                                // SECT-B2C-SELF-SERVICE : ENSEIGNANT dans étab PERSONNEL peut créer
   375→                                // des ÉTUDIANTS (la RLS User_insert limite à role='ETUDIANT').
   376→                                r.Use(middleware.RequireRoleOrPersonalEtab(s.dbPool, "RESPONSABLE", "ADMIN"))
   377→                                r.Post("/", s.createUser)
   378→                                r.Post("/import", s.importUsers) // ETUDIANTS-FIX-E2
   379→                                r.Patch("/{id}", s.updateUser)
   380→                                r.Delete("/{id}", s.deleteUser)
   381→                                // ETUDIANTS-FIX-E10 : dependencies pour preview suppression
   382→                                r.Get("/{id}/dependencies", s.getUserDependencies)
   383→                                // U5 (CRITICAL) : endpoints admin pour reset password + unlock account.
   384→                                // Avant ce fix, un user verrouillé était perdu (lockout permanent de fait).
   385→                                r.Post("/{id}/reset-password", s.resetUserPassword)
   386→                                r.Post("/{id}/unlock", s.unlockUserAccount)
   387→                        })
   388→                })
   389→
   390→                // /api/etablissements
   391→                // SECURITY-FIX (audit 2025) : RequireRole sur les mutations (avant :
   392→                // RequireAuth seul → un ETUDIANT pouvait créer/modifier/supprimer des
   393→                // établissements). Les GET restent ouverts à tous les rôles authentifiés
   394→                // (RLS filtre les lignes visibles).
   395→                r.Route("/api/etablissements", func(r chi.Router) {
