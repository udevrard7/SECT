   187→   187→function renderPieLabel({
   188→   188→  cx,
   189→   189→  cy,
   190→   190→  midAngle,
   191→   191→  outerRadius,
   192→   192→  percent,
   193→   193→  name,
   194→   194→  value,
   195→   195→}: {
   196→   196→  cx: number
   197→   197→  cy: number
   198→   198→  midAngle: number
   199→   199→  innerRadius: number
   200→   200→  outerRadius: number
   201→   201→  percent: number
   202→   202→  name: string
   203→   203→  value: number
   204→   204→}) {
   205→   205→  const RADIAN = Math.PI / 180
   206→   206→  const radius = outerRadius + 28
   207→   207→  const x = cx + radius * Math.cos(-midAngle * RADIAN)
   208→   208→  const y = cy + radius * Math.sin(-midAngle * RADIAN)
   209→   209→
   210→   210→  if (percent < 0.03) return null
   211→   211→
   212→   212→  return (
   213→   213→    <text
   214→   214→      x={x}
   215→   215→      y={y}
   216→   216→      textAnchor={x > cx ? 'start' : 'end'}
   217→   217→      dominantBaseline="central"
   218→   218→      className="text-xs fill-muted-foreground"
   219→   219→    >
   220→   220→      {name} ({value}, {`${(percent * 100).toFixed(0)}%`})
   221→   221→    </text>
   222→   222→  )
   223→   223→}
