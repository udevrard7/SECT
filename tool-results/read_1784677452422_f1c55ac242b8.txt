    29→// ──────────────────────────────────────────────────────────────────────────
    30→// B2 : PUT /api/sessions/{id} — bulk save OU log alerte
    31→// ──────────────────────────────────────────────────────────────────────────
    32→//
    33→// Le frontend passation-page.tsx envoie 2 formes de body sur cette route :
    34→//   1. { reponses: { questionId: contenu, ... } }  → bulk save (auto-save 30s)
    35→//   2. { alerte: { type, details, penalite } }      → log anti-fraude
    36→//
    37→// On dispatche selon la présence du champ "reponses" ou "alerte".
    38→
    39→func (s *Server) updateSessionBulk(w http.ResponseWriter, r *http.Request) {
    40→        claims, ok := middleware.ClaimsFromContext(r.Context())
    41→        if !ok || claims.UserID == "" {
    42→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
    43→                return
    44→        }
    45→        sessionID := chi.URLParam(r, "id")
    46→        if sessionID == "" {
    47→                writeJSONError(w, http.StatusBadRequest, "id session requis")
    48→                return
    49→        }
    50→
    51→        // Decoder le body de façon permissive (champs optionnels)
    52→        var body map[string]json.RawMessage
    53→        if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
    54→                writeJSONError(w, http.StatusBadRequest, "corps de requête invalide")
    55→                return
    56→        }
    57→
    58→        // Cas 1 : bulk save reponses
    59→        if rawReponses, ok := body["reponses"]; ok {
    60→                var reponses map[string]string
    61→                if err := json.Unmarshal(rawReponses, &reponses); err != nil {
    62→                        writeJSONError(w, http.StatusBadRequest, "format reponses invalide")
    63→                        return
    64→                }
    65→
    66→                // OPT-1 + OPT-2 : écriture via cache RAM + ownership cache.
    67→                //
    68→                // AVANT : chaque auto-save (30s) faisait 1 WithTx ownership + 1 BulkSaveReponses
    69→                // = 2 transactions DB par étudiant par save. Pour 5000 étudiants = 333 tx/s.
    70→                //
    71→                // APRÈS : l'auto-save écrit en RAM (< 1ms, 0 tx DB). Le worker goroutine
    72→                // synchronise vers Neon toutes les 30s (1 tx par session). L'ownership est
    73→                // vérifiée en cache après la première vérification DB.
    74→                //
    75→                // Pour 5000 étudiants : 5000 × 1 tx / 30s = 167 tx/s au lieu de 333 tx/s (-50%).
    76→                // Avec OPT-3 (batch flush) : ~10 tx/s au lieu de 167 tx/s (-95%).
    77→
    78→                // OWNERSHIP-CACHE-2 : vérifier ownership via le cache en mémoire d'abord,
    79→                // puis via DB si pas encore vérifié (même pattern que saveReponse).
    80→                if s.sessionCache != nil && s.sessionCache.IsOwnershipVerified(sessionID, claims.UserID) {
    81→                        // Ownership déjà vérifiée en cache → skip DB query (-1 tx par save)
    82→                } else {
    83→                        // Première vérification : requêter la DB
    84→                        var etudiantID string
    85→                        found := false
    86→                        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
    87→                                err := tx.QueryRow(r.Context(),
    88→                                        `SELECT "etudiantId" FROM "SessionPassation" WHERE "id" = $1`,
    89→                                        sessionID).Scan(&etudiantID)
    90→                                if err == nil {
    91→                                        found = true
    92→                                }
    93→                                return err
    94→                        })
    95→                        if !found || etudiantID != claims.UserID {
    96→                                writeJSONError(w, http.StatusNotFound, "session introuvable ou accès refusé")
    97→                                return
    98→                        }
    99→                        // Cacher l'ownership vérifiée pour les prochains saves
   100→                        if s.sessionCache != nil {
   101→                                s.sessionCache.MarkOwnershipVerified(sessionID, claims.UserID)
   102→                        }
   103→                }
   104→
   105→                // OPT-1 : écrire en RAM au lieu de DB directe.
   106→                // Le cache merge les réponses (map[string]string) et marque la session dirty.
   107→                // Le worker goroutine (FlushSessionToNeon toutes les 30s) synchronisera vers Neon.
   108→                // Le submit force un flush immédiat avant la soumission finale.
   109→                if s.sessionCache != nil {
   110→                        s.sessionCache.SaveAnswers(sessionID, "", claims.UserID, reponses)
   111→                }
   112→
   113→                w.Header().Set("Content-Type", "application/json")
   114→                json.NewEncoder(w).Encode(map[string]any{
   115→                        "saved": true,
   116→                        "count": len(reponses),
   117→                })
   118→                return
   119→        }
   120→
   121→        // Cas 2 : log alerte
   122→        if rawAlerte, ok := body["alerte"]; ok {
   123→                var alerte struct {
   124→                        Type      string  `json:"type"`
   125→                        Details   string  `json:"details"`
   126→                        Penalite  float64 `json:"penalite"`
   127→                }
   128→                if err := json.Unmarshal(rawAlerte, &alerte); err != nil {
   129→                        writeJSONError(w, http.StatusBadRequest, "format alerte invalide")
   130→                        return
   131→                }
   132→                if alerte.Type == "" {
   133→                        writeJSONError(w, http.StatusBadRequest, "type alerte requis")
   134→                        return
   135→                }
   136→
   137→                // OWNERSHIP-CACHE-2 : vérifier ownership via le cache d'abord (comme reponses).
   138→                // Les alertes sont moins fréquentes mais pendant un examen avec proctoring,
   139→                // elles peuvent être nombreuses (changement d'onglet, copier-coller...).
   140→                if s.sessionCache != nil && s.sessionCache.IsOwnershipVerified(sessionID, claims.UserID) {
   141→                        // Ownership déjà vérifiée → skip DB
   142→                } else {
   143→                        var etudiantID string
   144→                        found := false
   145→                        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   146→                                err := tx.QueryRow(r.Context(),
   147→                                        `SELECT "etudiantId" FROM "SessionPassation" WHERE "id" = $1`,
   148→                                        sessionID).Scan(&etudiantID)
   149→                                if err == nil {
   150→                                        found = true
   151→                                }
   152→                                return err
   153→                        })
   154→                        if !found || etudiantID != claims.UserID {
   155→                                writeJSONError(w, http.StatusNotFound, "session introuvable ou accès refusé")
   156→                                return
   157→                        }
   158→                        if s.sessionCache != nil {
   159→                                s.sessionCache.MarkOwnershipVerified(sessionID, claims.UserID)
   160→                        }
   161→                }
   162→
   163→                // AddAlerte (RLS off dans le repo — conforme au pattern existant)
   164→                alerteInput := domain.AlerteInput{
   165→                        Type:     alerte.Type,
   166→                        Details:  alerte.Details,
   167→                }
   168→                _ = s.sessionUC.AddAlerte(r.Context(), claims, sessionID, alerte.Penalite, alerteInput)
   169→
   170→                // OPT-7 : push WebSocket aux surveillants — alerte déclenchée.
   171→                // On récupère l'epreuveId de la session pour le broadcast ciblé.
   172→                var alerteEpreuveID string
   173→                _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   174→                        _ = tx.QueryRow(r.Context(),
   175→                                `SELECT "epreuveId" FROM "SessionPassation" WHERE "id" = $1`,
   176→                                sessionID).Scan(&alerteEpreuveID)
   177→                        return nil
   178→                })
   179→                s.surveillanceHub.BroadcastEvent("ALERT_TRIGGERED", alerteEpreuveID, map[string]any{
   180→                        "sessionId":  sessionID,
   181→                        "epreuveId":  alerteEpreuveID,
   182→                        "etudiantId": claims.UserID,
   183→                        "alertType":  alerte.Type,
   184→                        "penalite":   alerte.Penalite,
   185→                })
   186→
   187→                w.Header().Set("Content-Type", "application/json")
   188→                json.NewEncoder(w).Encode(map[string]any{
   189→                        "logged": true,
   190→                })
   191→                return
   192→        }
   193→
   194→        writeJSONError(w, http.StatusBadRequest, "body doit contenir 'reponses' ou 'alerte'")
   195→}

   197→// ──────────────────────────────────────────────────────────────────────────
   198→// B3 : POST /api/sessions/{id}/capture — persiste capture écran anti-fraude
   199→// ──────────────────────────────────────────────────────────────────────────
   200→//
   201→// Le frontend envoie { image: "data:image/jpeg;base64,..." }.
   202→// On decode le base64, on upload l'image vers R2, et on stocke les
   203→// métadonnées dans la table SessionCapture. On conserve aussi l'event
   204→// CAPTURE dans logEvents pour compatibilité ascendante.
   205→//
   206→// Si R2 n'est pas disponible (mode dev), on stocke juste les métadonnées
   207→// avec r2Key vide et l'image comme miniature base64.
   208→
   209→func (s *Server) captureSession(w http.ResponseWriter, r *http.Request) {
   210→        claims, ok := middleware.ClaimsFromContext(r.Context())
   211→        if !ok || claims.UserID == "" {
   212→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
   213→                return
   214→        }
   215→        sessionID := chi.URLParam(r, "id")
   216→        if sessionID == "" {
   217→                writeJSONError(w, http.StatusBadRequest, "id session requis")
   218→                return
   219→        }
   220→
   221→        var input struct {
   222→                Image string `json:"image"`
   223→        }
   224→        if err := json.NewDecoder(r.Body).Decode(&input); err != nil {
   225→                writeJSONError(w, http.StatusBadRequest, "corps de requête invalide")
   226→                return
   227→        }
   228→        if input.Image == "" {
   229→                writeJSONError(w, http.StatusBadRequest, "image requise")
   230→                return
   231→        }
   232→
   233→        // Vérifier ownership + récupérer epreuveId
   234→        var etudiantID, epreuveID string
   235→        found := false
   236→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   237→                err := tx.QueryRow(r.Context(),
   238→                        `SELECT "etudiantId", "epreuveId" FROM "SessionPassation" WHERE "id" = $1`,
   239→                        sessionID).Scan(&etudiantID, &epreuveID)
   240→                if err == nil {
   241→                        found = true
   242→                }
   243→                return err
   244→        })
   245→        if !found || etudiantID != claims.UserID {
   246→                writeJSONError(w, http.StatusNotFound, "session introuvable ou accès refusé")
   247→                return
   248→        }
   249→
   250→        // Décoder le base64 de l'image (accepte data:image/jpeg;base64,... ou base64 brut)
   251→        imageData := input.Image
   252→        if idx := strings.Index(imageData, ";base64,"); idx >= 0 {
   253→                imageData = imageData[idx+8:]
   254→        }
   255→        imageBytes, err := base64.StdEncoding.DecodeString(imageData)
   256→        if err != nil {
   257→                // Essayer RawStdEncoding au cas où
   258→                imageBytes, err = base64.RawStdEncoding.DecodeString(imageData)
   259→                if err != nil {
   260→                        writeJSONError(w, http.StatusBadRequest, "image base64 invalide")
   261→                        return
   262→                }
   263→        }
   264→
   265→        // Calculer SHA-256 pour intégrité
   266→        hash := sha256.Sum256(imageBytes)
   267→        imageHash := fmt.Sprintf("%x", hash)
   268→
   269→        // Déterminer le prochain captureIndex pour cette session
   270→        var captureIndex int
   271→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   272→                err := tx.QueryRow(r.Context(),
   273→                        `SELECT COALESCE(MAX("captureIndex"), 0) + 1 FROM "SessionCapture" WHERE "sessionId" = $1`,
   274→                        sessionID).Scan(&captureIndex)
   275→                return err
   276→        })
   277→        if captureIndex == 0 {
   278→                captureIndex = 1
   279→        }
   280→
   281→        // Upload vers R2 si disponible
   282→        captureID := uuid.New().String()
   283→        r2Key := ""
   284→        var fileSize int
   285→
   286→        if s.storage != nil {
   287→                // Générer la clé R2 : captures/{sessionID}/{timestamp}.jpg
   288→                ts := time.Now().UTC().Format("20060102-150405.000")
   289→                r2Key = fmt.Sprintf("captures/%s/%s.jpg", sessionID, ts)
   290→                fileSize = len(imageBytes)
   291→
   292→                _, err := s.storage.Upload(r.Context(), domain.StorageObject{
   293→                        Key:           r2Key,
   294→                        Content:       imageBytes,
   295→                        ContentType:   "image/jpeg",
   296→                        ContentLength: int64(fileSize),
   297→                })
   298→                if err != nil {
   299→                        // Log l'erreur mais ne pas bloquer la capture — on stocke juste les métadonnées
   300→                        fmt.Printf("[captureSession] R2 upload failed: %v\n", err)
   301→                        r2Key = ""
   302→                        fileSize = 0
   303→                }
   304→        }
   305→
   306→        // Insérer les métadonnées dans SessionCapture
   307→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   308→                _, err := tx.Exec(r.Context(),
   309→                        `INSERT INTO "SessionCapture" ("id", "sessionId", "etudiantId", "epreuveId", "r2Key", "imageHash", "fileSize", "captureIndex", "createdAt")
   310→                         VALUES ($1, $2, $3, $4, $5, $6, $7, $8, NOW())`,
   311→                        captureID, sessionID, etudiantID, epreuveID, r2Key, imageHash, fileSize, captureIndex)
   312→                return err
   313→        })
   314→
   315→        // Logger la capture dans logEvents (append un event CAPTURE)
   316→        // Compatibilité ascendante : le logEvent CAPTURE reste présent.
   317→        alerteInput := domain.AlerteInput{
   318→                Type:    "CAPTURE",
   319→                Details: fmt.Sprintf("Capture écran #%d prise (hash: %s...)", captureIndex, imageHash[:16]),
   320→        }
   321→        _ = s.sessionUC.AddAlerte(r.Context(), claims, sessionID, 0, alerteInput)
   322→
   323→        w.Header().Set("Content-Type", "application/json")
   324→        json.NewEncoder(w).Encode(map[string]any{
   325→                "captured":     true,
   326→                "captureId":    captureID,
   327→                "captureIndex": captureIndex,
   328→        })
   329→}

   331→// ──────────────────────────────────────────────────────────────────────────
   332→// GET /api/sessions/{id}/captures — liste des captures d'une session
   333→// ──────────────────────────────────────────────────────────────────────────
   334→//
   335→// Retourne la liste des captures avec URL présignée R2 (valide 5 minutes).
   336→// Accès : ENSEIGNANT/ADMIN/RESPONSABLE ou l'étudiant propriétaire.
   337→
   338→func (s *Server) listSessionCaptures(w http.ResponseWriter, r *http.Request) {
   339→        claims, ok := middleware.ClaimsFromContext(r.Context())
   340→        if !ok || claims.UserID == "" {
   341→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
   342→                return
   343→        }
   344→        sessionID := chi.URLParam(r, "id")
   345→        if sessionID == "" {
   346→                writeJSONError(w, http.StatusBadRequest, "id session requis")
   347→                return
   348→        }
   349→
   350→        // Vérifier l'accès : l'étudiant propriétaire OU enseignant/admin/responsable
   351→        role := domain.Role(claims.Role)
   352→        isStaff := role == domain.RoleEnseignant || role == domain.RoleAdmin || role == domain.RoleResponsable
   353→
   354→        var etudiantID string
   355→        found := false
   356→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   357→                err := tx.QueryRow(r.Context(),
   358→                        `SELECT "etudiantId" FROM "SessionPassation" WHERE "id" = $1`,
   359→                        sessionID).Scan(&etudiantID)
   360→                if err == nil {
   361→                        found = true
   362→                }
   363→                return err
   364→        })
   365→        if !found {
   366→                writeJSONError(w, http.StatusNotFound, "session introuvable")
   367→                return
   368→        }
   369→        if !isStaff && etudiantID != claims.UserID {
   370→                writeJSONError(w, http.StatusForbidden, "accès refusé")
   371→                return
   372→        }
   373→
   374→        // Récupérer les captures
   375→        type captureRow struct {
   376→                ID           string     `json:"id"`
   377→                R2Key        string     `json:"r2Key"`
   378→                ImageHash    *string    `json:"imageHash,omitempty"`
   379→                FileSize     *int       `json:"fileSize,omitempty"`
   380→                CaptureIndex int        `json:"captureIndex"`
   381→                CreatedAt    time.Time  `json:"createdAt"`
   382→        }
   383→
   384→        var captures []captureRow
   385→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   386→                rows, err := tx.Query(r.Context(),
   387→                        `SELECT "id", "r2Key", "imageHash", "fileSize", "captureIndex", "createdAt"
   388→                         FROM "SessionCapture"
   389→                         WHERE "sessionId" = $1
   390→                         ORDER BY "captureIndex" ASC`,
   391→                        sessionID)
   392→                if err != nil {
   393→                        return err
   394→                }
   395→                defer rows.Close()
   396→                for rows.Next() {
   397→                        var c captureRow
   398→                        if err := rows.Scan(&c.ID, &c.R2Key, &c.ImageHash, &c.FileSize, &c.CaptureIndex, &c.CreatedAt); err != nil {
   399→                                return err
   400→                        }
   401→                        captures = append(captures, c)
   402→                }
   403→                return rows.Err()
   404→        })
   405→
   406→        // Enrichir avec URL présignées si R2 est disponible
   407→        type captureResponse struct {
   408→                ID           string  `json:"id"`
   409→                URL          string  `json:"url,omitempty"`
   410→                R2Key        string  `json:"r2Key,omitempty"`
   411→                ImageHash    *string `json:"imageHash,omitempty"`
   412→                FileSize     *int    `json:"fileSize,omitempty"`
   413→                CaptureIndex int     `json:"captureIndex"`
   414→                CreatedAt    string  `json:"createdAt"`
   415→        }
   416→
   417→        result := make([]captureResponse, 0, len(captures))
   418→        for _, c := range captures {
   419→                resp := captureResponse{
   420→                        ID:           c.ID,
   421→                        R2Key:        c.R2Key,
   422→                        ImageHash:    c.ImageHash,
   423→                        FileSize:     c.FileSize,
   424→                        CaptureIndex: c.CaptureIndex,
   425→                        CreatedAt:    c.CreatedAt.UTC().Format(time.RFC3339),
   426→                }
   427→                // Générer URL présignée (5 min = 300s) si R2 disponible et clé présente
   428→                if s.storage != nil && c.R2Key != "" {
   429→                        url, err := s.storage.PresignURL(r.Context(), c.R2Key, 300)
   430→                        if err == nil {
   431→                                resp.URL = url
   432→                        }
   433→                }
   434→                result = append(result, resp)
   435→        }
   436→
   437→        w.Header().Set("Content-Type", "application/json")
   438→        json.NewEncoder(w).Encode(map[string]any{
   439→                "captures": result,
   440→        })
   441→}

   443→// ──────────────────────────────────────────────────────────────────────────
   444→// B4 : GET /api/epreuves/auto-close?epreuveId=X — détection clôture auto
   445→// ──────────────────────────────────────────────────────────────────────────
   446→//
   447→// Le frontend poll toutes les 15s pendant l'examen pour détecter :
   448→//   - si l'épreuve a été clôturée manuellement (clotureeAt non null)
   449→//   - si la dateFin est dépassée (avec période de grâce)
   450→//
   451→// Response: { isClosed, raisonCloture, inGracePeriod, gracePeriodEndsAt }
   452→
   453→func (s *Server) epreuveAutoClose(w http.ResponseWriter, r *http.Request) {
   454→        claims, ok := middleware.ClaimsFromContext(r.Context())
   455→        if !ok || claims.UserID == "" {
   456→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
   457→                return
   458→        }
   459→        epreuveID := r.URL.Query().Get("epreuveId")
   460→        if epreuveID == "" {
   461→                writeJSONError(w, http.StatusBadRequest, "epreuveId requis")
   462→                return
   463→        }
   464→
   465→        var (
   466→                clotureeAt     *time.Time
   467→                raisonCloture  *string
   468→                dateFin        *time.Time
   469→                delaiGrace     int
   470→                statut         string
   471→        )
   472→        found := false
   473→        _ = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   474→                err := tx.QueryRow(r.Context(), `
   475→                        SELECT "clotureeAt", "raisonCloture", "dateFin", COALESCE("delaiGrace", 0), "statut"::text
   476→                        FROM "Epreuve"
   477→                        WHERE "id" = $1 AND "deletedAt" IS NULL
   478→                `, epreuveID).Scan(&clotureeAt, &raisonCloture, &dateFin, &delaiGrace, &statut)
   479→                if err == nil {
   480→                        found = true
   481→                }
   482→                return err
   483→        })
   484→        if !found {
   485→                writeJSONError(w, http.StatusNotFound, "épreuve introuvable")
   486→                return
   487→        }
   488→
   489→        now := time.Now().UTC()
   490→        isClosed := false
   491→        inGracePeriod := false
   492→        var gracePeriodEndsAt *time.Time
   493→
   494→        if clotureeAt != nil {
   495→                // Clôturée manuellement
   496→                isClosed = true
   497→        } else if statut == "CLOTUREE" {
   498→                isClosed = true
   499→        } else if dateFin != nil && now.After(*dateFin) {
   500→                // Date fin dépassée — vérifier période de grâce
   501→                graceEnd := dateFin.Add(time.Duration(delaiGrace) * time.Minute)
   502→                if now.Before(graceEnd) {
   503→                        inGracePeriod = true
   504→                        gracePeriodEndsAt = &graceEnd
   505→                } else {
   506→                        isClosed = true
   507→                        rc := "Délai dépassé (fin de période + grâce)"
   508→                        raisonCloture = &rc
   509→
   510→                        // BUGFIX (CLOTURE-AUTO) : l'ancien code détectait isClosed=true
