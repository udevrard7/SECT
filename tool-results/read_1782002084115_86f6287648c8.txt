    42→    42→/* ─── Magnetic Button ─── */
    43→    43→function MagneticButton({
    44→    44→  children,
    45→    45→  className = '',
    46→    46→  ...props
    47→    47→}: React.ComponentProps<typeof Button> & { children: React.ReactNode }) {
    48→    48→  const ref = useRef<HTMLButtonElement>(null)
    49→    49→  const xTo = useRef<gsap.QuickToFunc | null>(null)
    50→    50→  const yTo = useRef<gsap.QuickToFunc | null>(null)
    51→    51→
    52→    52→  useEffect(() => {
    53→    53→    if (!ref.current) return
    54→    54→    xTo.current = gsap.quickTo(ref.current, 'x', { duration: 0.3, ease: 'power2.out' })
    55→    55→    yTo.current = gsap.quickTo(ref.current, 'y', { duration: 0.3, ease: 'power2.out' })
    56→    56→    return () => {
    57→    57→      xTo.current = null
    58→    58→      yTo.current = null
    59→    59→    }
    60→    60→  }, [])
    61→    61→
    62→    62→  const handleMouseMove = useCallback(
    63→    63→    (e: React.MouseEvent<HTMLButtonElement>) => {
    64→    64→      if (!ref.current || !xTo.current || !yTo.current) return
    65→    65→      const rect = ref.current.getBoundingClientRect()
    66→    66→      const x = e.clientX - rect.left - rect.width / 2
    67→    67→      const y = e.clientY - rect.top - rect.height / 2
    68→    68→      xTo.current(x * 0.3)
    69→    69→      yTo.current(y * 0.3)
    70→    70→    },
    71→    71→    []
    72→    72→  )
    73→    73→
    74→    74→  const handleMouseLeave = useCallback(() => {
    75→    75→    xTo.current?.(0)
    76→    76→    yTo.current?.(0)
    77→    77→  }, [])
    78→    78→
    79→    79→  return (
    80→    80→    <Button
    81→    81→      ref={ref}
    82→    82→      className={className}
    83→    83→      onMouseMove={handleMouseMove}
    84→    84→      onMouseLeave={handleMouseLeave}
    85→    85→      {...props}
    86→    86→    >
    87→    87→      {children}
    88→    88→    </Button>
    89→    89→  )
    90→    90→}

   268→   268→function HeroSection({ onDemo }: { onDemo: () => void }) {
   269→   269→  const sectionRef = useRef<HTMLDivElement>(null)
   270→   270→  const headlineRef = useRef<HTMLHeadingElement>(null)
   271→   271→  const subRef = useRef<HTMLParagraphElement>(null)

   273→   273→  const imageRef = useRef<HTMLDivElement>(null)
   274→   274→  const typingRef = useRef<HTMLSpanElement>(null)
   275→   275→
   276→   276→  useEffect(() => {
   277→   277→    const ctx = gsap.context(() => {
   278→   278→      // Split headline into words and animate
   279→   279→      if (headlineRef.current) {
   280→   280→        const words = headlineRef.current.querySelectorAll('.hero-word')
   281→   281→        gsap.fromTo(
   282→   282→          words,
   283→   283→          { y: 80, opacity: 0, rotateX: -40 },
   284→   284→          {
   285→   285→            y: 0,
   286→   286→            opacity: 1,
   287→   287→            rotateX: 0,
   288→   288→            duration: 1.2,
   289→   289→            stagger: 0.08,
   290→   290→            ease: 'power3.out',
   291→   291→            delay: 0.3,
   292→   292→          }
   293→   293→        )
   294→   294→      }
   295→   295→
   296→   296→      // Subtitle fade in
   297→   297→      if (subRef.current) {
   298→   298→        gsap.fromTo(
   299→   299→          subRef.current,
   300→   300→          { opacity: 0, y: 20 },
   301→   301→          { opacity: 1, y: 0, duration: 1, ease: 'power2.out', delay: 1.2 }
   302→   302→        )
   303→   303→      }
   304→   304→
   305→   305→      // CTA buttons
   306→   306→      if (ctaRef.current) {
   307→   307→        gsap.fromTo(
   308→   308→          ctaRef.current.children,
   309→   309→          { opacity: 0, y: 20 },
   310→   310→          { opacity: 1, y: 0, duration: 0.8, stagger: 0.15, ease: 'power2.out', delay: 1.5 }
   311→   311→        )
   312→   312→      }
   313→   313→
   314→   314→      // Dashboard image parallax
   315→   315→      if (imageRef.current) {
   316→   316→        gsap.fromTo(
   317→   317→          imageRef.current,
   318→   318→          { y: 60, opacity: 0, scale: 0.95 },
   319→   319→          { y: 0, opacity: 1, scale: 1, duration: 1.4, ease: 'power3.out', delay: 1.8 }
   320→   320→        )
   321→   321→
   322→   322→        // Parallax on scroll
   323→   323→        gsap.to(imageRef.current, {
   324→   324→          y: -40,
   325→   325→          ease: 'none',
   326→   326→          scrollTrigger: {
   327→   327→            trigger: sectionRef.current,
   328→   328→            start: 'top top',
   329→   329→            end: 'bottom top',
   330→   330→            scrub: 1,
   331→   331→          },
   332→   332→        })
   333→   333→      }
   334→   334→    }, sectionRef)
   335→   335→
   336→   336→    // 3D tilt on mouse move
   337→   337→    const handleMouseMove = (e: MouseEvent) => {
   338→   338→      if (!imageRef.current) return
   339→   339→      const rect = imageRef.current.getBoundingClientRect()
   340→   340→      const centerX = rect.left + rect.width / 2
   341→   341→      const centerY = rect.top + rect.height / 2
   342→   342→      const rotateY = ((e.clientX - centerX) / (rect.width / 2)) * 5
   343→   343→      const rotateX = -((e.clientY - centerY) / (rect.height / 2)) * 5
   344→   344→      gsap.to(imageRef.current, {
   345→   345→        rotateX,
   346→   346→        rotateY,
   347→   347→        duration: 0.5,
   348→   348→        ease: 'power2.out',
   349→   349→        transformPerspective: 1200,
   350→   350→      })
   351→   351→    }
   352→   352→
   353→   353→    const handleMouseLeave = () => {
   354→   354→      if (!imageRef.current) return
   355→   355→      gsap.to(imageRef.current, {
   356→   356→        rotateX: 0,
   357→   357→        rotateY: 0,
   358→   358→        duration: 0.8,
   359→   359→        ease: 'power2.out',
   360→   360→      })
   361→   361→    }
   362→   362→
   363→   363→    window.addEventListener('mousemove', handleMouseMove)
   364→   364→    imageRef.current?.addEventListener('mouseleave', handleMouseLeave)
   365→   365→
   366→   366→    return () => {
   367→   367→      ctx.revert()
   368→   368→      window.removeEventListener('mousemove', handleMouseMove)
   369→   369→    }
   370→   370→  }, [])
   371→   371→
   372→   372→  // Typing effect

   389→   389→        charIndex++

   391→   391→          typingRef.current.textContent = currentPhrase.slice(0, charIndex)
   392→   392→        }

   394→   394→          isDeleting = true
   395→   395→          timeout = setTimeout(type, 2000)
   396→   396→          return
   397→   397→        }
   398→   398→        timeout = setTimeout(type, 60)
   399→   399→      } else {
   400→   400→        charIndex--
   401→   401→        if (typingRef.current) {
   402→   402→          typingRef.current.textContent = currentPhrase.slice(0, charIndex)
   403→   403→        }
   404→   404→        if (charIndex === 0) {
   405→   405→          isDeleting = false
   406→   406→          phraseIndex = (phraseIndex + 1) % phrases.length
   407→   407→          timeout = setTimeout(type, 500)
   408→   408→          return

   417→   417→
   418→   418→  const headlineWords = ["L'examen", 'reinvente', 'par', "l'Intelligence", 'Artificielle']
   419→   419→
   420→   420→  return (

   427→   427→        {/* Badge */}
   428→   428→        <motion.div

   431→   431→          transition={{ delay: 0.1, duration: 0.6 }}
   432→   432→          className="inline-flex items-center gap-2 px-4 py-1.5 mb-8 rounded-full border border-white/[0.08] bg-white/[0.03] backdrop-blur-sm"
   433→   433→        >

   441→   441→        <h1
