     2→-- Migration 000007 — Policies RLS par table (Option A)
     3→-- ============================================================
     4→--
     5→-- Policies organisées par groupe d'accès. Chaque policy utilise les
     6→-- fonctions helper définies dans 000006 :
     7→--   current_user_id(), current_role_claim(), current_etablissement_id()
     8→--   is_admin(), is_responsable(), is_enseignant(), is_etudiant()
     9→--   belongs_to_etablissement(eta_id), admin_has_etablissement_access(eta_id)
    10→--
    11→-- Rappel PostgreSQL : pour une même commande (SELECT/INSERT/...), les
    12→-- policies multiples sont combinées par OR. Entre commandes différentes
    13→-- (ex: SELECT vs INSERT), elles sont indépendantes.
    14→--
    15→-- Convention de nommage : "{Table}_{operation}" (ex: "User_select")

   322→-- Question (créée par un enseignant — auteurId)
   323→CREATE POLICY "Question_select" ON "Question"
   324→  FOR SELECT TO neondb_owner
   325→  USING (
   326→    -- ENSEIGNANT : ses questions (auteurId) + questions partagées (auteurId IS NULL)
   327→    (is_enseignant() AND ("auteurId" = current_user_id() OR "auteurId" IS NULL))
   328→    -- RESPONSABLE : questions de son établissement (via filière de l'épreuve)
   329→    OR (is_responsable() AND EXISTS (
   330→      SELECT 1 FROM "EpreuveQuestion" eq
   331→      JOIN "Epreuve" e ON e."id" = eq."epreuveId"
   332→      JOIN "Filiere" f ON f."id" = e."filiereId"
   333→      WHERE eq."questionId" = "Question"."id"
   334→        AND f."etablissementId" = current_etablissement_id()
   335→    ))
   336→    -- ETUDIANT : questions des épreuves où il a une session
   337→    OR (is_etudiant() AND EXISTS (
   338→      SELECT 1 FROM "EpreuveQuestion" eq
   339→      JOIN "SessionPassation" sp ON sp."epreuveId" = eq."epreuveId"
   340→      WHERE eq."questionId" = "Question"."id"
   341→        AND sp."etudiantId" = current_user_id()
   342→    ))
   343→    -- ADMIN : via accès établissement
   344→    OR (is_admin() AND EXISTS (
   345→      SELECT 1 FROM "EpreuveQuestion" eq
   346→      JOIN "Epreuve" e ON e."id" = eq."epreuveId"
   347→      JOIN "Filiere" f ON f."id" = e."filiereId"
   348→      WHERE eq."questionId" = "Question"."id"
   349→        AND admin_has_etablissement_access(f."etablissementId")
   350→    ))
   351→  );
