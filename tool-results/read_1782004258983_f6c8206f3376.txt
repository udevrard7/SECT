    45→/* ─── Magnetic Button ─── */
    46→function MagneticButton({
    47→  children,
    48→  className = '',
    49→  ...props
    50→}: React.ComponentProps<typeof Button> & { children: React.ReactNode }) {
    51→  const ref = useRef<HTMLButtonElement>(null)
    52→  const xTo = useRef<gsap.QuickToFunc | null>(null)
    53→  const yTo = useRef<gsap.QuickToFunc | null>(null)
    54→
    55→  useEffect(() => {
    56→    if (!ref.current) return
    57→    xTo.current = gsap.quickTo(ref.current, 'x', { duration: 0.3, ease: 'power2.out' })
    58→    yTo.current = gsap.quickTo(ref.current, 'y', { duration: 0.3, ease: 'power2.out' })
    59→    return () => {
    60→      xTo.current = null
    61→      yTo.current = null
    62→    }
    63→  }, [])
    64→
    65→  const handleMouseMove = useCallback(
    66→    (e: React.MouseEvent<HTMLButtonElement>) => {
    67→      if (!ref.current || !xTo.current || !yTo.current) return
    68→      const rect = ref.current.getBoundingClientRect()
    69→      const x = e.clientX - rect.left - rect.width / 2
    70→      const y = e.clientY - rect.top - rect.height / 2
    71→      xTo.current(x * 0.3)
    72→      yTo.current(y * 0.3)
    73→    },
    74→    []
    75→  )
    76→
    77→  const handleMouseLeave = useCallback(() => {
    78→    xTo.current?.(0)
    79→    yTo.current?.(0)
    80→  }, [])
    81→
    82→  return (
    83→    <Button
    84→      ref={ref}
    85→      className={className}
    86→      onMouseMove={handleMouseMove}
    87→      onMouseLeave={handleMouseLeave}
    88→      {...props}
    89→    >
    90→      {children}
    91→    </Button>
    92→  )
    93→}

   276→  const imageRef = useRef<HTMLDivElement>(null)
   277→  const typingRef = useRef<HTMLSpanElement>(null)
   278→
   279→  useEffect(() => {
   280→    const ctx = gsap.context(() => {
   281→      // Split headline into words and animate
   282→      if (headlineRef.current) {
   283→        const words = headlineRef.current.querySelectorAll('.hero-word')
   284→        gsap.fromTo(
   285→          words,
   286→          { y: 80, opacity: 0, rotateX: -40 },
   287→          {
   288→            y: 0,
   289→            opacity: 1,
   290→            rotateX: 0,
   291→            duration: 1.2,
   292→            stagger: 0.08,
   293→            ease: 'power3.out',
   294→            delay: 0.3,
   295→          }
   296→        )
   297→      }
   298→
   299→      // Subtitle fade in
   300→      if (subRef.current) {
   301→        gsap.fromTo(
   302→          subRef.current,
   303→          { opacity: 0, y: 20 },
   304→          { opacity: 1, y: 0, duration: 1, ease: 'power2.out', delay: 1.2 }
   305→        )
   306→      }
   307→
   308→      // CTA buttons
   309→      if (ctaRef.current) {
   310→        gsap.fromTo(
   311→          ctaRef.current.children,
   312→          { opacity: 0, y: 20 },
   313→          { opacity: 1, y: 0, duration: 0.8, stagger: 0.15, ease: 'power2.out', delay: 1.5 }
   314→        )
   315→      }
   316→
   317→      // Dashboard image parallax
   318→      if (imageRef.current) {
   319→        gsap.fromTo(
   320→          imageRef.current,
   321→          { y: 60, opacity: 0, scale: 0.95 },
   322→          { y: 0, opacity: 1, scale: 1, duration: 1.4, ease: 'power3.out', delay: 1.8 }
   323→        )
   324→
   325→        // Parallax on scroll
   326→        gsap.to(imageRef.current, {
   327→          y: -40,
   328→          ease: 'none',
   329→          scrollTrigger: {
   330→            trigger: sectionRef.current,
   331→            start: 'top top',
   332→            end: 'bottom top',
   333→            scrub: 1,
   334→          },
   335→        })
   336→      }
   337→    }, sectionRef)
   338→
   339→    // 3D tilt on mouse move
   340→    const handleMouseMove = (e: MouseEvent) => {
   341→      if (!imageRef.current) return
   342→      const rect = imageRef.current.getBoundingClientRect()
   343→      const centerX = rect.left + rect.width / 2
   344→      const centerY = rect.top + rect.height / 2
   345→      const rotateY = ((e.clientX - centerX) / (rect.width / 2)) * 5
   346→      const rotateX = -((e.clientY - centerY) / (rect.height / 2)) * 5
   347→      gsap.to(imageRef.current, {
   348→        rotateX,
   349→        rotateY,
   350→        duration: 0.5,
   351→        ease: 'power2.out',
   352→        transformPerspective: 1200,
   353→      })
   354→    }
   355→
   356→    const handleMouseLeave = () => {
   357→      if (!imageRef.current) return
   358→      gsap.to(imageRef.current, {
   359→        rotateX: 0,
   360→        rotateY: 0,
   361→        duration: 0.8,
   362→        ease: 'power2.out',
   363→      })
   364→    }
   365→
   366→    window.addEventListener('mousemove', handleMouseMove)
   367→    imageRef.current?.addEventListener('mouseleave', handleMouseLeave)
   368→
   369→    return () => {
   370→      ctx.revert()
   371→      window.removeEventListener('mousemove', handleMouseMove)
   372→    }
   373→  }, [])
   374→
   375→  // Typing effect
   376→  useEffect(() => {
   377→    if (!typingRef.current) return
   378→    const phrases = [
   379→      "Corriger 100 copies en 5 minutes",
   380→      "Generer des examens avec l'IA",
   381→      "Analyser les resultats en temps reel",
   382→      "Securiser chaque epreuve",

   385→    let charIndex = 0
   386→    let isDeleting = false
   387→    let timeout: ReturnType<typeof setTimeout>
   388→
   389→    const type = () => {
   390→      const currentPhrase = phrases[phraseIndex]
   391→      if (!isDeleting) {
   392→        charIndex++
   393→        if (typingRef.current) {
   394→          typingRef.current.textContent = currentPhrase.slice(0, charIndex)
   395→        }
   396→        if (charIndex === currentPhrase.length) {
   397→          isDeleting = true
   398→          timeout = setTimeout(type, 2000)
   399→          return
   400→        }
   401→        timeout = setTimeout(type, 60)
   402→      } else {
   403→        charIndex--
   404→        if (typingRef.current) {
   405→          typingRef.current.textContent = currentPhrase.slice(0, charIndex)
   406→        }
   407→        if (charIndex === 0) {
   408→          isDeleting = false
   409→          phraseIndex = (phraseIndex + 1) % phrases.length
   410→          timeout = setTimeout(type, 500)
   411→          return
   412→        }
   413→        timeout = setTimeout(type, 30)
   414→      }
   415→    }
   416→
   417→    timeout = setTimeout(type, 2000)
   418→    return () => clearTimeout(timeout)
   419→  }, [])
   420→

   434→          transition={{ delay: 0.1, duration: 0.6 }}
   435→          className="inline-flex items-center gap-2 px-4 py-1.5 mb-6 rounded-full border border-white/[0.08] bg-white/[0.03] backdrop-blur-sm"
   436→        >
   437→          <Sparkles className="h-3.5 w-3.5 text-emerald-400" />

   439→            Nouveau : IA Generative integree
