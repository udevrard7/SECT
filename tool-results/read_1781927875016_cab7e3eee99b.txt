   107→const MONTHS_FR = [
   108→  'janvier', 'février', 'mars', 'avril', 'mai', 'juin',
   109→  'juillet', 'août', 'septembre', 'octobre', 'novembre', 'décembre',
   110→]
   111→
   112→const DAYS_FR = [
   113→  'dimanche', 'lundi', 'mardi', 'mercredi', 'jeudi', 'vendredi', 'samedi',
   114→]
   115→
   116→function formatDateFR(dateStr: string | Date): string {
   117→  const d = typeof dateStr === 'string' ? new Date(dateStr) : dateStr
   118→  return `${d.getDate()} ${MONTHS_FR[d.getMonth()]} ${d.getFullYear()}`
   119→}
   120→
   121→function formatDateTimeFR(dateStr: string | Date): string {
   122→  const d = typeof dateStr === 'string' ? new Date(dateStr) : dateStr
   123→  const day = DAYS_FR[d.getDay()]
   124→  const hours = d.getHours().toString().padStart(2, '0')
   125→  const minutes = d.getMinutes().toString().padStart(2, '0')
   126→  return `${day} ${d.getDate()} ${MONTHS_FR[d.getMonth()]} ${d.getFullYear()} à ${hours}h${minutes}`
   127→}
   128→
   129→function parseJsonSafe<T>(value: string | null | unknown, fallback: T): T {
   130→  if (!value) return fallback
   131→  try {
   132→    return JSON.parse(typeof value === 'string' ? value : JSON.stringify(value)) as T
   133→  } catch {
   134→    return fallback
   135→  }
   136→}
   137→
   138→function getScoreBadgeClasses(score: number): string {
   139→  if (score >= 10) {
   140→    return 'bg-emerald-100 text-emerald-800 border-emerald-200 dark:bg-emerald-900/40 dark:text-emerald-300 dark:border-emerald-800'
   141→  }
   142→  if (score >= 8) {
   143→    return 'bg-amber-100 text-amber-800 border-amber-200 dark:bg-amber-900/40 dark:text-amber-300 dark:border-amber-800'
   144→  }
   145→  return 'bg-red-100 text-red-800 border-red-200 dark:bg-red-900/40 dark:text-red-300 dark:border-red-800'
   146→}
   147→
   148→function getScoreColor(score: number): string {
   149→  if (score >= 10) return 'text-emerald-700 dark:text-emerald-400'
   150→  if (score >= 8) return 'text-amber-700 dark:text-amber-400'
   151→  return 'text-red-700 dark:text-red-400'
   152→}
   153→
   154→function getProgressColor(score: number): string {
   155→  if (score >= 10) return 'bg-emerald-500'
   156→  if (score >= 8) return 'bg-amber-500'
   157→  return 'bg-red-500'
   158→}
   159→
   160→function getProgressBg(score: number): string {
   161→  if (score >= 10) return 'bg-emerald-100 dark:bg-emerald-900/30'
   162→  if (score >= 8) return 'bg-amber-100 dark:bg-amber-900/30'
   163→  return 'bg-red-100 dark:bg-red-900/30'
   164→}
   165→
   166→function getQuestionTypeBadgeClasses(type: string): string {
   167→  switch (type?.toUpperCase()) {
   168→    case 'QCU':
   169→      return 'bg-sky-100 text-sky-800 border-sky-200 dark:bg-sky-900/40 dark:text-sky-300 dark:border-sky-800'
   170→    case 'QCM':
   171→      return 'bg-amber-100 text-amber-800 border-amber-200 dark:bg-amber-900/40 dark:text-amber-300 dark:border-amber-800'
   172→    case 'QRC':
   173→      return 'bg-emerald-100 text-emerald-800 border-emerald-200 dark:bg-emerald-900/40 dark:text-emerald-300 dark:border-emerald-800'
   174→    case 'TRS':
   175→      return 'bg-rose-100 text-rose-800 border-rose-200 dark:bg-rose-900/40 dark:text-rose-300 dark:border-rose-800'
   176→    default:
   177→      return 'bg-gray-100 text-gray-800 border-gray-200 dark:bg-gray-900/40 dark:text-gray-300 dark:border-gray-800'
   178→  }
   179→}

   189→  // Detail dialog state

   191→  const [selectedResult, setSelectedResult] = useState<StudentSession | null>(null)
   192→
   193→  // ─── Fetch results ───
   194→  const fetchResultats = useCallback(async () => {
   195→    if (!user?.id) return
   196→    setIsLoading(true)
   197→    try {
   198→      const res = await fetch(`/api/resultats?etudiantId=${user.id}`)
   199→      if (res.ok) {
   200→        const data = await res.json()
   201→        setResultats(data.resultats ?? [])
   202→      }
   203→    } catch {
   204→      toast.error('Erreur de chargement', {
   205→        description: 'Impossible de charger vos résultats.',
   206→      })
   207→    } finally {
   208→      setIsLoading(false)
   209→    }
   210→  }, [user])
   211→
   212→  useEffect(() => {
   213→    fetchResultats()
   214→  }, [fetchResultats])
   215→
   216→  // ─── Statistics ───
   217→  const globalStats = useMemo(() => {
   218→    const normalizedScores = resultats
   219→      .filter(r => r.statut === 'RETOURNEE' || r.statut === 'CORRIGEE')
   220→      .map((r) => {
   221→        const rawScore = r.resultat?.scoreFinal ?? r.score ?? 0
   222→        const totalPossible = r.resultat?.totalPossible ?? 20
   223→        return totalPossible > 0 ? (rawScore / totalPossible) * 20 : 0
   224→      })
   225→      .filter((s) => s > 0)
   226→
   227→    if (normalizedScores.length === 0) {
   228→      return { moyenne: 0, count: 0, best: 0 }
   229→    }
   230→
   231→    const moyenne = normalizedScores.reduce((a, b) => a + b, 0) / normalizedScores.length
   232→    const best = Math.max(...normalizedScores)
   233→
   234→    return {
   235→      moyenne: Math.round(moyenne * 10) / 10,
   236→      count: resultats.length,
   237→      best: Math.round(best * 10) / 10,
   238→    }
   239→  }, [resultats])
   240→
   241→  // ─── Detail dialog handlers ───
   242→  const handleViewDetail = (result: StudentSession) => {
   243→    setSelectedResult(result)
   244→    setDetailDialogOpen(true)
   245→  }
   246→
   247→  // ─── Build question details for dialog ───
   248→  const dialogQuestionDetails = useMemo(() => {
   249→    if (!selectedResult) return []
   250→
   251→    const epreuveQuestions = selectedResult.epreuve.questions || []
   252→    const reponses = selectedResult.reponses || []
   253→
   254→    // If we have rich epreuve.questions data (from /api/resultats), always build from it
   255→    // This ensures consistent format with enonce, pointsMax, pointsObtenus, etc.
   256→    if (epreuveQuestions.length > 0) {
   257→      return epreuveQuestions.map((eq, idx) => {
   258→        const reponse = reponses.find((r) => r.questionId === eq.questionId)
   259→        const pointsObtenus = reponse?.score ?? null
   260→        const isGraded = pointsObtenus !== null && pointsObtenus !== undefined
   261→
   262→        return {
   263→          index: idx + 1,
   264→          type: eq.question.type,
   265→          enonce: eq.question.enonce,
   266→          pointsMax: eq.bareme,
   267→          pointsObtenus,
   268→          correct: isGraded
   269→            ? pointsObtenus! >= eq.bareme * 0.5
   270→            : null,
   271→          reponseEtudiant: reponse?.contenu ?? null,
   272→          reponseAttendue: null,
   273→          commentaire: reponse?.commentaire ?? null,
   274→        }
   275→      })
   276→    }
   277→
   278→    // Fallback: try to use detailParQuestion if epreuve.questions is empty
   279→    // Map stored format {questionId, type, bareme, score, repondu} to frontend format
   280→    const details = selectedResult.resultat?.detailParQuestion
   281→    if (details && Array.isArray(details) && details.length > 0) {
   282→      return details.map((q: Record<string, unknown>, idx: number) => {
   283→        const score = typeof q.score === 'number' ? q.score : null
   284→        const bareme = typeof q.bareme === 'number' ? q.bareme : 1
   285→        const isGraded = score !== null
   286→
   287→        return {
   288→          index: idx + 1,
   289→          type: String(q.type || ''),
   290→          enonce: String(q.enonce || `Question ${idx + 1}`),
   291→          pointsMax: bareme,
   292→          pointsObtenus: score,
   293→          correct: isGraded ? (score! >= bareme * 0.5) : null,
   294→          reponseEtudiant: typeof q.reponseEtudiant === 'string' ? q.reponseEtudiant : null,
   295→          reponseAttendue: typeof q.reponseAttendue === 'string' ? q.reponseAttendue : null,
   296→          commentaire: typeof q.commentaire === 'string' ? q.commentaire : null,
   297→        }
   298→      })
   299→    }
   300→
   301→    return []
   302→  }, [selectedResult])
   303→
   304→  return (
   305→    <div className="space-y-6">
   306→      {/* ─── Header ─── */}
   307→      <div>
   308→        <h1 className="text-2xl font-bold tracking-tight md:text-3xl flex items-center gap-2">
   309→          <Trophy className="h-7 w-7 text-emerald-600 dark:text-emerald-400" />
   310→          Mes Résultats
   311→        </h1>
   312→        <p className="mt-1 text-sm text-muted-foreground">
   313→          Consultez vos notes et résultats
   314→        </p>
   315→      </div>
   316→
   317→      {/* ─── Statistics card ─── */}
   318→      {!isLoading && resultats.length > 0 && (
   319→        <div className="grid gap-4 sm:grid-cols-3">
   320→          <Card className="border-l-4 border-l-emerald-500">
   321→            <CardContent className="p-4">
   322→              <div className="flex items-center gap-3">
   323→                <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-emerald-100 dark:bg-emerald-900/40">
   324→                  <Target className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   325→                </div>
   326→                <div>
   327→                  <p className="text-xs font-medium text-muted-foreground">Moyenne générale</p>
   328→                  <p className={`text-2xl font-bold ${getScoreColor(globalStats.moyenne)}`}>
   329→                    {globalStats.moyenne.toFixed(1)}<span className="text-sm text-muted-foreground">/20</span>
   330→                  </p>
   331→                </div>
   332→              </div>
   333→            </CardContent>
   334→          </Card>
   335→
   336→          <Card className="border-l-4 border-l-teal-500">
   337→            <CardContent className="p-4">
   338→              <div className="flex items-center gap-3">
   339→                <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-teal-100 dark:bg-teal-900/40">
   340→                  <BookOpen className="h-5 w-5 text-teal-600 dark:text-teal-400" />
   341→                </div>
   342→                <div>
   343→                  <p className="text-xs font-medium text-muted-foreground">Nombre d&apos;épreuves</p>
   344→                  <p className="text-2xl font-bold">
   345→                    {globalStats.count}
   346→                  </p>
   347→                </div>
   348→              </div>
   349→            </CardContent>
   350→          </Card>
   351→
   352→          <Card className="border-l-4 border-l-emerald-600">
   353→            <CardContent className="p-4">
   354→              <div className="flex items-center gap-3">
   355→                <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-emerald-100 dark:bg-emerald-900/40">
   356→                  <Award className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   357→                </div>
   358→                <div>
   359→                  <p className="text-xs font-medium text-muted-foreground">Meilleure note</p>
   360→                  <p className={`text-2xl font-bold ${getScoreColor(globalStats.best)}`}>
   361→                    {globalStats.best.toFixed(1)}<span className="text-sm text-muted-foreground">/20</span>
   362→                  </p>
   363→                </div>
   364→              </div>
   365→            </CardContent>
   366→          </Card>
   367→        </div>

   378→                    <div className="h-5 w-2/3 rounded bg-muted" />
