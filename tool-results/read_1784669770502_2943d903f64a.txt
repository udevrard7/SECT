   155→// SUBMIT-JITTER-1 (OPT-7) : amplitude du jitter appliqué à l'auto-submit de
   156→// fin de temps. Quand le chronomètre atteint 0, au lieu de soumettre à l'instant
   157→// (t=0 pour toute la promotion → pic qui sature le backend), on tire un délai
   158→// aléatoire dans [0, SUBMIT_JITTER_MS] pour étaler les soumissions sur cette
   159→// fenêtre. 45s étaie 1000 étudiants sur ~6-7 submits/s, juste sous la capacité
   160→// Render free. Overridable via NEXT_PUBLIC_SUBMIT_JITTER_MS (ex. 60000 en prod).
   161→const SUBMIT_JITTER_MS = (() => {
   162→  const raw = typeof process !== 'undefined' ? process.env.NEXT_PUBLIC_SUBMIT_JITTER_MS : undefined
   163→  const parsed = raw ? parseInt(raw, 10) : NaN
   164→  return Number.isFinite(parsed) && parsed >= 0 && parsed <= 120000 ? parsed : 45000
   165→})()
   166→// Nombre max de retries si le backend renvoie 202 (file pleine). 10 × 3s = 30s

   170→// ─── Utility: format time as HH:MM:SS ──────────────────────────────────────
   171→
   172→function formatTime(totalSeconds: number): string {
   173→  if (totalSeconds <= 0) return '00:00:00'
   174→  const h = Math.floor(totalSeconds / 3600)
   175→  const m = Math.floor((totalSeconds % 3600) / 60)
   176→  const s = totalSeconds % 60
   177→  return [h, m, s].map((v) => String(v).padStart(2, '0')).join(':')
   178→}

   182→function parseJsonSafe<T>(value: string | null | undefined, fallback: T): T {
   183→  if (!value) return fallback
   184→  try {
   185→    return JSON.parse(value) as T
   186→  } catch {
   187→    return fallback
   188→  }
   189→}

   195→// BUGFIX (E2E-TEST) : les propositions générées par l'IA sont des objets

   197→// la passation attendait des strings. React error #31 quand il tentait de
   198→// rendre {prop} (objet) comme enfant de <span>. Ce helper gère les 2 formats.
   199→function getPropText(prop: unknown): string {
   200→  if (typeof prop === 'string') return prop
   201→  if (prop && typeof prop === 'object' && 'text' in prop) {
   202→    return String((prop as { text: unknown }).text)
   203→  }
   204→  return String(prop)
   205→}

   478→              const logs = JSON.parse(activeSession.logEvents as string)

   480→            } catch {
   481→              // Ignore parse errors
   482→            }

   487→        }
   488→      }

   492→        saveExamSnapshot({
   493→          epreuveId,
   494→          userId: user.id,
   495→          epreuve: epreuveInfo,
   496→          questions: questionsData,
   497→          session: activeSession,
   498→          reponses,

   500→          timeRemaining,

   509→        securityConfig,

   524→    refetchOnReconnect: false,

   527→  // Sync query.data → local state (préserve le comportement du fetch original
   528→  // qui setEpreuve/setSecurityConfig/etc. au fur et à mesure). Le useEffect ne

   547→    // elles reflètent la dernière action de l'étudiant.
   548→    if (d.session?.id) {

   551→          // Fusion : IndexedDB prime sur le serveur pour les questions
   552→          // présentes dans le cache (plus récentes)

   554→        } else {
   555→          setReponses(d.reponses)
   556→        }
   557→      })
   558→    } else {
   559→      setReponses(d.reponses)
   560→    }
   561→  }, [epreuveQuery.data])
   562→

   564→  useEffect(() => {
   565→    if (epreuveQuery.error) {

   568→        description: err instanceof Error ? err.message : 'Impossible de charger l\'épreuve',
   569→      })

   571→  }, [epreuveQuery.error])
   572→

   574→  // du guard original), sinon isLoading de la query.

   577→  // ─── Start exam ────────────────────────────────────────────────────────
   578→  const startExam = useCallback(async () => {

   643→    }

   646→  // ─── Auto-save answers ─────────────────────────────────────────────────
   647→  // BUGFIX (EXAM-OFFLINE-1) : saveAnswers gère maintenant le hors-ligne.
   648→  // Si le fetch échoue (micro-coupure), les réponses sont déjà en IndexedDB
   649→  // (via handleAnswerChange) et un retry est programmé au retour online.
   650→  //
   651→  // OPT-5 : skip l'auto-save si les réponses n'ont pas changé depuis le dernier save.
   652→  // Réduit ~50% des requêtes HTTP pendant l'examen (les étudiants ne modifient
   653→  // pas leurs réponses en continu). Pour 5000 étudiants, on passe de 167 saves/s
   654→  // à ~83 saves/s en moyenne.
   655→  const saveAnswers = useCallback(async () => {
   656→    if (!sessionRef.current || phaseRef.current !== 'in-exam') return
   657→
   658→    try {
   659→      const currentReponses = reponsesRef.current
   660→      const entries = Object.entries(currentReponses)
   661→
   662→      if (entries.length === 0) {
   663→        setLastSaved(new Date())
   664→        return
   665→      }
   666→
   667→      // OPT-5 : comparer les réponses actuelles avec les dernières sauvegardées.
   668→      // Si rien n'a changé, on skip le save (économise 1 requête HTTP + 1 write DB).
   669→      const currentHash = JSON.stringify(entries.sort())
   670→      if (currentHash === lastSavedReponsesHashRef.current) {
   671→        return // Rien n'a changé, skip
   672→      }
   673→
   674→      // Batch save all answers
   675→      const res = await fetch(`/api/sessions/${sessionRef.current.id}`, {
   676→        method: 'PUT',
   677→        headers: { 'Content-Type': 'application/json' },
   678→        body: JSON.stringify({ reponses: currentReponses }),
   679→      })
   680→
   681→      if (res.ok) {
   682→        lastSavedReponsesHashRef.current = currentHash
   683→        setLastSaved(new Date())
   684→      }
   685→    } catch {
   686→      // BUGFIX (EXAM-OFFLINE-1) : réseau coupé — les réponses sont déjà
   687→      // en IndexedDB (sauvegardées à chaque handleAnswerChange). Un retry
   688→      // automatique se déclenchera au retour du réseau (online event).
   689→      // On ne perturbe pas l'étudiant (silent fail).
   690→    }
   691→  }, [])
   692→
   693→  // OPT-9 : version debouncée de saveAnswers.
   694→  // Quand l'étudiant navigue rapidement entre les questions, on coalesce
   695→  // les saves en un seul appel API après 2s d'inactivité.
   696→  // Le flush immédiat reste disponible pour le submit (debouncedSave.flush()).
   697→  const debouncedSaveAnswers = useMemo(
   698→    () => debounce(saveAnswers, 2000),
   699→    [saveAnswers]
   700→  )
   701→
   702→  // Cleanup du debounce au démontage
   703→  useEffect(() => {
   704→    return () => {
   705→      debouncedSaveAnswers.cancel()
   706→    }
   707→  }, [debouncedSaveAnswers])
   708→
   709→  // Annule le jitter si l'étudiant soumet manuellement avant la fin du compte
   710→  // à rebours. Appelé depuis submitExam (via cleanup) et au unmount.
   711→  const cancelPendingAutoSubmit = useCallback(() => {
   712→    if (pendingAutoSubmitTimeoutRef.current) {
   713→      clearTimeout(pendingAutoSubmitTimeoutRef.current)
   714→      pendingAutoSubmitTimeoutRef.current = null
   715→    }
   716→    setPendingAutoSubmitIn(null)
   717→  }, [])
   718→
   719→  // ─── Submit exam ───────────────────────────────────────────────────────
   720→  // SUBMIT-JITTER-1 + SUBMIT-RATELIMIT-1 (OPT-7) : gère le 202 Accepted du
   721→  // backend (file de submits pleine) en réessayant après le Retry-After
   722→  // indiqué. Le submit reste idempotent côté backend (statut session vérifié),
   723→  // donc un retry est sûr. Max SUBMIT_MAX_RETRIES pour rester sous le timeout
   724→  // de session frontend.
   725→  const submitExam = useCallback(async (autoSubmit: boolean = false, reason: AutoSubmitReason = null) => {
   726→    if (!sessionRef.current || isAutoSubmittingRef.current) return
   727→    if (isSubmitting && !autoSubmit) return
   728→
   729→    // SUBMIT-JITTER-1 : si l'étudiant soumet manuellement pendant la fenêtre
   730→    // de jitter, annuler le submit jitteré planifié.
   731→    cancelPendingAutoSubmit()
   732→
   733→    isAutoSubmittingRef.current = true
   734→    setIsSubmitting(true)
   735→
   736→    try {
   737→      // Save answers first
   738→      // OPT-9 : flush immédiat de tous les saves en attente avant le submit
   739→      debouncedSaveAnswers.flush()
   740→      // Small delay to let the flush complete
   741→      await new Promise(resolve => setTimeout(resolve, 100))
   742→
   743→      type SubmitResponse = {
   744→        penalite?: number
   745→        scenario?: 'A' | 'B'
   746→        scenarioMessage?: string
   747→        score?: number
   748→        totalPossible?: number
   749→        autoGradableTotal?: number
   750→        pendingCorrection?: number
   751→      }
   752→
   753→      // SUBMIT-RATELIMIT-1 : on ne peut pas utiliser submitOffline pour le
   754→      // submit final car le hook traite tout 2xx (dont 202) comme un succès.
   755→      // Or 202 = "file pleine, réessayez" → il faut retry avec Retry-After.
   756→      // On fait donc un fetch direct avec boucle de retry, en gardant le
   757→      // fallback offline (outbox) si le réseau lâche complètement.
   758→      const submitUrl = `/api/sessions/${sessionRef.current.id}/submit`
   759→      const submitBody = JSON.stringify({ autoSubmit, reponses: reponsesRef.current })
   760→
   761→      let data: SubmitResponse | undefined
   762→      let lastError: string | undefined
   763→      let attempt = 0
   764→      let queued = false
   765→
   766→      while (attempt < SUBMIT_MAX_RETRIES) {
   767→        attempt += 1
   768→
   769→        // Offline → outbox (une seule fois, puis break)
   770→        if (typeof navigator !== 'undefined' && !navigator.onLine) {
   771→          const result = await submitOffline<SubmitResponse>(
   772→            submitUrl,
   773→            { autoSubmit, reponses: reponsesRef.current },
   774→            { type: 'submit-exam', meta: { examTitle: epreuve?.titre ?? 'Épreuve' } }
   775→          )
   776→          if (!result.synced) {
   777→            // Hook a déjà toastyé. Si l'outbox échoue aussi, on remonte.
   778→            if (result.error) {
   779→              toast.error('Sauvegarde hors ligne impossible', { description: result.error })
   780→            }
   781→            return
   782→          }
   783→          if (result.error) {
   784→            throw new Error(result.error)
   785→          }
   786→          data = result.data
   787→          break
   788→        }
   789→
   790→        // Online → fetch direct pour pouvoir distinguer 202 des vrais succès
   791→        let res: Response
   792→        try {
   793→          res = await fetch(submitUrl, {
   794→            method: 'POST',
   795→            headers: { 'Content-Type': 'application/json' },
   796→            body: submitBody,
   797→            credentials: 'same-origin',
   798→          })
   799→        } catch (err) {
   800→          // Erreur réseau inattendue → fallback offline outbox (une fois)
   801→          const result = await submitOffline<SubmitResponse>(
   802→            submitUrl,
   803→            { autoSubmit, reponses: reponsesRef.current },
   804→            { type: 'submit-exam', meta: { examTitle: epreuve?.titre ?? 'Épreuve' } }
   805→          )
   806→          if (!result.synced) {
   807→            if (result.error) {
   808→              toast.error('Sauvegarde hors ligne impossible', { description: result.error })
   809→            }
   810→            return
   811→          }
   812→          if (result.error) throw new Error(result.error)
   813→          data = result.data
   814→          break
   815→        }
   816→
   817→        // SUBMIT-RATELIMIT-1 : 202 Accepted = file pleine, retry après Retry-After
   818→        if (res.status === 202) {
   819→          const body = await res.json().catch(() => ({} as { retryAfter?: number }))
   820→          const retryAfter = (body as { retryAfter?: number }).retryAfter ?? 3
   821→          queued = true
   822→          // Indicateur visuel pendant l'attente
   823→          if (attempt === 1) {
   824→            toast.info('Soumission en file d\'attente…', {
   825→              description: `Le serveur traite d'autres copies. Nouvelle tentative dans ${retryAfter}s.`,
   826→              duration: retryAfter * 1000 + 500,
   827→            })
   828→          }
   829→          await new Promise((resolve) => setTimeout(resolve, retryAfter * 1000))
   830→          continue
   831→        }
   832→
   833→        // Vrai succès (200) ou erreur HTTP
   834→        if (!res.ok) {
   835→          const err = await res.json().catch(() => ({}))
   836→          throw new Error((err as { error?: string }).error ?? `HTTP ${res.status}`)
   837→        }
   838→
   839→        data = (await res.json().catch(() => undefined)) as SubmitResponse | undefined
   840→        break
   841→      }
   842→
   843→      if (queued && !data) {
   844→        // Épuisé les retries sans succès — l'étudiant garde sa copie en IndexedDB
   845→        // (déjà sauvée) et le SW retryera. On ne perd rien.
   846→        toast.warning('Soumission en cours de traitement', {
   847→          description: 'Le serveur est très sollicité. Votre copie est sauvegardée et sera soumise automatiquement.',
   848→          duration: 6000,
   849→        })
   850→        return
   851→      }
   852→
   853→      if (!data) {
   854→        throw new Error(lastError ?? 'Réponse vide du serveur')
   855→      }
   856→
   857→      if (autoSubmit) {
   858→        setAutoSubmitted(true)
   859→        setAutoSubmitReason(reason)
   860→      }
   861→
   862→      // Update penalty from server response
   863→      if (data?.penalite !== undefined) {
   864→        setPenalite(data.penalite)
   865→      }
   866→
   867→      // Store submit result for scenario-specific display
   868→      if (data?.scenario) {
   869→        setSubmitResult({
   870→          score: data.score ?? 0,
   871→          totalPossible: data.totalPossible ?? 0,
   872→          autoGradedTotal: data.autoGradableTotal ?? data.totalPossible ?? 0,
   873→          pendingCorrection: data.pendingCorrection ?? 0,
   874→          scenario: data.scenario,
   875→          scenarioMessage: data.scenarioMessage ?? '',
   876→        })
   877→      }
   878→
   879→      setPhase('post-exam')
   880→
   881→      // BUGFIX (EXAM-OFFLINE-1) : nettoyer le cache IndexedDB après submit
   882→      // réussi (les réponses sont maintenant sur le serveur).
   883→      if (sessionRef.current) {
   884→        clearCachedAnswers(sessionRef.current.id)
   885→      }
   886→
   887→      // OPT-10: nettoyer le snapshot IndexedDB après submit réussi
   888→      clearExamSnapshot(epreuveId).catch(() => {})
   889→
   890→      // Exit fullscreen
   891→      try {
   892→        if (document.fullscreenElement) {
   893→          await document.exitFullscreen()
   894→        }
   895→      } catch {
   896→        // Ignore
   897→      }
   898→
   899→      // Clean up intervals
   900→      if (autoSaveIntervalRef.current) clearInterval(autoSaveIntervalRef.current)
   901→      if (timerIntervalRef.current) clearInterval(timerIntervalRef.current)
   902→      if (inactivityIntervalRef.current) clearInterval(inactivityIntervalRef.current)
   903→      if (captureIntervalRef.current) clearInterval(captureIntervalRef.current)
   904→    } catch (err) {
   905→      toast.error('Erreur', {
   906→        description: err instanceof Error ? err.message : 'Impossible de soumettre l\'épreuve',
   907→      })
   908→    } finally {
   909→      setIsSubmitting(false)
   910→      isAutoSubmittingRef.current = false
   911→    }
   912→  }, [isSubmitting, debouncedSaveAnswers, submitOffline, epreuve, cancelPendingAutoSubmit])
   913→
   914→  // ─── Log alert (enhanced with penalty) ─────────────────────────────────
   915→  const logAlert = useCallback(async (type: string, details?: string, alertPenalite?: number) => {
   916→    if (!sessionRef.current) return
   917→
   918→    try {
   919→      await fetch(`/api/sessions/${sessionRef.current.id}`, {
   920→        method: 'PUT',
   921→        headers: { 'Content-Type': 'application/json' },
   922→        body: JSON.stringify({
   923→          alerte: {
   924→            type,
   925→            details,
   926→            penalite: alertPenalite || 0,
   927→          },
   928→        }),
   929→      })
   930→
   931→      // Increment client-side alert count
   932→      setTotalAlertCount((prev) => {
   933→        const newCount = prev + 1
   934→        // Check if we need to auto-submit
   935→        const config = securityConfigRef.current
   936→        if (config.autoSubmitOnViolation && newCount >= config.nbAlertesMax && !isAutoSubmittingRef.current) {
   937→          // Trigger auto-submit via violation dialog
   938→          setAutoSubmitReason('violations')
   939→          setShowViolationDialog(true)
   940→        }
   941→        return newCount
   942→      })
   943→
   944→      // Update penalty
   945→      if (alertPenalite && alertPenalite > 0) {
   946→        setPenalite((prev) => prev + alertPenalite)
   947→      }
   948→    } catch {
   949→      // Silent fail
   950→    }
   951→  }, [])
   952→
   953→  // ─── Handle violation dialog confirmation ──────────────────────────────
   954→  const handleViolationSubmit = useCallback(() => {
   955→    setShowViolationDialog(false)
   956→    submitExam(true, 'violations')
   957→  }, [submitExam])
   958→
   959→  // ─── Capture screenshot ───────────────────────────────────────────────
   960→  const captureScreen = useCallback(async () => {
   961→    if (!sessionRef.current || !examContainerRef.current) return
   962→
   963→    try {
   964→      // Dynamic import of html2canvas-pro
   965→      const html2canvas = (await import('html2canvas-pro')).default
   966→      const canvas = await html2canvas(examContainerRef.current, {
   967→        scale: 0.5, // Lower resolution for smaller payload
   968→        useCORS: true,
   969→        logging: false,
   970→        allowTaint: true,
   971→      })
   972→
   973→      const image = canvas.toDataURL('image/jpeg', 0.4) // Compress to JPEG 40% quality
   974→
   975→      // Upload to server
   976→      await fetch(`/api/sessions/${sessionRef.current.id}/capture`, {
   977→        method: 'POST',
   978→        headers: { 'Content-Type': 'application/json' },
   979→        body: JSON.stringify({ image }),
   980→      })
   981→
   982→      // Show brief flash indicator
   983→      setShowCaptureFlash(true)
   984→      setLastCaptureTime(new Date())
   985→      setTimeout(() => setShowCaptureFlash(false), 1500)
   986→    } catch {
   987→      // Silent fail — don't disrupt the exam
   988→    }
   989→  }, [])
   990→
   991→  // ─── SUBMIT-JITTER-1 (OPT-7) : planification du submit jitteré ────────
   992→  // Quand le temps expire, on ne soumet pas immédiatement. On tire un délai
   993→  // aléatoire dans [0, SUBMIT_JITTER_MS] et on planifie le submit. L'étudiant
   994→  // voit un compte à rebours pendant cette fenêtre. Si l'étudiant soumet
   995→  // manuellement avant (bouton « Terminer »), le jitter est annulé.
