    72→function getAcademicYear(date: Date = new Date()): string {
    73→  const year = date.getFullYear()
    74→  const month = date.getMonth()
    75→  const startYear = month >= 8 ? year : year - 1
    76→  return `${startYear}-${startYear + 1}`
    77→}

    79→function formatDateFR(date: Date | string): string {
    80→  const d = typeof date === 'string' ? new Date(date) : date
    81→  return d.toLocaleDateString('fr-FR', {
    82→    day: '2-digit', month: '2-digit', year: 'numeric',
    83→  })
    84→}

    86→function getTypeLabel(type: string): string {
    87→  switch (type) {
    88→    case 'QCU': return 'QCU (Choix unique)'
    89→    case 'QCM': return 'QCM (Choix multiples)'
    90→    case 'QRC': return 'QRC (Réponse courte)'
    91→    case 'REFLEXION': return 'Réflexion'
    92→    case 'CODE': return 'Code (Programmation)'
    93→    default: return type
    94→  }
    95→}

   101→/**
   102→ * Wrap text to fit within a given width. Returns array of lines.
   103→ */
   104→function wrapText(doc: jsPDF, text: string, maxWidth: number): string[] {
   105→  if (!text) return []
   106→  return doc.splitTextToSize(text, maxWidth) as string[]
   107→}
   108→
   109→/**
   110→ * Get the height needed to render a block of text.
   111→ */
   112→function getTextHeight(doc: jsPDF, text: string, maxWidth: number, fontSize: number = FONT_BODY): number {
   113→  doc.setFontSize(fontSize)
   114→  const lines = wrapText(doc, text, maxWidth)
   115→  return lines.length * (fontSize * 0.3528 * 1.4) + 2
   116→}

   120→function addHeader(doc: jsPDF, data: EpreuvePDFData, isCorrige: boolean = false): void {
   121→  const y = MARGIN_TOP
   122→
   123→  // Logo (if available)
   124→  let textOffsetX = MARGIN_LEFT
   125→  if (data.etablissement.logo) {
   126→    try {
   127→      const logoData = data.etablissement.logo
   128→      // SVG is not supported by jsPDF - skip it
   129→      if (logoData.includes('image/svg+xml')) {
   130→        console.warn('[PDF] SVG logos are not supported in PDF generation, skipping')
   131→      } else {
   132→        // Detect format from data URL
   133→        let format = 'PNG'
   134→        if (logoData.includes('image/jpeg') || logoData.includes('image/jpg')) {
   135→          format = 'JPEG'
   136→        } else if (logoData.includes('image/webp')) {
   137→          format = 'PNG' // jsPDF may not support WEBP; will try as PNG fallback
   138→        }
   139→        const logoWidth = 18  // mm
   140→        const logoHeight = 18 // mm
   141→        doc.addImage(logoData, format, MARGIN_LEFT, y, logoWidth, logoHeight)
   142→        textOffsetX = MARGIN_LEFT + logoWidth + 4
   143→      }
   144→    } catch (e) {
   145→      // If logo fails to render, just skip it
   146→      console.warn('[PDF] Failed to render logo:', e)
   147→    }
   148→  }
   149→
   150→  // Left side: Etablissement info
   151→  doc.setFontSize(12)
   152→  doc.setFont('helvetica', 'bold')
   153→  doc.text(data.etablissement.nom, textOffsetX, y + 5)
   154→
   155→  doc.setFontSize(FONT_SMALL)
   156→  doc.setFont('helvetica', 'normal')
   157→  let infoY = y + 10
   158→  if (data.etablissement.ville) {
   159→    doc.text(`${data.etablissement.ville}${data.etablissement.pays ? `, ${data.etablissement.pays}` : ''}`, textOffsetX, infoY)
   160→    infoY += 4
   161→  }
   162→
   163→  // Academic year
   164→  doc.text(`Année universitaire : ${getAcademicYear()}`, textOffsetX, infoY)
   165→
   166→  // Right side: UE & Filière info
   167→  const rightX = PAGE_WIDTH - MARGIN_RIGHT
   168→  doc.setFontSize(FONT_SMALL)
   169→  doc.setFont('helvetica', 'normal')
   170→
   171→  let rightY = y + 5
   172→  if (data.uniteEnseignement) {
   173→    doc.setFont('helvetica', 'bold')
   174→    const ueText = `UE : ${data.uniteEnseignement.code} - ${data.uniteEnseignement.nom}`
   175→    const ueLines = wrapText(doc, ueText, 90)
   176→    for (const line of ueLines) {
   177→      doc.text(line, rightX, rightY, { align: 'right' })
   178→      rightY += 4
   179→    }
   180→    doc.setFont('helvetica', 'normal')
   181→  }
   182→  if (data.filiere) {
   183→    doc.text(`Filière : ${data.filiere.nom}${data.filiere.code ? ` (${data.filiere.code})` : ''}`, rightX, rightY, { align: 'right' })
   184→    rightY += 4
   185→  }
   186→
   187→  // Separator line
   188→  const sepY = HEADER_BOTTOM - 4
   189→  doc.setDrawColor(180, 180, 180)
   190→  doc.setLineWidth(0.5)
   191→  doc.line(MARGIN_LEFT, sepY, PAGE_WIDTH - MARGIN_RIGHT, sepY)
   192→
   193→  // Watermark for Corrigé
   194→  if (isCorrige) {
   195→    doc.setFontSize(9)
   196→    doc.setFont('helvetica', 'bold')
   197→    doc.setTextColor(220, 50, 50)
   198→    doc.text('CORRIGÉ TYPE — CONFIDENTIEL', PAGE_WIDTH / 2, sepY - 2, { align: 'center' })
   199→    doc.setTextColor(0, 0, 0)
   200→  }
   201→}

   203→function addFooter(doc: jsPDF, data: EpreuvePDFData, pageNum: number, totalPages: number, isCorrige: boolean = false): void {
   204→  const y = FOOTER_TOP
   205→
   206→  // Separator line
   207→  doc.setDrawColor(180, 180, 180)
   208→  doc.setLineWidth(0.3)
   209→  doc.line(MARGIN_LEFT, y, PAGE_WIDTH - MARGIN_RIGHT, y)
   210→
   211→  doc.setFontSize(FONT_TINY)
   212→  doc.setFont('helvetica', 'normal')
   213→  doc.setTextColor(120, 120, 120)
   214→
   215→  // Left: Confidential
   216→  const label = isCorrige ? 'Corrigé confidentiel' : 'Confidentiel'
   217→  doc.text(label, MARGIN_LEFT, y + 4)
   218→
   219→  // Center: Epreuve title (truncated)
   220→  const maxTitleLen = 60
   221→  const title = data.titre.length > maxTitleLen ? data.titre.slice(0, maxTitleLen) + '…' : data.titre
   222→  doc.text(title, PAGE_WIDTH / 2, y + 4, { align: 'center' })
   223→
   224→  // Right: Page number
   225→  doc.text(`Page ${pageNum} / ${totalPages}`, PAGE_WIDTH - MARGIN_RIGHT, y + 4, { align: 'right' })
   226→
   227→  doc.setTextColor(0, 0, 0)
   228→}

   232→function addNewPage(doc: jsPDF, data: EpreuvePDFData, isCorrige: boolean = false): void {

   237→/**
   238→ * Check if we need a new page. If so, add one and return the new Y position.
   239→ * Returns the Y position to continue from (either current or top of new page).
   240→ */
   241→function ensureSpace(
   242→  doc: jsPDF,
   243→  data: EpreuvePDFData,
   244→  currentY: number,
   245→  requiredHeight: number,
   246→  isCorrige: boolean = false
   247→): number {
   248→  if (currentY + requiredHeight > CONTENT_BOTTOM) {
   249→    addNewPage(doc, data, isCorrige)
   250→    return CONTENT_TOP
   251→  }
   252→  return currentY
   253→}

   257→/**
   258→ * Estimate the height needed for a question in Sujet mode.
   259→ */
   260→function estimateQuestionHeightSujet(doc: jsPDF, q: PDFQuestion): number {
   261→  let height = 0
   262→
   263→  // Question header line
   264→  height += 8
   265→
   266→  // Question text
   267→  height += getTextHeight(doc, q.enonce, CONTENT_WIDTH - 5, FONT_BODY)
   268→
   269→  // Propositions for QCU/QCM
   270→  if ((q.type === 'QCU' || q.type === 'QCM') && q.propositions && q.propositions.length > 0) {
   271→    height += 3 // gap
   272→    for (const prop of q.propositions) {
   273→      height += getTextHeight(doc, prop.text, CONTENT_WIDTH - 15, FONT_BODY) + 4
   274→    }
   275→  }
   276→
   277→  // Response lines for QRC
   278→  if (q.type === 'QRC') {
   279→    height += 3 + 5 * ANSWER_LINE_SPACING
   280→  }
   281→
   282→  // Response lines for REFLEXION
   283→  if (q.type === 'REFLEXION') {
   284→    height += 3 + 8 * ANSWER_LINE_SPACING
   285→  }
   286→
   287→  // Code section for CODE type
   288→  if (q.type === 'CODE') {
   289→    height += 3
   290→    // Language badge + signature
   291→    height += 8
   292→    // Starter code box
   293→    if (q.codeInitial) {
   294→      const codeLines = q.codeInitial.split('\n').length
   295→      height += Math.max(codeLines * 4 + 10, 40) // 4mm per code line, min 40mm
   296→    } else {
   297→      height += 40 // Empty code area
   298→    }
   299→    // Public tests table
   300→    if (q.testsPublics && q.testsPublics.length > 0) {
   301→      height += 8 + q.testsPublics.length * 8
   302→    }
   303→  }
   304→
   305→  // Bottom gap + separator
   306→  height += QUESTION_GAP + 3
   307→
   308→  return height
   309→}

   311→/**
   312→ * Render a single question in Sujet mode.
   313→ * Returns the Y position after rendering.
   314→ */
   315→function renderQuestionSujet(doc: jsPDF, q: PDFQuestion, index: number, startY: number): number {
   316→  let y = startY
   317→
   318→  // Question header — measure width at the SAME font size before switching
   319→  doc.setFontSize(FONT_HEADING)
   320→  doc.setFont('helvetica', 'bold')
   321→  doc.text(`Question ${index + 1}`, MARGIN_LEFT, y)
   322→  const questionLabelWidth = doc.getTextWidth(`Question ${index + 1}  `) // Measure at size 11 bold
   323→
   324→  // Type and bareme
   325→  doc.setFontSize(FONT_SMALL)
   326→  doc.setFont('helvetica', 'normal')
   327→  const typeLabel = getTypeLabel(q.type)
   328→  doc.text(typeLabel, MARGIN_LEFT + questionLabelWidth, y)
   329→
   330→  // Bareme on the right
   331→  doc.setFont('helvetica', 'bold')
   332→  const baremeText = `${q.bareme} pt${q.bareme > 1 ? 's' : ''}`
   333→  doc.text(baremeText, PAGE_WIDTH - MARGIN_RIGHT, y, { align: 'right' })
   334→
   335→  y += 6
   336→
   337→  // Question text
   338→  doc.setFontSize(FONT_BODY)
   339→  doc.setFont('helvetica', 'normal')
   340→  const lines = wrapText(doc, q.enonce, CONTENT_WIDTH - 5)
   341→  for (const line of lines) {
   342→    doc.text(line, MARGIN_LEFT, y)
   343→    y += LINE_HEIGHT_BODY
   344→  }
   345→
   346→  y += 2
   347→
   348→  // Propositions for QCU/QCM
   349→  if ((q.type === 'QCU' || q.type === 'QCM') && q.propositions && q.propositions.length > 0) {
   350→    for (let pi = 0; pi < q.propositions.length; pi++) {
   351→      const prop = q.propositions[pi]
   352→      const letter = String.fromCharCode(65 + pi)
   353→
   354→      if (q.type === 'QCU') {
   355→        // Draw circle for QCU
   356→        doc.setDrawColor(80, 80, 80)
   357→        doc.setLineWidth(0.4)
   358→        doc.circle(MARGIN_LEFT + 3, y - 1, 2.5)
   359→        doc.setFontSize(FONT_SMALL)
   360→        doc.setFont('helvetica', 'bold')
   361→        doc.text(letter, MARGIN_LEFT + 3, y - 0.5, { align: 'center' })
   362→      } else {
   363→        // Draw square for QCM
   364→        doc.setDrawColor(80, 80, 80)
   365→        doc.setLineWidth(0.4)
   366→        doc.rect(MARGIN_LEFT + 0.5, y - 3.5, 5, 5)
   367→        doc.setFontSize(FONT_SMALL)
   368→        doc.setFont('helvetica', 'bold')
   369→        doc.text(letter, MARGIN_LEFT + 3, y - 0.5, { align: 'center' })
   370→      }
   371→
   372→      // Proposition text
   373→      doc.setFontSize(FONT_BODY)
   374→      doc.setFont('helvetica', 'normal')
   375→      const propLines = wrapText(doc, prop.text, CONTENT_WIDTH - 15)
   376→      for (const line of propLines) {
   377→        doc.text(line, MARGIN_LEFT + 10, y)
   378→        y += LINE_HEIGHT_BODY
   379→      }
   380→      y += 1
   381→    }
   382→  }
   383→
   384→  // Response lines for QRC
   385→  if (q.type === 'QRC') {
   386→    y += 2
   387→    const numLines = 5
   388→    for (let i = 0; i < numLines; i++) {
   389→      doc.setDrawColor(190, 190, 190)
   390→      doc.setLineWidth(0.3)
   391→      doc.line(MARGIN_LEFT, y, PAGE_WIDTH - MARGIN_RIGHT, y)
   392→      y += ANSWER_LINE_SPACING
   393→    }
   394→  }
   395→
   396→  // Response lines for REFLEXION
   397→  if (q.type === 'REFLEXION') {
   398→    y += 2
   399→    const numLines = 8
   400→    for (let i = 0; i < numLines; i++) {
   401→      doc.setDrawColor(190, 190, 190)
   402→      doc.setLineWidth(0.3)
   403→      doc.line(MARGIN_LEFT, y, PAGE_WIDTH - MARGIN_RIGHT, y)
   404→      y += ANSWER_LINE_SPACING
   405→    }
   406→  }
   407→
   408→  // Code section for CODE type
   409→  if (q.type === 'CODE') {
   410→    y += 2
   411→
   412→    // Language + signature info
   413→    doc.setFontSize(FONT_SMALL)
   414→    doc.setFont('helvetica', 'bold')
   415→    doc.setTextColor(100, 60, 160)
   416→    const langLabel = q.langage ? q.langage.toUpperCase() : 'CODE'
   417→    doc.text(`Langage : ${langLabel}`, MARGIN_LEFT, y)
   418→    if (q.fonctionSignature) {
   419→      doc.setFont('helvetica', 'normal')
   420→      doc.setTextColor(80, 80, 80)
   421→      doc.text(`Signature : ${q.fonctionSignature}`, MARGIN_LEFT + 50, y)
   422→    }
   423→    doc.setTextColor(0, 0, 0)
   424→    y += 6
   425→
   426→    // Starter code box
   427→    doc.setFillColor(245, 245, 250)
   428→    doc.setDrawColor(180, 170, 210)
   429→    doc.setLineWidth(0.4)
   430→    const codeText = q.codeInitial || '// Écrire votre code ici'
   431→    const codeLines = codeText.split('\n')
   432→    const codeBoxHeight = Math.max(codeLines.length * 4 + 10, 35)
   433→    doc.roundedRect(MARGIN_LEFT, y - 3, CONTENT_WIDTH, codeBoxHeight, 2, 2, 'FD')
   434→
   435→    // Code text
   436→    doc.setFontSize(9)
   437→    doc.setFont('courier', 'normal')
   438→    doc.setTextColor(40, 40, 60)
   439→    let codeY = y + 2
   440→    for (const line of codeLines.slice(0, 20)) { // Max 20 lines to prevent overflow
   441→      if (codeY > y + codeBoxHeight - 5) break
   442→      doc.text(line.substring(0, 80), MARGIN_LEFT + 4, codeY) // Truncate long lines
   443→      codeY += 4
   444→    }
   445→    doc.setTextColor(0, 0, 0)
   446→    y += codeBoxHeight + 4
   447→
   448→    // Public tests table
   449→    if (q.testsPublics && q.testsPublics.length > 0) {
   450→      doc.setFontSize(FONT_SMALL)
   451→      doc.setFont('helvetica', 'bold')
   452→      doc.text('Tests publics :', MARGIN_LEFT, y)
   453→      y += 5
   454→
   455→      const testHead = [['N°', 'Entrée', 'Sortie attendue']]
   456→      const testBody = q.testsPublics.map((t, i) => [
   457→        `${i + 1}`,
   458→        t.entree.substring(0, 40),
   459→        t.sortieAttendue.substring(0, 40),
   460→      ])
   461→
   462→      autoTable(doc, {
   463→        head: testHead,
   464→        body: testBody,
   465→        startY: y,
   466→        margin: { left: MARGIN_LEFT + 5, right: MARGIN_RIGHT },
   467→        styles: {
   468→          fontSize: FONT_TINY,
   469→          cellPadding: 2,
   470→          lineColor: [180, 170, 210],
   471→          lineWidth: 0.2,
   472→        },
   473→        headStyles: {
   474→          fillColor: [100, 60, 160],
   475→          textColor: [255, 255, 255],
   476→          fontStyle: 'bold',
   477→          fontSize: 7,
   478→        },
   479→        columnStyles: {
   480→          0: { cellWidth: 10 },
   481→          1: { cellWidth: 60 },
   482→          2: { cellWidth: 60 },
   483→        },
   484→      })
   485→
   486→      y = (doc as any).lastAutoTable?.finalY ?? y + 20
   487→      y += 3
   488→    }
   489→  }
   490→
   491→  // Separator line between questions
   492→  y += 3
   493→  doc.setDrawColor(220, 220, 220)
   494→  doc.setLineWidth(0.2)
   495→  doc.line(MARGIN_LEFT, y, PAGE_WIDTH - MARGIN_RIGHT, y)
   496→  y += QUESTION_GAP
   497→
   498→  return y
   499→}

   502→
   503→/**
   504→ * Estimate the height needed for a question in Corrigé mode.
   505→ */
   506→function estimateQuestionHeightCorrige(doc: jsPDF, q: PDFQuestion): number {
   507→  let height = estimateQuestionHeightSujet(doc, q)
   508→
   509→  // Additional height for correct answers / model response
   510→  if ((q.type === 'QCU' || q.type === 'QCM') && q.reponseCorrecte) {
   511→    height += LINE_HEIGHT_BODY + 12  // Label line + answer content + spacing
   512→  }
   513→
   514→  if ((q.type === 'QRC' || q.type === 'REFLEXION') && q.reponseCorrecte) {
   515→    const text = Array.isArray(q.reponseCorrecte) ? q.reponseCorrecte.join('\n') : q.reponseCorrecte
   516→    height += getTextHeight(doc, text, CONTENT_WIDTH - 15, FONT_BODY) + LINE_HEIGHT_BODY + 8
   517→  }
   518→
   519→  if (q.explication) {
   520→    height += getTextHeight(doc, q.explication, CONTENT_WIDTH - 15, FONT_BODY) + LINE_HEIGHT_BODY + 8
   521→  }
   522→
   523→  return height
   524→}

   526→/**
   527→ * Render a single question in Corrigé mode.
   528→ * Returns the Y position after rendering.
   529→ */
   530→function renderQuestionCorrige(doc: jsPDF, q: PDFQuestion, index: number, startY: number): number {
   531→  let y = startY
   532→
   533→  // First render the question like Sujet
   534→  y = renderQuestionSujet(doc, q, index, y)
   535→
   536→  // Correct answers for QCU/QCM
   537→  if ((q.type === 'QCU' || q.type === 'QCM') && q.propositions && q.reponseCorrecte) {
   538→    const correctIds = Array.isArray(q.reponseCorrecte) ? q.reponseCorrecte : [q.reponseCorrecte]
   539→
   540→    doc.setFontSize(FONT_SMALL)
   541→    doc.setFont('helvetica', 'bold')
   542→    doc.setTextColor(0, 130, 60)
   543→    doc.text('Réponse correcte :', MARGIN_LEFT, y)
   544→    y += LINE_HEIGHT_BODY  // Advance past the label to avoid overlap
   545→
   546→    const correctLabels = q.propositions
   547→      .filter((p) => correctIds.includes(p.id))
   548→      .map((p) => {
   549→        const idx = q.propositions!.indexOf(p)
   550→        return `${String.fromCharCode(65 + idx)}. ${p.text}`
   551→      })
   552→
   553→    doc.setFont('helvetica', 'normal')
   554→    doc.setFontSize(FONT_BODY)
   555→    for (const label of correctLabels) {
   556→      const lines = wrapText(doc, label, CONTENT_WIDTH - 20)
   557→      for (const line of lines) {
   558→        doc.text(line, MARGIN_LEFT + 10, y)
   559→        y += LINE_HEIGHT_BODY
   560→      }
   561→    }
   562→    doc.setTextColor(0, 0, 0)
   563→    y += 3
   564→  }
   565→
   566→  // Model response for QRC / REFLEXION
   567→  if ((q.type === 'QRC' || q.type === 'REFLEXION') && q.reponseCorrecte) {
   568→    const text = Array.isArray(q.reponseCorrecte) ? q.reponseCorrecte.join('\n') : q.reponseCorrecte
   569→    const label = q.type === 'QRC' ? 'Réponse modèle :' : 'Guide de correction :'
   570→
   571→    doc.setFontSize(FONT_SMALL)
   572→    doc.setFont('helvetica', 'bold')
   573→    doc.setTextColor(0, 100, 50)
   574→    doc.text(label, MARGIN_LEFT, y)
   575→    doc.setTextColor(0, 0, 0)
   576→    y += LINE_HEIGHT_BODY  // Advance past the label to avoid overlap
   577→
   578→    doc.setFontSize(FONT_BODY)
   579→    doc.setFont('helvetica', 'normal')
   580→    const lines = wrapText(doc, text, CONTENT_WIDTH - 15)
   581→    const boxHeight = lines.length * LINE_HEIGHT_BODY + 6
   582→
   583→    doc.setFillColor(245, 250, 245)
   584→    doc.setDrawColor(0, 160, 80)
   585→    doc.roundedRect(MARGIN_LEFT, y - 3, CONTENT_WIDTH, boxHeight, 2, 2, 'FD')
   586→
   587→    doc.setTextColor(0, 80, 40)
   588→    for (const line of lines) {
   589→      doc.text(line, MARGIN_LEFT + 5, y)
   590→      y += LINE_HEIGHT_BODY
   591→    }
   592→    doc.setTextColor(0, 0, 0)
   593→    y += 5
   594→  }
   595→
   596→  // Explanation
   597→  if (q.explication) {
   598→    doc.setFontSize(FONT_SMALL)
   599→    doc.setFont('helvetica', 'bold')
   600→    doc.setTextColor(80, 80, 140)
   601→    doc.text('Explication :', MARGIN_LEFT, y)
   602→    doc.setTextColor(0, 0, 0)
   603→    y += LINE_HEIGHT_BODY  // Advance past the label to avoid overlap
   604→
   605→    doc.setFontSize(FONT_BODY)
   606→    doc.setFont('helvetica', 'normal')
   607→    doc.setTextColor(80, 80, 80)
   608→    const lines = wrapText(doc, q.explication, CONTENT_WIDTH - 10)
   609→    for (const line of lines) {
   610→      doc.text(line, MARGIN_LEFT + 5, y)
   611→      y += LINE_HEIGHT_BODY
   612→    }
   613→    doc.setTextColor(0, 0, 0)
   614→    y += 3
   615→  }
   616→
   617→  return y
   618→}

   707→  return doc

   796→/**
   797→ * Generate the Feuille de Réponses PDF (answer sheet for QCM/QCU).
   798→ */
   799→export function generateFeuilleReponsesPDF(data: EpreuvePDFData): jsPDF {
   800→  const doc = new jsPDF('p', 'mm', 'a4')
   801→  doc.setProperties({
   802→    title: `Feuille de réponses - ${data.titre}`,
   803→    author: data.etablissement.nom,
   804→    subject: `Feuille de réponses - ${data.titre}`,
   805→  })
   806→
   807→  addHeader(doc, data, false)
   808→
   809→  let y = CONTENT_TOP
   810→
   811→  // Title
   812→  doc.setFontSize(FONT_TITLE)
   813→  doc.setFont('helvetica', 'bold')
   814→  doc.text('FEUILLE DE RÉPONSES', MARGIN_LEFT, y)
   815→  y += 8
   816→
   817→  doc.setFontSize(FONT_SUBTITLE)
   818→  doc.text(data.titre, MARGIN_LEFT, y)
   819→  y += 8
   820→
   821→  // Student info fields
   822→  doc.setFontSize(FONT_BODY)
   823→  doc.setFont('helvetica', 'normal')
   824→  doc.text('Nom : ', MARGIN_LEFT, y)
   825→  doc.setDrawColor(150, 150, 150)
   826→  doc.line(MARGIN_LEFT + 15, y, MARGIN_LEFT + 80, y)
   827→  doc.text('Prénom : ', MARGIN_LEFT + 85, y)
   828→  doc.line(MARGIN_LEFT + 102, y, PAGE_WIDTH - MARGIN_RIGHT, y)
   829→  y += 8
   830→
   831→  doc.text('Matricule : ', MARGIN_LEFT, y)
   832→  doc.line(MARGIN_LEFT + 25, y, MARGIN_LEFT + 80, y)
   833→  doc.text('Filière : ', MARGIN_LEFT + 85, y)
   834→  doc.line(MARGIN_LEFT + 105, y, PAGE_WIDTH - MARGIN_RIGHT, y)
   835→  y += 12
   836→
   837→  // Separator
   838→  doc.setDrawColor(200, 200, 200)
   839→  doc.setLineWidth(0.3)
   840→  doc.line(MARGIN_LEFT, y, PAGE_WIDTH - MARGIN_RIGHT, y)
   841→  y += 8
   842→
   843→  // Filter QCU and QCM questions
   844→  const mcqQuestions = data.questions.filter((q) => q.type === 'QCU' || q.type === 'QCM')
   845→
   846→  if (mcqQuestions.length === 0) {
   847→    doc.setFontSize(FONT_BODY)
   848→    doc.setFont('helvetica', 'italic')
   849→    doc.text('Aucune question QCM/QCU dans cette épreuve.', MARGIN_LEFT, y)
   850→  } else {
   851→    // Build the answer grid using autoTable
   852→    const maxProps = Math.max(
   853→      ...mcqQuestions.map((q) => q.propositions?.length || 0),
   854→      4
   855→    )
   856→    const letterCols = Array.from({ length: maxProps }, (_, i) =>
   857→      String.fromCharCode(65 + i)
   858→    )
   859→
   860→    // Table header
   861→    const head = [['N°', 'Type', ...letterCols]]
   862→
   863→    // Table body
   864→    const body = mcqQuestions.map((q) => {
   865→      const originalIdx = data.questions.indexOf(q)
   866→      const cells: string[] = [
   867→        `${originalIdx + 1}`,
   868→        q.type,
   869→      ]
   870→      for (let c = 0; c < maxProps; c++) {
   871→        cells.push(q.propositions && c < q.propositions.length ? '__CIRCLE__' : '')
   872→      }
   873→      return cells
   874→    })
   875→
   876→    autoTable(doc, {
   877→      head,
   878→      body,
   879→      startY: y,
   880→      margin: { left: MARGIN_LEFT, right: MARGIN_RIGHT },
   881→      styles: {
   882→        fontSize: FONT_BODY,
   883→        cellPadding: 3,
   884→        halign: 'center',
   885→        valign: 'middle',
   886→        lineColor: [180, 180, 180],
   887→        lineWidth: 0.3,
   888→      },
   889→      headStyles: {
   890→        fillColor: [35, 120, 75],
   891→        textColor: [255, 255, 255],
   892→        fontStyle: 'bold',
   893→        fontSize: FONT_SMALL,
   894→      },
   895→      columnStyles: {
   896→        0: { halign: 'center', cellWidth: 15, fontStyle: 'bold' },
   897→        1: { halign: 'center', cellWidth: 20 },
   898→      },
   899→      alternateRowStyles: {
   900→        fillColor: [245, 250, 248],
   901→      },
   902→      didParseCell: (hookData) => {
   903→        // Replace __CIRCLE__ with empty string so it doesn't render as text
   904→        if (hookData.cell.raw === '__CIRCLE__') {
   905→          hookData.cell.raw = ''
   906→          hookData.cell.text = ''  // Clear text content to prevent any rendering
   907→          ;(hookData.cell as any)._isCircle = true
   908→        }
   909→      },
   910→      didDrawCell: (hookData) => {
   911→        // Draw a circle in cells that had __CIRCLE__
   912→        if ((hookData.cell as any)._isCircle) {
   913→          const { x, y: cellY, width, height } = hookData.cell
   914→          const cx = x + width / 2
   915→          const cy = cellY + height / 2
   916→          const radius = Math.min(width, height) / 2 - 1.5
   917→          doc.setDrawColor(60, 60, 60)
   918→          doc.setLineWidth(0.5)
   919→          doc.circle(cx, cy, radius)
   920→        }
   921→      },
   922→    })
   923→
   924→    // Get the Y position after the table
   925→    y = (doc as any).lastAutoTable?.finalY ?? y + 20
   926→    y += 10
   927→  }
   928→
   929→  // Open-ended questions section
   930→  const openQuestions = data.questions.filter((q) => q.type === 'QRC' || q.type === 'REFLEXION')
   931→
   932→  if (openQuestions.length > 0) {
   933→    y = ensureSpace(doc, data, y, 20, false)
   934→
   935→    doc.setFontSize(FONT_HEADING)
   936→    doc.setFont('helvetica', 'bold')
   937→    doc.text('Questions ouvertes', MARGIN_LEFT, y)
   938→    y += 8
   939→
   940→    for (const q of openQuestions) {
   941→      const originalIdx = data.questions.indexOf(q)
   942→      const numLines = q.type === 'QRC' ? 5 : 8
   943→      const neededHeight = numLines * ANSWER_LINE_SPACING + 20
   944→
   945→      y = ensureSpace(doc, data, y, neededHeight, false)
   946→
   947→      doc.setFontSize(FONT_BODY)
   948→      doc.setFont('helvetica', 'bold')
   949→      doc.text(`Question ${originalIdx + 1} (${q.type} - ${q.bareme} pts)`, MARGIN_LEFT, y)
   950→      y += 5
   951→
   952→      doc.setFont('helvetica', 'normal')
   953→      const lines = wrapText(doc, q.enonce, CONTENT_WIDTH - 5)
   954→      for (const line of lines) {
   955→        doc.text(line, MARGIN_LEFT, y)
   956→        y += LINE_HEIGHT_BODY
   957→      }
   958→      y += 3
   959→
   960→      // Response lines
   961→      for (let i = 0; i < numLines; i++) {
   962→        doc.setDrawColor(190, 190, 190)
   963→        doc.setLineWidth(0.3)
   964→        doc.line(MARGIN_LEFT, y, PAGE_WIDTH - MARGIN_RIGHT, y)
   965→        y += ANSWER_LINE_SPACING
   966→      }
   967→      y += 5
   968→    }
   969→  }
   970→
   971→  // Add footers
   972→  const totalPages = doc.getNumberOfPages()
   973→  for (let p = 1; p <= totalPages; p++) {
   974→    doc.setPage(p)
   975→    addFooter(doc, data, p, totalPages, false)
   976→  }
   977→
   978→  return doc
   979→}
