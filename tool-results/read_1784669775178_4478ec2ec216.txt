   115→// computeSurvRiskScore — calcule un score 0-100 à partir des métriques.
   116→// Formule : min(100, alertes*8 + pénalité*2 + bonus par événements critiques).
   117→func computeSurvRiskScore(alertes int, penalite float64, fraudEvents []survLogEvent) int {
   118→        criticalCount := 0
   119→        for _, e := range fraudEvents {
   120→                switch e.Type {
   121→                case "FULLSCREEN_EXIT", "TAB_SWITCH", "DEVTOOLS_ATTEMPT":
   122→                        criticalCount++
   123→                }
   124→        }
   125→        score := alertes*8 + int(penalite)*2 + criticalCount*5
   126→        if score > 100 {
   127→                score = 100
   128→        }
   129→        if score < 0 {
   130→                score = 0
   131→        }
   132→        return score
   133→}
   134→
   135→// riskLevelFromScore — dérive le niveau de risque depuis le score.
   136→func riskLevelFromScore(score int) string {
   137→        switch {
   138→        case score >= 70:
   139→                return "critical"
   140→        case score >= 40:
   141→                return "high"
   142→        case score >= 15:
   143→                return "moderate"
   144→        default:
   145→                return "safe"
   146→        }
   147→}

   149→func (s *Server) surveillanceListSessions(w http.ResponseWriter, r *http.Request) {
   150→        claims, ok := middleware.ClaimsFromContext(r.Context())
   151→        if !ok || claims.UserID == "" {
   152→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
   153→                return
   154→        }
   155→
   156→        // Scope enseignant : un ENSEIGNANT ne voit que ses sessions ; un ADMIN/RESPONSABLE
   157→        // peut explicitement filtrer via ?enseignantId=.
   158→        enseignantID := r.URL.Query().Get("enseignantId")
   159→        if enseignantID == "" && claims.Role == "ENSEIGNANT" {
   160→                enseignantID = claims.UserID
   161→        }
   162→
   163→        // Filtres UI (SURVEILLANCE-FIX-2 S5) : epreuveId, severity, type, search.
   164→        // UX-IMPROVE : ajout dateDebut/dateFin pour filtrer par date d'épreuve.
   165→        epreuveFilter := r.URL.Query().Get("epreuveId")
   166→        severityFilter := r.URL.Query().Get("severity")
   167→        typeFilter := r.URL.Query().Get("type")
   168→        searchFilter := r.URL.Query().Get("search")
   169→        dateDebutFilter := r.URL.Query().Get("dateDebut")
   170→        dateFinFilter := r.URL.Query().Get("dateFin")
   171→        // UX-IMPROVE : optionsOnly=true → retourner uniquement la liste des
   172→        // épreuves (pour le dropdown) sans fetcher les sessions. Évite de
   173→        // charger 200 sessions juste pour peupler un dropdown.
   174→        optionsOnly := r.URL.Query().Get("optionsOnly") == "true"
   175→
   176→        type rawSession struct {
   177→                ID              string
   178→                EtudiantID      string
   179→                EtudiantNom     string
   180→                EtudiantEmail   string
   181→                EpreuveID       string
   182→                EpreuveTitre    string
   183→                EpreuveStatut   string
   184→                EpreuveDebut    *time.Time
   185→                EpreuveFin      *time.Time
   186→                ProctoringActif bool
   187→                SessionStatut   string
   188→                DateDebut       *time.Time
   189→                DateFin         *time.Time
   190→                Score           *float64
   191→                Alertes         int
   192→                Penalite        float64
   193→                LogEventsRaw    []byte // logEvents TEXT parsé en []byte
   194→                Flagged         bool   // EXISTS Alerte FRAUDE
   195→        }
   196→
   197→        var rawSessions []rawSession
   198→        var epreuveOptions []survEpreuveOption
   199→        var txErr error
   200→
   201→        txErr = appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   202→                var where []string
   203→                var args []any
   204→                argIdx := 1
   205→
   206→                if enseignantID != "" {
   207→                        where = append(where, fmt.Sprintf(`e."enseignantId" = $%d`, argIdx))
   208→                        args = append(args, enseignantID)
   209→                        argIdx++
   210→                }
   211→                if epreuveFilter != "" {
   212→                        where = append(where, fmt.Sprintf(`s."epreuveId" = $%d`, argIdx))
   213→                        args = append(args, epreuveFilter)
   214→                        argIdx++
   215→                }
   216→                if searchFilter != "" {
   217→                        where = append(where, fmt.Sprintf(`(u."name" ILIKE $%d OR u."email" ILIKE $%d)`, argIdx, argIdx))
   218→                        args = append(args, "%"+searchFilter+"%")
   219→                        argIdx++
   220→                }
   221→                // UX-IMPROVE : skip date filters si optionsOnly (on veut toutes
   222→                // les épreuves, pas juste celles d'une date spécifique).
   223→                // UX-IMPROVE : filtre par date d'épreuve (dateDebut/dateFin).
   224→                // Permet à l'enseignant de cibler une session spécifique dans le temps
   225→                // au lieu de charger toutes les sessions d'un coup.
   226→                if dateDebutFilter != "" {
   227→                        where = append(where, fmt.Sprintf(`e."dateDebut" >= $%d`, argIdx))
   228→                        args = append(args, dateDebutFilter)
   229→                        argIdx++
   230→                }
   231→                if dateFinFilter != "" {
   232→                        where = append(where, fmt.Sprintf(`e."dateFin" <= $%d`, argIdx))
   233→                        args = append(args, dateFinFilter)
   234→                        argIdx++
   235→                }
   236→
   237→                whereClause := ""
   238→                if len(where) > 0 {
   239→                        whereClause = "WHERE " + strings.Join(where, " AND ")
   240→                }
   241→
   242→                // UX-IMPROVE : si optionsOnly, skip la query sessions (on veut
   243→                // juste les epreuve options pour le dropdown). Économise le
   244→                // fetch de 200 rows.
   245→                if !optionsOnly {
   246→                        query := fmt.Sprintf(`
   247→                        SELECT s."id", s."etudiantId",
   248→                               COALESCE(u."name", '') AS etudiant_nom,
   249→                               COALESCE(u."email", '') AS etudiant_email,
   250→                               s."epreuveId",
   251→                               COALESCE(e."titre", '') AS epreuve_titre,
   252→                               COALESCE(e."statut"::text, '') AS epreuve_statut,
   253→                               e."dateDebut" AS epreuve_debut,
   254→                               e."dateFin" AS epreuve_fin,
   255→                               COALESCE(e."proctoringActif", false) AS proctoring,
   256→                               s."statut"::text AS session_statut,
   257→                               s."dateDebut", s."dateFin", s."score",
   258→                               COALESCE(s."alertes", 0), COALESCE(s."penalite", 0),
   259→                               COALESCE(s."logEvents"::text, '[]')::bytea AS log_events,
   260→                               EXISTS(
   261→                                 SELECT 1 FROM "Alerte" a
   262→                                 WHERE a."epreuveId" = s."epreuveId"
   263→                                   AND a."userId" = s."etudiantId"
   264→                                   AND a."type" = 'FRAUDE'
   265→                               ) AS flagged
   266→                        FROM "SessionPassation" s
   267→                        LEFT JOIN "User" u ON u."id" = s."etudiantId"
   268→                        LEFT JOIN "Epreuve" e ON e."id" = s."epreuveId"
   269→                        %s
   270→                        ORDER BY s."createdAt" DESC
   271→                        LIMIT 200
   272→                `, whereClause)
   273→
   274→                        rows, err := tx.Query(r.Context(), query, args...)
   275→                        if err != nil {
   276→                                return fmt.Errorf("query sessions: %w", err)
   277→                        }
   278→                        defer rows.Close()
   279→
   280→                        for rows.Next() {
   281→                                rs := rawSession{}
   282→                                if err := rows.Scan(
   283→                                        &rs.ID, &rs.EtudiantID, &rs.EtudiantNom, &rs.EtudiantEmail,
   284→                                        &rs.EpreuveID, &rs.EpreuveTitre, &rs.EpreuveStatut,
   285→                                        &rs.EpreuveDebut, &rs.EpreuveFin, &rs.ProctoringActif,
   286→                                        &rs.SessionStatut, &rs.DateDebut, &rs.DateFin, &rs.Score,
   287→                                        &rs.Alertes, &rs.Penalite, &rs.LogEventsRaw, &rs.Flagged,
   288→                                ); err != nil {
   289→                                        return fmt.Errorf("scan session: %w", err)
   290→                                }
   291→                                rawSessions = append(rawSessions, rs)
   292→                        }
   293→                        if err := rows.Err(); err != nil {
   294→                                return fmt.Errorf("rows iteration: %w", err)
   295→                        }
   296→                } // fin if !optionsOnly
   297→
   298→                // Récupération des épreuves pour le dropdown filtre (S4).
   299→                var eArgs []any
   300→                eIdx := 1
   301→                eWhere := ""
   302→                if enseignantID != "" {
   303→                        eWhere = fmt.Sprintf(`WHERE e."enseignantId" = $%d`, eIdx)
   304→                        eArgs = append(eArgs, enseignantID)
   305→                        eIdx++
   306→                }
   307→                epreuvesQuery := fmt.Sprintf(`
   308→                        SELECT e."id", e."titre", e."statut"::text, e."dateDebut", e."dateFin",
   309→                               COALESCE(e."proctoringActif", false),
   310→                               COALESCE(sum(s."alertes"), 0) AS total_alerts,
   311→                               count(*) FILTER (WHERE s."alertes" > 0) AS sessions_with_alerts,
   312→                               count(*) AS total_sessions
   313→                        FROM "Epreuve" e
   314→                        LEFT JOIN "SessionPassation" s ON s."epreuveId" = e."id"
   315→                        %s
   316→                        GROUP BY e."id", e."titre", e."statut", e."dateDebut", e."dateFin", e."proctoringActif"
   317→                        ORDER BY e."titre" ASC
   318→                        LIMIT 100
   319→                `, eWhere)
   320→                epRows, err := tx.Query(r.Context(), epreuvesQuery, eArgs...)
   321→                if err != nil {
   322→                        return fmt.Errorf("query epreuves: %w", err)
   323→                }
   324→                defer epRows.Close()
   325→                for epRows.Next() {
   326→                        eo := survEpreuveOption{}
   327→                        var debut, fin *time.Time
   328→                        if err := epRows.Scan(&eo.ID, &eo.Titre, &eo.Statut, &debut, &fin,
   329→                                &eo.ProctoringActif, &eo.TotalAlerts, &eo.SessionsWithAlerts, &eo.TotalSessions); err != nil {
   330→                                return fmt.Errorf("scan epreuve: %w", err)
   331→                        }
   332→                        if debut != nil {
   333→                                ts := debut.UTC().Format(time.RFC3339)
   334→                                eo.DateDebut = &ts
   335→                        }
   336→                        if fin != nil {
   337→                                ts := fin.UTC().Format(time.RFC3339)
   338→                                eo.DateFin = &ts
   339→                        }
   340→                        epreuveOptions = append(epreuveOptions, eo)
   341→                }
   342→                if err := epRows.Err(); err != nil {
   343→                        return fmt.Errorf("epreuves rows: %w", err)
   344→                }
   345→
   346→                return nil
   347→        })
   348→
   349→        if txErr != nil {
   350→                // SURVEILLANCE-FIX-2 S12 : propager l'erreur SQL au lieu de renvoyer 200 vide.
   351→                http.Error(w, `{"error":"`+txErr.Error()+`"}`, http.StatusInternalServerError)
   352→                return
   353→        }
   354→
   355→        // Mapping rawSession → survSession avec partition logEvents.
   356→        sessions := make([]survSession, 0, len(rawSessions))
   357→        for _, rs := range rawSessions {
   358→                // Parse logEvents JSON → []survLogEvent.
   359→                var logEvents []survLogEvent
   360→                if len(rs.LogEventsRaw) > 0 && string(rs.LogEventsRaw) != "[]" {
   361→                        if err := json.Unmarshal(rs.LogEventsRaw, &logEvents); err != nil {
   362→                                // logEvents corrompu : on ne crash pas, on logge et continue avec tableau vide.
   363→                                logEvents = []survLogEvent{}
   364→                        }
   365→                } else {
   366→                        logEvents = []survLogEvent{}
   367→                }
   368→
   369→                // Partition : fraudEvents / screenshotEvents / submissionEvents.
   370→                fraudEvents := make([]survLogEvent, 0)
   371→                screenshotEvents := make([]survLogEvent, 0)
   372→                submissionEvents := make([]survLogEvent, 0)
   373→                for _, evt := range logEvents {
   374→                        switch {
   375→                        case screenshotEventTypes[evt.Type]:
   376→                                screenshotEvents = append(screenshotEvents, evt)
   377→                        case submissionEventTypes[evt.Type]:
   378→                                submissionEvents = append(submissionEvents, evt)
   379→                        case benignEventTypes[evt.Type]:
   380→                                // ignoré
   381→                        default:
   382→                                fraudEvents = append(fraudEvents, evt)
   383→                        }
   384→                }
   385→
   386→                // Filtre severity (post-fetch) : high/medium/low/info selon getSeverityLevel.
   387→                if severityFilter != "" && severityFilter != "all" {
   388→                        keep := false
   389→                        for _, evt := range logEvents {
   390→                                if severityOfEvent(evt.Type) == severityFilter {
   391→                                        keep = true
   392→                                        break
   393→                                }
   394→                        }
   395→                        if !keep {
   396→                                continue
   397→                        }
   398→                }
   399→
   400→                // Filtre type (post-fetch) : au moins un événement de ce type dans la session.
   401→                if typeFilter != "" && typeFilter != "all" {
   402→                        keep := false
   403→                        for _, evt := range logEvents {
   404→                                if evt.Type == typeFilter {
   405→                                        keep = true
   406→                                        break
   407→                                }
   408→                        }
   409→                        if !keep {
   410→                                continue
   411→                        }
   412→                }
   413→
   414→                // Calcul riskScore / riskLevel / totalPenalite (S6).
   415→                totalPenalite := rs.Penalite
   416→                for _, evt := range logEvents {
   417→                        totalPenalite += evt.Penalite
   418→                }
   419→                riskScore := computeSurvRiskScore(rs.Alertes, totalPenalite, fraudEvents)
   420→                riskLevel := riskLevelFromScore(riskScore)
   421→
   422→                ss := survSession{
   423→                        ID:       rs.ID,
   424→                        Statut:   rs.SessionStatut,
   425→                        Score:    rs.Score,
   426→                        Penalite: rs.Penalite,
   427→                        Alertes:  rs.Alertes,
   428→                        Etudiant: survEtudiantRef{
   429→                                ID:    rs.EtudiantID,
   430→                                Name:  rs.EtudiantNom,
   431→                                Email: rs.EtudiantEmail,
   432→                        },
   433→                        Epreuve: survEpreuveRef{
   434→                                ID:              rs.EpreuveID,
   435→                                Titre:           rs.EpreuveTitre,
   436→                                Statut:          rs.EpreuveStatut,
   437→                                ProctoringActif: rs.ProctoringActif,
   438→                        },
   439→                        LogEvents:        logEvents,
   440→                        FraudEvents:      fraudEvents,
   441→                        ScreenshotEvents: screenshotEvents,
   442→                        SubmissionEvents: submissionEvents,
   443→                        TotalPenalite:    totalPenalite,
   444→                        RiskScore:        riskScore,
   445→                        RiskLevel:        riskLevel,
   446→                        Flagged:          rs.Flagged,
   447→                }
   448→                if rs.DateDebut != nil {
   449→                        ts := rs.DateDebut.UTC().Format(time.RFC3339)
   450→                        ss.DateDebut = &ts
   451→                }
   452→                if rs.DateFin != nil {
   453→                        ts := rs.DateFin.UTC().Format(time.RFC3339)
   454→                        ss.DateFin = &ts
   455→                }
   456→                if rs.EpreuveDebut != nil {
   457→                        ts := rs.EpreuveDebut.UTC().Format(time.RFC3339)
   458→                        ss.Epreuve.DateDebut = &ts
   459→                }
   460→                if rs.EpreuveFin != nil {
   461→                        ts := rs.EpreuveFin.UTC().Format(time.RFC3339)
   462→                        ss.Epreuve.DateFin = &ts
   463→                }
   464→                sessions = append(sessions, ss)
   465→        }
   466→
   467→        w.Header().Set("Content-Type", "application/json")
   468→        json.NewEncoder(w).Encode(map[string]any{
   469→                "sessions": sessions,
   470→                "epreuves": epreuveOptions,
   471→                "total":    len(sessions),
   472→        })
   473→}

   481→        case "COPY_ATTEMPT", "PASTE_ATTEMPT", "PRINTSCREEN_ATTEMPT", "PRINT_ATTEMPT", "ALT_TAB":
   482→                return "medium"
   483→        case "INACTIVITY":
   484→                return "low"
   485→        }
   486→        return "info"
   487→}

   564→                // Compter les screenshots (SCREEN_CAPTURE + PRINTSCREEN_ATTEMPT) dans logEvents.

   580→                        ActiveSessions:     active,
   581→                        SessionsWithAlerts: withAlerts,
   582→                        TotalAlerts:        totalAlerts,
   583→                        TotalPenalite:      totalPen,
   584→                        FlaggedSessions:    flagged,

   690→// ──────────────────────────────────────────────────────────────────────────
   691→// POST /api/surveillance/{id}/flag — signale une session (crée une Alerte FRAUDE)
   692→//
   693→// SURVEILLANCE-FIX-2 S2 :
   694→//   - Route manquante implémentée.
   695→//   - Body optionnel : { reason?: string }
   696→//   - Vérifie ownership (enseignant = propriétaire de l'épreuve).
   697→//   - 409 si déjà signalée (EXISTS Alerte FRAUDE).
   698→//   - Crée une ligne dans "Alerte" (titre, description, severity=WARNING,
   699→//     type=FRAUDE, epreuveId, userId=etudiantId).
   700→// ──────────────────────────────────────────────────────────────────────────
   701→
   702→func (s *Server) surveillanceFlagSession(w http.ResponseWriter, r *http.Request) {
   703→        claims, ok := middleware.ClaimsFromContext(r.Context())
   704→        if !ok || claims.UserID == "" {
   705→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
   706→                return
   707→        }
   708→
   709→        sessionID := chi.URLParam(r, "id")
   710→        if sessionID == "" {
   711→                writeJSONError(w, http.StatusBadRequest, "session id required")
   712→                return
   713→        }
   714→
   715→        var body struct {
   716→                Reason string `json:"reason"`
   717→        }
   718→        _ = json.NewDecoder(r.Body).Decode(&body)
   719→
   720→        var alerteID string
   721→        txErr := appdb.WithTx(r.Context(), s.dbPool, claims, func(tx pgx.Tx) error {
   722→                // 1. Récupérer la session + vérifier ownership (enseignant = propriétaire épreuve).
   723→                var epreuveID, etudiantID, epreuveTitre, etudiantNom string
   724→                var alertes int
   725→                var penalite float64
   726→                err := tx.QueryRow(r.Context(), `
   727→                        SELECT s."epreuveId", s."etudiantId",
   728→                               COALESCE(e."titre", 'Épreuve'), COALESCE(u."name", 'Étudiant'),
   729→                               COALESCE(s."alertes", 0), COALESCE(s."penalite", 0)
   730→                        FROM "SessionPassation" s
   731→                        LEFT JOIN "Epreuve" e ON e."id" = s."epreuveId"
   732→                        LEFT JOIN "User" u ON u."id" = s."etudiantId"
   733→                        WHERE s."id" = $1
   734→                `, sessionID).Scan(&epreuveID, &etudiantID, &epreuveTitre, &etudiantNom, &alertes, &penalite)
   735→                if err != nil {
   736→                        return &domain.NotFoundError{Entity: "SessionPassation", ID: sessionID}
   737→                }
   738→
   739→                // Ownership : ENSEIGNANT doit être propriétaire de l'épreuve.
   740→                if claims.Role == "ENSEIGNANT" {
   741→                        var ownerID string
   742→                        err = tx.QueryRow(r.Context(), `SELECT "enseignantId" FROM "Epreuve" WHERE "id" = $1`, epreuveID).Scan(&ownerID)
   743→                        if err != nil {
   744→                                return &domain.NotFoundError{Entity: "Epreuve", ID: epreuveID}
   745→                        }
   746→                        if ownerID != claims.UserID {
   747→                                return &domain.UnauthorizedError{Message: "accès refusé à cette session"}
   748→                        }
   749→                }
   750→
   751→                // 2. Vérifier si déjà signalée (EXISTS Alerte FRAUDE).
   752→                var alreadyFlagged bool
   753→                err = tx.QueryRow(r.Context(), `
   754→                        SELECT EXISTS(
   755→                          SELECT 1 FROM "Alerte"
   756→                          WHERE "epreuveId" = $1 AND "userId" = $2 AND "type" = 'FRAUDE'
   757→                        )
   758→                `, epreuveID, etudiantID).Scan(&alreadyFlagged)
   759→                if err != nil {
   760→                        return fmt.Errorf("check existing alerte: %w", err)
   761→                }
   762→                if alreadyFlagged {
   763→                        return &domain.ConflictError{Message: "Cette session a déjà été signalée."}
   764→                }
   765→
   766→                // 3. Construire titre + description.
   767→                shortID := sessionID
   768→                if len(shortID) > 8 {
   769→                        shortID = shortID[:8]
   770→                }
   771→                titre := fmt.Sprintf("Session signalée — %s (%s)", etudiantNom, epreuveTitre)
   772→                description := fmt.Sprintf("Session %s signalée par %s. %d alerte(s), pénalité %.2f.",
   773→                        shortID, claims.UserID, alertes, penalite)
   774→                if body.Reason != "" {
   775→                        description += " Motif : " + body.Reason
   776→                }
   777→
   778→                // 4. Insérer l'alerte.
   779→                err = tx.QueryRow(r.Context(), `
   780→                        INSERT INTO "Alerte" ("titre", "description", "severity", "type", "epreuveId", "userId", "lue", "resolu", "createdAt")
   781→                        VALUES ($1, $2, 'WARNING', 'FRAUDE', $3, $4, false, false, now())
   782→                        RETURNING "id"
   783→                `, titre, description, epreuveID, etudiantID).Scan(&alerteID)
   784→                if err != nil {
   785→                        return fmt.Errorf("insert alerte: %w", err)
   786→                }
   787→                return nil
   788→        })
   789→
   790→        if txErr != nil {
   791→                middleware.MapDomainError(w, txErr)
   792→                return
   793→        }
   794→
   795→        // SECT-ALERTES-FIX-1 P3 : notifier le RESPONSABLE de l'établissement + l'enseignant
   796→        // propriétaire de l'épreuve qu'une alerte fraude a été créée. Non bloquant.
   797→        if s.notifDispatcher != nil {
   798→                // Récupérer l'épreuve (ID + titre + enseignant) + le responsable de l'étab
   799→                var epreuveIDEnv, epreuveTitreEnv, enseignantID, responsableID string
   800→                _ = s.dbPool.QueryRow(r.Context(), `
   801→                        SELECT e."id", e."titre", e."enseignantId"
   802→                        FROM "SessionPassation" s JOIN "Epreuve" e ON e."id" = s."epreuveId"
   803→                        WHERE s."id" = $1`, sessionID).Scan(&epreuveIDEnv, &epreuveTitreEnv, &enseignantID)
   804→                if epreuveIDEnv != "" {
   805→                        _ = s.dbPool.QueryRow(r.Context(), `
   806→                                SELECT u.id FROM "User" u
   807→                                WHERE u.role = 'RESPONSABLE' AND u."etablissementId" = (
   808→                                        SELECT f."etablissementId" FROM "Epreuve" e JOIN "Filiere" f ON f.id = e."filiereId"
   809→                                        WHERE e.id = $1
   810→                                ) AND u."deletedAt" IS NULL LIMIT 1`, epreuveIDEnv).Scan(&responsableID)
   811→                }
   812→
   813→                notifEvent := notification.Event{
   814→                        Type:        "ALERTE_FRAUDE",
   815→                        Titre:       "Alerte fraude — session signalée ⚠️",
   816→                        Message:     fmt.Sprintf("Session signalée — %s", epreuveTitreEnv),
   817→                        Categorie:   "evaluation",
   818→                        Priorite:    "warning",
   819→                        ActionURL:   "/alertes",
   820→                        ActionLabel: "Voir l'alerte",
   821→                        Icone:       "ShieldAlert",
   822→                }
   823→
   824→                if responsableID != "" {
   825→                        notifEvent.UserID = responsableID
   826→                        s.notifDispatcher.Dispatch(r.Context(), notifEvent)
   827→                }
   828→                if enseignantID != "" && enseignantID != responsableID {
   829→                        notifEvent.UserID = enseignantID
   830→                        s.notifDispatcher.Dispatch(r.Context(), notifEvent)
   831→                }
   832→        }
   833→
   834→        w.Header().Set("Content-Type", "application/json")
   835→        w.WriteHeader(http.StatusCreated)
   836→        json.NewEncoder(w).Encode(map[string]any{
   837→                "alerte": map[string]string{
   838→                        "id":       alerteID,
   839→                        "titre":    "Session signalée",
   840→                        "severity": "WARNING",
   841→                        "type":     "FRAUDE",
   842→                },
   843→                "message": "Session signalée — alerte fraude créée",
   844→        })
   845→}

   848→// SSE-STREAM-1 : GET /api/surveillance/stream — Server-Sent Events
   849→// ──────────────────────────────────────────────────────────────────────────
   850→
   851→func (s *Server) surveillanceStream(w http.ResponseWriter, r *http.Request) {
   852→        claims, ok := middleware.ClaimsFromContext(r.Context())
   853→        if !ok || claims.UserID == "" {
   854→                writeJSONError(w, http.StatusUnauthorized, "authentication required")
   855→                return
   856→        }
   857→
   858→        enseignantID := r.URL.Query().Get("enseignantId")
   859→        if enseignantID == "" && claims.Role == "ENSEIGNANT" {
   860→                enseignantID = claims.UserID
   861→        }
   862→
   863→        w.Header().Set("Content-Type", "text/event-stream")
   864→        w.Header().Set("Cache-Control", "no-cache")
   865→        w.Header().Set("Connection", "keep-alive")
   866→        w.Header().Set("X-Accel-Buffering", "no")
   867→
   868→        flusher, _ := w.(http.Flusher)
   869→        flushIfNeeded := func() {
   870→                if flusher != nil {
   871→                        flusher.Flush()
   872→                }
   873→        }
   874→
   875→        stats := s.fetchSurveillanceStats(r, enseignantID)
   876→        if statsJSON, err := json.Marshal(stats); err == nil {
   877→                fmt.Fprintf(w, "data: %s\n\n", statsJSON)
   878→                flushIfNeeded()
   879→        }
   880→
   881→        ticker := time.NewTicker(10 * time.Second)
   882→        heartbeat := time.NewTicker(15 * time.Second)
   883→        defer ticker.Stop()
   884→        defer heartbeat.Stop()
   885→
   886→        for {
   887→                select {
   888→                case <-r.Context().Done():
   889→                        return
   890→                case <-ticker.C:
   891→                        stats := s.fetchSurveillanceStats(r, enseignantID)
   892→                        if statsJSON, err := json.Marshal(stats); err == nil {
   893→                                fmt.Fprintf(w, "data: %s\n\n", statsJSON)
   894→                                flushIfNeeded()
   895→                        }
   896→                case <-heartbeat.C:
   897→                        fmt.Fprintf(w, ": heartbeat\n\n")
   898→                        flushIfNeeded()
   899→                }
   900→        }
   901→}
