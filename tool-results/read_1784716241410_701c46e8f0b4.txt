    76→interface AccessRecord {
    77→  id: string
    78→  adminId: string
    79→  etablissementId: string
    80→  motif: string
    81→  statut: 'EN_ATTENTE' | 'APPROUVE' | 'REFUSE' | 'EXPIRE'
    82→  dateDebut: string | null
    83→  dateFin: string | null
    84→  commentaire: string | null
    85→  approuvePar: string | null
    86→  createdAt: string
    87→  // BUGFIX (ADMIN-AUDIT-4) : admin + etablissement optionnels (l'API peut
    88→  // ne pas inclure les relations). Optional chaining + fallback partout.
    89→  admin?: { id: string; name: string; email: string }
    90→  etablissement?: { id: string; nom: string; ville: string | null; actif: boolean }
    91→}

    93→interface AuthorizedEtablissement {
    94→  id: string
    95→  nom: string
    96→  type: string | null
    97→  ville: string | null
    98→  pays: string | null
    99→  email: string | null
   100→  telephone: string | null
   101→  actif: boolean
   102→  // BUGFIX (ADMIN-AUDIT-4b) : access optionnel (l'API peut ne pas l'inclure).
   103→  // Optional chaining partout pour éviter le crash etab.access.dateFin.
   104→  access?: {
   105→    id: string
   106→    motif: string
   107→    dateDebut: string | null

   125→function formatDate(dateStr: string | null) {
   126→  if (!dateStr) return '—'
   127→  return new Date(dateStr).toLocaleDateString('fr-FR', {
   128→    day: '2-digit',
   129→    month: '2-digit',
   130→    year: 'numeric',
   131→  })
   132→}
   133→
   134→function getStatutBadge(statut: string) {
   135→  switch (statut) {
   136→    case 'EN_ATTENTE':
   137→      return (
   138→        <Badge className="bg-warning/10 text-warning border-warning/30">
   139→          <Clock className="h-3 w-3 mr-1" />
   140→          En attente
   141→        </Badge>
   142→      )
   143→    case 'APPROUVE':
   144→      return (
   145→        <Badge className="bg-success/10 text-success-text border-success/30">
   146→          <CheckCircle2 className="h-3 w-3 mr-1" />
   147→          Approuvé
   148→        </Badge>
   149→      )
   150→    case 'REFUSE':
   151→      return (
   152→        <Badge className="bg-destructive/10 text-destructive border-destructive/30">
   153→          <X className="h-3 w-3 mr-1" />
   154→          Refusé
   155→        </Badge>
   156→      )
   157→    case 'EXPIRE':
   158→      return (
   159→        <Badge className="bg-gray-100 text-gray-600 border-gray-200 dark:bg-gray-800 dark:text-gray-400 dark:border-gray-700">
   160→          <AlertCircle className="h-3 w-3 mr-1" />
   161→          Expiré
   162→        </Badge>
   163→      )
   164→    default:
   165→      return <Badge variant="outline">{statut}</Badge>
   166→  }
   167→}
   168→
   169→function getMotifLabel(motif: string) {
   170→  switch (motif) {
   171→    case 'Audit':
   172→      return 'Audit'
   173→    case 'Support technique':
   174→      return 'Support technique'
   175→    case 'Inspection':
   176→      return 'Inspection'
   177→    case 'Urgence':
   178→      return 'Urgence'
   179→    default:
   180→      return motif
   181→  }
   182→}

   186→export function AccesEtablissementsPage() {
   187→  const { user } = useAuthStore()
   188→  const queryClient = useQueryClient()
   189→  const router = useRouter()
   190→
   191→  // ─── Data state (BUGFIX QUERY-MIGRATION-GROUP-A : TanStack Query) ───
   192→  // Le cache survit au démontage → 0 refetch au retour, 0 skeleton, navigation
   193→  // instantanée. Les 3 ressources sont indépendantes → 3 useQuery séparés.
   194→  const accessQuery = useQuery<{ accessRecords: AccessRecord[] }>({
   195→    queryKey: ['etablissement-access', user?.id],
   196→    queryFn: async () => {
   197→      const res = await fetch(`/api/etablissement-access?adminId=${user!.id}`)
   198→      if (!res.ok) throw new Error('Failed to fetch access records')
   199→      return res.json()
   200→    },
   201→    enabled: !!user?.id,
   202→    staleTime: 60 * 1000,
   203→    refetchOnWindowFocus: false,
   204→  })
   205→
   206→  const authorizedQuery = useQuery<{ etablissements: AuthorizedEtablissement[] }>({
   207→    queryKey: ['authorized-etablissements', user?.id],
   208→    queryFn: async () => {
   209→      const res = await fetch(`/api/etablissement-access/authorized-etablissements?adminId=${user!.id}`)
   210→      if (!res.ok) throw new Error('Failed to fetch authorized etablissements')
   211→      return res.json()
   212→    },
   213→    enabled: !!user?.id,
   214→    staleTime: 60 * 1000,
   215→    refetchOnWindowFocus: false,
   216→  })
   217→
   218→  const etablissementsQuery = useQuery<{ etablissements: EtablissementOption[] }>({
   219→    queryKey: ['etablissements'],
   220→    queryFn: async () => {
   221→      const res = await fetch('/api/etablissements')
   222→      if (!res.ok) throw new Error('Failed to fetch etablissements')
   223→      return res.json()
   224→    },
   225→    staleTime: 60 * 1000,
   226→    refetchOnWindowFocus: false,
   227→  })
   228→
   229→  const accessRecords = accessQuery.data?.accessRecords ?? []
   230→  const authorizedEtablissements = authorizedQuery.data?.etablissements ?? []
   231→  const etablissements = useMemo(
   232→    () =>
   233→      (etablissementsQuery.data?.etablissements ?? []).map((e: EtablissementOption) => ({
   234→        id: e.id,
   235→        nom: e.nom,
   236→        ville: e.ville,
   237→        type: e.type,
   238→        actif: e.actif,
   239→        _count: e._count ?? { filieres: 0, users: 0 },
   240→      })),
   241→    [etablissementsQuery.data],
   242→  )
   243→  const isLoading =
   244→    accessQuery.isLoading || authorizedQuery.isLoading || etablissementsQuery.isLoading
   245→
   246→  // BUGFIX B-9 : toasts one-shot sur erreur de fetch. Les useEffect ne se
   247→  // redéclenchent que sur transition false→true de isError (tableau de dep
   248→  // à un seul booléen), donc pas de spam de toasts au re-render.
   249→  useEffect(() => {
   250→    if (accessQuery.isError)
   251→      toast.error('Erreur', { description: 'Impossible de charger vos demandes d\'accès.' })
   252→  }, [accessQuery.isError])
   253→  useEffect(() => {
   254→    if (authorizedQuery.isError)
   255→      toast.error('Erreur', { description: 'Impossible de charger vos établissements autorisés.' })
   256→  }, [authorizedQuery.isError])
   257→  useEffect(() => {
   258→    if (etablissementsQuery.isError)
   259→      toast.error('Erreur', { description: 'Impossible de charger la liste des établissements.' })
   260→  }, [etablissementsQuery.isError])
   261→
   262→  // Helper pour invalider le cache après mutation (create/delete).
   263→  const refreshData = async () => {
   264→    await Promise.all([
   265→      queryClient.invalidateQueries({ queryKey: ['etablissement-access', user?.id] }),
   266→      queryClient.invalidateQueries({ queryKey: ['authorized-etablissements', user?.id] }),
   267→      queryClient.invalidateQueries({ queryKey: ['etablissements'] }),
   268→    ])
   269→  }
   270→
   271→  // ─── Dialog state ───
   272→  const [cancelTarget, setCancelTarget] = useState<AccessRecord | null>(null)
   273→  // ACCESS-WORKFLOW-UI : cible pour la révocation d'un accès APPROUVE par l'ADMIN.
   274→  // Le bouton "Révoquer" apparaît à côté de "Voir l'établissement" pour les
   275→  // enregistrements APPROUVE. La mutation PATCH envoie {statut: REFUSE,
   276→  // commentaire: "Accès révoqué par l'admin"} → le backend Update() accepte ce
   277→  // statut (validStatuts contient AccessRefuse) sans vérifier l'ownership côté
   278→  // ADMIN (le usecase ne fait le check EtablissementID que pour RESPONSABLE).
   279→  const [revokeTarget, setRevokeTarget] = useState<AccessRecord | null>(null)
   280→  const [isSubmitting, setIsSubmitting] = useState(false)
   281→  // BUGFIX B-7 : protection anti double-clic sur les boutons "Annuler" et
   282→  // "Révoquer" des AlertDialogs. Désactive l'AlertDialogAction pendant la
   283→  // mutation pour éviter les requêtes dupliquées (et donc les 404/409 sur le
   284→  // DELETE/PATCH d'un ID déjà traité).
   285→  const [cancelling, setCancelling] = useState(false)
   286→  const [revoking, setRevoking] = useState(false)
   287→  const [activeTab, setActiveTab] = useState('mes-autorisations')
   288→  // ASSISTANCE-MODE-FRONTEND : ID de l'établissement dont l'activation du mode
   289→  // assistance est en cours. Désactive le bouton correspondant (Loader2) et
   290→  // empêche les double-clics. Vide = aucune activation en cours.
   291→  const [assistanceLoadingId, setAssistanceLoadingId] = useState<string | null>(null)
   292→
   293→  // ─── Form state ───
   294→  const [formEtablissementId, setFormEtablissementId] = useState('')
   295→  const [formMotif, setFormMotif] = useState('')
   296→  const [formDateDebut, setFormDateDebut] = useState('')
   297→  const [formDateFin, setFormDateFin] = useState('')
   298→  const [formCommentaire, setFormCommentaire] = useState('')
   299→
   300→  // ─── Stats ───
   301→  const approuveCount = accessRecords.filter((r) => r.statut === 'APPROUVE').length
   302→  const enAttenteCount = accessRecords.filter((r) => r.statut === 'EN_ATTENTE').length
   303→  const expireCount = accessRecords.filter((r) => r.statut === 'EXPIRE').length
   304→  const activeEtablissementsCount = etablissements.filter((e) => e.actif).length
   305→
   306→  // ─── Available establishments for the form ───
   307→  // Only show establishments where admin doesn't already have APPROUVE or EN_ATTENTE access
   308→  const unavailableEtabIds = new Set(
   309→    accessRecords
   310→      .filter((r) => r.statut === 'APPROUVE' || r.statut === 'EN_ATTENTE')
   311→      .map((r) => r.etablissementId)
   312→  )
   313→  const availableEtablissements = etablissements.filter(
   314→    (e) => e.actif && !unavailableEtabIds.has(e.id)
   315→  )
   316→
   317→  // ─── Cancel access request ───
   318→  const handleCancelRequest = async () => {
   319→    if (!cancelTarget) return
   320→    setCancelling(true)
   321→    try {
   322→      const res = await fetch(`/api/etablissement-access/${cancelTarget.id}`, {
   323→        method: 'DELETE',
   324→      })
   325→      if (!res.ok) {
   326→        const err = await res.json().catch(() => ({}))
   327→        throw new Error(err.error || 'Erreur lors de l\'annulation')
   328→      }
   329→      toast.success('Demande annulée', {
   330→        description: `La demande d'accès à ${cancelTarget.etablissement?.nom ?? 'cet établissement'} a été annulée.`,
   331→      })
   332→      setCancelTarget(null)
   333→      await refreshData()
   334→    } catch (err) {
   335→      toast.error('Erreur', {
   336→        description: err instanceof Error ? err.message : 'Impossible d\'annuler la demande.',
   337→      })
   338→    } finally {
   339→      setCancelling(false)
   340→    }
   341→  }
   342→
   343→  // ─── Revoke access (ACCESS-WORKFLOW-UI) ───
   344→  // Permet à l'ADMIN de révoquer un accès qu'il avait précédemment obtenu.
   345→  // Le backend Update() accepte statut=REFUSE pour un accès APPROUVE et
   346→  // enregistre le commentaire. L'auto-révocation est aussi gérée côté backend
   347→  // via admin_has_etablissement_access() qui vérifie dateFin >= now().
   348→  // Côté frontend on propose une révocation manuelle anticipative.
   349→  const handleRevokeAccess = async () => {
   350→    if (!revokeTarget) return
   351→    setRevoking(true)
   352→    try {
   353→      const res = await fetch(`/api/etablissement-access/${revokeTarget.id}`, {
   354→        method: 'PATCH',
   355→        headers: { 'Content-Type': 'application/json' },
   356→        body: JSON.stringify({
   357→          statut: 'REFUSE',
   358→          commentaire: 'Accès révoqué par l\'admin',
   359→        }),
   360→      })
   361→      if (!res.ok) {
   362→        const err = await res.json().catch(() => ({}))
   363→        throw new Error(err.error || 'Erreur lors de la révocation')
   364→      }
   365→      toast.success('Accès révoqué', {
   366→        description: `L'accès à ${revokeTarget.etablissement?.nom ?? 'cet établissement'} a été révoqué.`,
   367→      })
   368→      setRevokeTarget(null)
   369→      await refreshData()
   370→    } catch (err) {
   371→      toast.error('Erreur', {
   372→        description: err instanceof Error ? err.message : 'Impossible de révoquer l\'accès.',
   373→      })
   374→    } finally {
   375→      setRevoking(false)
   376→    }
   377→  }
   378→
   379→  // ─── Assistance mode (ASSISTANCE-MODE-FRONTEND) ───
   380→  // Active le "mode assistance" pour un établissement APPROUVE : le backend
   381→  // /api/auth/assistance-mode émet de nouveaux tokens JWT avec
   382→  // etablissementId positionné, et renvoie le user mis à jour. On stocke les
   383→  // nouveaux tokens via le shim Next.js (cookies httpOnly), on met à jour
   384→  // l'auth store (l'ADMIN "devient" RESPONSABLE pour la nav), puis on
   385→  // redirige vers /dashboard (vue responsable).
   386→  const handleAssistanceMode = async (record: AccessRecord) => {
   387→    if (!record.etablissementId) return
   388→    setAssistanceLoadingId(record.etablissementId)
   389→    try {
   390→      const res = await fetch('/api/go-auth/assistance-mode', {
   391→        method: 'POST',
   392→        headers: { 'Content-Type': 'application/json' },
   393→        body: JSON.stringify({ etablissementId: record.etablissementId }),
   394→      })
   395→      const data = await res.json().catch(() => ({}))
   396→      if (!res.ok || !data?.user) {
   397→        throw new Error(data?.error || 'Impossible d\'activer le mode assistance')
   398→      }
   399→      // Normalisation du user (champs optionnels → null pour éviter les
   400→      // "undefined" dans la sidebar / header).
   401→      useAuthStore.setState({
   402→        user: {
   403→          id: data.user.id,
   404→          email: data.user.email ?? '',
   405→          name: data.user.name ?? '',
   406→          role: data.user.role,
   407→          etablissementId: data.user.etablissementId ?? null,
   408→          filiereId: data.user.filiereId ?? null,
   409→          etablissement: data.user.etablissement ?? null,
   410→          filiere: data.user.filiere ?? null,
   411→          image: data.user.image ?? null,
   412→          actif: data.user.actif,
   413→          matricule: data.user.matricule ?? null,
   414→          mustChangePwd: data.user.mustChangePwd,
   415→          derniereConnexion: data.user.derniereConnexion ?? null,
   416→        },
   417→      })
   418→      toast.success('Mode assistance activé', {
   419→        description: `Vous accédez maintenant aux données de ${record.etablissement?.nom ?? 'cet établissement'}.`,
   420→      })
   421→      router.push('/dashboard')
   422→    } catch (err) {
   423→      toast.error('Erreur', {
   424→        description: err instanceof Error ? err.message : 'Une erreur est survenue.',
   425→      })
   426→    } finally {
   427→      setAssistanceLoadingId(null)
   428→    }
   429→  }
   430→
   431→  // ─── Submit access request ───
   432→  const handleSubmitRequest = async () => {
   433→    if (!formEtablissementId || !formMotif || !user?.id) {
   434→      toast.error('Champs manquants', {
   435→        description: 'L\'établissement et le motif sont obligatoires.',
   436→      })
   437→      return
   438→    }
   439→
   440→    setIsSubmitting(true)
   441→    try {
   442→      // ACCES-ETABLISSEMENTS-FIX : convertir les dates "YYYY-MM-DD" (input type=date)
   443→      // en RFC3339 "YYYY-MM-DDT00:00:00Z" car le backend Go décode *time.Time qui
   444→      // attend le format RFC3339. Sans conversion, json.Decoder échoue → 400 "JSON invalide".
   445→      const toRFC3339 = (d: string): string | null => {
   446→        if (!d) return null
   447→        return `${d}T00:00:00Z`
   448→      }
   449→      const res = await fetch('/api/etablissement-access', {
   450→        method: 'POST',
   451→        headers: { 'Content-Type': 'application/json' },
   452→        body: JSON.stringify({
   453→          adminId: user.id,
   454→          etablissementId: formEtablissementId,
   455→          motif: formMotif,
   456→          dateDebut: toRFC3339(formDateDebut),
   457→          dateFin: toRFC3339(formDateFin),
   458→          commentaire: formCommentaire || null,
   459→        }),
   460→      })
   461→      if (!res.ok) {
   462→        const err = await res.json().catch(() => ({}))
   463→        // ACCES-ETABLISSEMENTS-FIX : si 409 (demande déjà existante), afficher un
   464→        // message spécifique avec CTA vers l'onglet Mes autorisations pour annuler.
   465→        if (res.status === 409) {
   466→          toast.error('Demande déjà existante', {
   467→            description: 'Vous avez déjà une demande en attente ou approuvée pour cet établissement. Annulez-la d\'abord dans l\'onglet « Mes autorisations ».',
   468→            action: {
   469→              label: 'Voir mes demandes',
   470→              onClick: () => setActiveTab('mes-autorisations'),
   471→            },
   472→          })
   473→          setIsSubmitting(false)
   474→          return
   475→        }
   476→        throw new Error(err.error || 'Erreur lors de la création')
   477→      }
   478→      toast.success('Demande envoyée', {
   479→        description: 'Votre demande d\'accès a été soumise avec succès.',
   480→      })
   481→      // Reset form
   482→      setFormEtablissementId('')
   483→      setFormMotif('')
   484→      setFormDateDebut('')
   485→      setFormDateFin('')
   486→      setFormCommentaire('')
   487→      setActiveTab('mes-autorisations')
   488→      await refreshData()
   489→    } catch (err) {
   490→      toast.error('Erreur', {
   491→        description: err instanceof Error ? err.message : 'Une erreur est survenue.',
   492→      })
   493→    } finally {
   494→      setIsSubmitting(false)
   495→    }
   496→  }
   497→
   498→  // ─── Renew / Relancer (create new request for expired/refused) ───
   499→  const handleRenew = (record: AccessRecord) => {
   500→    // Pre-fill the form with the same etablissement and motif
   501→    setFormEtablissementId(record.etablissementId)
   502→    setFormMotif(record.motif)
   503→    setFormDateDebut('')
   504→    setFormDateFin('')
   505→    setFormCommentaire('')
   506→    setActiveTab('demander-acces')
   507→    toast.info('Renouvellement', {
   508→      description: `Formulaire pré-rempli pour ${record.etablissement?.nom ?? 'cet établissement'}.`,
   509→    })
   510→  }
   511→
   512→  return (
   513→    <div className="space-y-6">
   514→      {/* ─── Header ─── */}
   515→      <div className="ds-kente-pattern -mx-4 -mt-4 rounded-lg px-4 py-4 sm:-mx-6 sm:px-6">
   516→        <h1 className="text-2xl font-bold tracking-tight md:text-3xl flex items-center gap-2 font-display">
   517→          <KeyRound className="h-7 w-7 text-success-text" />
   518→          Accès aux Établissements
   519→        </h1>
   520→        <p className="mt-1 text-sm text-muted-foreground">
   521→          Gérez vos autorisations d&apos;accès aux données des établissements
   522→        </p>
   523→      </div>
   524→
   525→      {/* ─── Notice box ─── */}
