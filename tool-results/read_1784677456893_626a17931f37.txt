    60→function formatDateTime(date: string | Date | null): string {
    61→  if (!date) return '—'
    62→  const d = typeof date === 'string' ? new Date(date) : date
    63→  return d.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit', second: '2-digit' })
    64→}
    65→
    66→function formatRelativeDate(dateStr: string): string {
    67→  const date = new Date(dateStr)
    68→  const now = new Date()
    69→  const diffMs = now.getTime() - date.getTime()
    70→  const diffMin = Math.floor(diffMs / 60000)
    71→  const diffH = Math.floor(diffMin / 60)
    72→  const diffD = Math.floor(diffH / 24)
    73→  if (diffMin < 1) return "À l'instant"
    74→  if (diffMin < 60) return `Il y a ${diffMin} min`
    75→  if (diffH < 24) return `Il y a ${diffH}h`
    76→  if (diffD < 7) return `Il y a ${diffD}j`
    77→  return date.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit' })
    78→}
    79→
    80→function formatTime(date: string): string {
    81→  return new Date(date).toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit', second: '2-digit' })
    82→}

   146→function useDebounce<T>(value: T, delay = 350): T {
   147→  const [debounced, setDebounced] = useState(value)
   148→  useEffect(() => {
   149→    const t = setTimeout(() => setDebounced(value), delay)
   150→    return () => clearTimeout(t)
   151→  }, [value, delay])
   152→  return debounced
   153→}

   865→// ─── Captures Viewer (R2) ───
   866→function CapturesViewer({ captures, index, onClose, onIndexChange }: {
   867→  captures: SessionCapture[]; index: number; onClose: () => void; onIndexChange: (index: number) => void
   868→}) {
   869→  const current = captures[index]
   870→  if (!current) return null
   871→
   872→  const formatFileSize = (bytes?: number) => {
   873→    if (!bytes) return '—'
   874→    if (bytes < 1024) return `${bytes} o`
   875→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} Ko`
   876→    return `${(bytes / (1024 * 1024)).toFixed(1)} Mo`
   877→  }
   878→
   879→  return (
   880→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   881→      <DialogContent className="sm:max-w-4xl">
   882→        <DialogHeader>
   883→          <DialogTitle className="flex items-center justify-between gap-2">
   884→            <span>Capture #{current.captureIndex} — {index + 1} / {captures.length}</span>
   885→            <span className="text-xs font-normal text-muted-foreground">
   886→              {formatDateTime(current.createdAt)}
   887→              {current.fileSize ? ` • ${formatFileSize(current.fileSize)}` : ''}
   888→            </span>
   889→          </DialogTitle>
   890→        </DialogHeader>
   891→        <div className="relative flex items-center justify-center">
   892→          {current.url ? (
   893→            <img
   894→              src={current.url}
   895→              alt={`Capture #${current.captureIndex}`}
   896→              className="max-h-[60vh] w-auto rounded-lg border border-border/60"
   897→            />
   898→          ) : (
   899→            <div className="flex h-64 w-full flex-col items-center justify-center rounded-lg border border-border/60 bg-muted">
   900→              <ImageOff className="h-12 w-12 text-muted-foreground/40" />
   901→              <p className="mt-2 text-xs text-muted-foreground">Image non disponible (R2 non configuré)</p>
   902→            </div>
   903→          )}
   904→          {captures.length > 1 && (<>
   905→            <Button size="icon" variant="outline" onClick={() => onIndexChange((index - 1 + captures.length) % captures.length)} className="absolute left-2" aria-label="Capture précédente"><ChevronRight className="h-4 w-4 rotate-180" /></Button>
   906→            <Button size="icon" variant="outline" onClick={() => onIndexChange((index + 1) % captures.length)} className="absolute right-2" aria-label="Capture suivante"><ChevronRight className="h-4 w-4" /></Button>
   907→          </>)}
   908→        </div>
   909→        {/* Thumbnails strip */}
   910→        {captures.length > 1 && (
   911→          <div className="mt-3 flex gap-1.5 overflow-x-auto pb-1 scrollbar-thin">
   912→            {captures.map((c, i) => (
   913→              <button
   914→                key={c.id}
   915→                onClick={() => onIndexChange(i)}
   916→                className={`shrink-0 rounded-md border-2 transition-all ${
   917→                  i === index ? 'border-primary ring-1 ring-primary/30' : 'border-border/60 opacity-60 hover:opacity-100'
   918→                }`}
   919→                aria-label={`Capture #${c.captureIndex}`}
   920→              >
   921→                {c.url ? (
   922→                  <img src={c.url} alt={`Miniature #${c.captureIndex}`} className="h-12 w-20 rounded object-cover" />
   923→                ) : (
   924→                  <div className="flex h-12 w-20 items-center justify-center rounded bg-muted">
   925→                    <Camera className="h-4 w-4 text-muted-foreground/40" />
   926→                  </div>
   927→                )}
   928→              </button>
   929→            ))}
   930→          </div>
   931→        )}
   932→      </DialogContent>
   933→    </Dialog>
