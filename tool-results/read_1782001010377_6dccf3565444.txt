    55→/* ─── Magnetic Button ─── */
    56→function MagneticButton({
    57→  children,
    58→  className = '',
    59→  ...props
    60→}: React.ComponentProps<typeof Button> & { children: React.ReactNode }) {
    61→  const ref = useRef<HTMLButtonElement>(null)
    62→  const xTo = useRef<gsap.QuickToFunc | null>(null)
    63→  const yTo = useRef<gsap.QuickToFunc | null>(null)
    64→
    65→  useEffect(() => {
    66→    if (!ref.current) return
    67→    xTo.current = gsap.quickTo(ref.current, 'x', { duration: 0.3, ease: 'power2.out' })
    68→    yTo.current = gsap.quickTo(ref.current, 'y', { duration: 0.3, ease: 'power2.out' })
    69→    return () => {
    70→      xTo.current = null
    71→      yTo.current = null
    72→    }
    73→  }, [])
    74→
    75→  const handleMouseMove = (e: React.MouseEvent) => {
    76→    if (!ref.current || !xTo.current || !yTo.current) return
    77→    const rect = ref.current.getBoundingClientRect()
    78→    const dx = e.clientX - rect.left - rect.width / 2
    79→    const dy = e.clientY - rect.top - rect.height / 2
    80→    xTo.current(dx * 0.3)
    81→    yTo.current(dy * 0.3)
    82→  }
    83→
    84→  const handleMouseLeave = () => {
    85→    xTo.current?.(0)
    86→    yTo.current?.(0)
    87→  }
    88→
    89→  return (
    90→    <Button
    91→      ref={ref}
    92→      className={className}
    93→      onMouseMove={handleMouseMove}
    94→      onMouseLeave={handleMouseLeave}
    95→      {...props}
    96→    >
    97→      {children}
    98→    </Button>
    99→  )
   100→}

   102→/* ─── Animated Counter ─── */
   103→function AnimatedCounter({ target, suffix = '', prefix = '' }: { target: number; suffix?: string; prefix?: string }) {
   104→  const ref = useRef<HTMLSpanElement>(null)
   105→  const counted = useRef(false)
   106→
   107→  useEffect(() => {
   108→    if (!ref.current) return
   109→    const el = ref.current
   110→    const trigger = ScrollTrigger.create({
   111→      trigger: el,
   112→      start: 'top 85%',
   113→      once: true,
   114→      onEnter: () => {
   115→        if (counted.current) return
   116→        counted.current = true
   117→        const obj = { val: 0 }
   118→        gsap.to(obj, {
   119→          val: target,
   120→          duration: 2,
   121→          ease: 'power2.out',
   122→          onUpdate: () => {
   123→            el.textContent = prefix + Math.round(obj.val).toLocaleString() + suffix
   124→          },
   125→        })
   126→      },
   127→    })
   128→    return () => trigger.kill()
   129→  }, [target, suffix, prefix])
   130→
   131→  return <span ref={ref}>{prefix}0{suffix}</span>
   132→}

   134→/* ─── Typing Effect ─── */
   135→function TypingText({ texts, speed = 80, pause = 2000 }: { texts: string[]; speed?: number; pause?: number }) {
   136→  const [display, setDisplay] = useState('')
   137→  const [textIdx, setTextIdx] = useState(0)
   138→  const [charIdx, setCharIdx] = useState(0)
   139→  const [deleting, setDeleting] = useState(false)
   140→
   141→  useEffect(() => {
   142→    const current = texts[textIdx]
   143→    if (!deleting && charIdx < current.length) {
   144→      const t = setTimeout(() => {
   145→        setDisplay(current.slice(0, charIdx + 1))
   146→        setCharIdx(charIdx + 1)
   147→      }, speed)
   148→      return () => clearTimeout(t)
   149→    }
   150→    if (!deleting && charIdx === current.length) {
   151→      const t = setTimeout(() => setDeleting(true), pause)
   152→      return () => clearTimeout(t)
   153→    }
   154→    if (deleting && charIdx > 0) {
   155→      const t = setTimeout(() => {
   156→        setDisplay(current.slice(0, charIdx - 1))
   157→        setCharIdx(charIdx - 1)
   158→      }, speed / 2)
   159→      return () => clearTimeout(t)
   160→    }
   161→    if (deleting && charIdx === 0) {
   162→      const t = setTimeout(() => {
   163→        setDeleting(false)
   164→        setTextIdx((prev) => (prev + 1) % texts.length)
   165→      }, speed)
   166→      return () => clearTimeout(t)
   167→    }
   168→  }, [charIdx, deleting, textIdx, texts, speed, pause])
   169→
   170→  return (
   171→    <span>
   172→      {display}
   173→      <span className="animate-pulse text-emerald-400">|</span>
   174→    </span>
   175→  )
   176→}
