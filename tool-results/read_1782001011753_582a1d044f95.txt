   253→   253→// ─── Typewriter Hook ───
   254→   254→function useTypewriter(words: string[], typingSpeed = 80, deletingSpeed = 40, pauseTime = 2000) {
   255→   255→  const [text, setText] = useState('')
   256→   256→  const [wordIndex, setWordIndex] = useState(0)
   257→   257→  const [isDeleting, setIsDeleting] = useState(false)
   258→   258→
   259→   259→  useEffect(() => {
   260→   260→    const currentWord = words[wordIndex]
   261→   261→
   262→   262→    const timeout = setTimeout(() => {
   263→   263→      if (!isDeleting) {
   264→   264→        setText(currentWord.substring(0, text.length + 1))
   265→   265→        if (text.length === currentWord.length) {
   266→   266→          setTimeout(() => setIsDeleting(true), pauseTime)
   267→   267→        }
   268→   268→      } else {
   269→   269→        setText(currentWord.substring(0, text.length - 1))
   270→   270→        if (text.length === 0) {
   271→   271→          setIsDeleting(false)
   272→   272→          setWordIndex((prev) => (prev + 1) % words.length)
   273→   273→        }
   274→   274→      }
   275→   275→    }, isDeleting ? deletingSpeed : typingSpeed)
   276→   276→
   277→   277→    return () => clearTimeout(timeout)
   278→   278→  }, [text, isDeleting, wordIndex, words, typingSpeed, deletingSpeed, pauseTime])
   279→   279→
   280→   280→  return text
   281→   281→}

   284→   284→function FeatureCard({ feature, index }: { feature: typeof features[0]; index: number }) {
   285→   285→  const cardRef = useRef<HTMLDivElement>(null)
   286→   286→  const iconRef = useRef<HTMLDivElement>(null)
   287→   287→  const statRef = useRef<HTMLSpanElement>(null)
   288→   288→  const [hovered, setHovered] = useState(false)
   289→   289→
   290→   290→  useEffect(() => {
   291→   291→    if (!cardRef.current) return
   292→   292→
   293→   293→    const card = cardRef.current
   294→   294→
   295→   295→    const handleMouseMove = (e: MouseEvent) => {
   296→   296→      const rect = card.getBoundingClientRect()
   297→   297→      const x = e.clientX - rect.left - rect.width / 2
   298→   298→      const y = e.clientY - rect.top - rect.height / 2
   299→   299→      const rotateX = (y / rect.height) * -12
   300→   300→      const rotateY = (x / rect.width) * 12
   301→   301→
   302→   302→      gsap.to(card, {
   303→   303→        rotateX,
   304→   304→        rotateY,
   305→   305→        duration: 0.3,
   306→   306→        ease: 'power2.out',
   307→   307→        transformPerspective: 600,
   308→   308→      })
   309→   309→    }
   310→   310→
   311→   311→    const handleMouseLeave = () => {
   312→   312→      gsap.to(card, {
   313→   313→        rotateX: 0,
   314→   314→        rotateY: 0,
   315→   315→        duration: 0.6,
   316→   316→        ease: 'elastic.out(1, 0.5)',
   317→   317→        transformPerspective: 600,
   318→   318→      })
   319→   319→      setHovered(false)
   320→   320→    }
   321→   321→
   322→   322→    const handleMouseEnter = () => {
   323→   323→      setHovered(true)
   324→   324→      if (iconRef.current) {
   325→   325→        gsap.to(iconRef.current, {
   326→   326→          scale: 1.15,
   327→   327→          rotation: 8,
   328→   328→          duration: 0.3,
   329→   329→          ease: 'back.out(2)',
   330→   330→        })

   334→   334→          { scale: 1 },
   335→   335→          { scale: 1.2, duration: 0.3, yoyo: true, repeat: 1, ease: 'power2.out' }
   336→   336→        )
   337→   337→      }
   338→   338→    }
   339→   339→
   340→   340→    card.addEventListener('mousemove', handleMouseMove)
   341→   341→    card.addEventListener('mouseleave', handleMouseLeave)
   342→   342→    card.addEventListener('mouseenter', handleMouseEnter)
   343→   343→
   344→   344→    return () => {
   345→   345→      card.removeEventListener('mousemove', handleMouseMove)
   346→   346→      card.removeEventListener('mouseleave', handleMouseLeave)
   347→   347→      card.removeEventListener('mouseenter', handleMouseEnter)
   348→   348→    }
   349→   349→  }, [index])
   350→   350→
   351→   351→  // Reset icon on unhover via GSAP
   352→   352→  useEffect(() => {
   353→   353→    if (!hovered && iconRef.current) {
   354→   354→      gsap.to(iconRef.current, {
   355→   355→        scale: 1,
   356→   356→        rotation: 0,
   357→   357→        duration: 0.4,
   358→   358→        ease: 'power2.out',
   359→   359→      })
   360→   360→    }
   361→   361→  }, [hovered])
   362→   362→
   363→   363→  return (
   364→   364→    <div
   365→   365→      ref={cardRef}
   366→   366→      className={`flex items-center gap-3 rounded-2xl px-4 py-3.5 border transition-colors duration-300 cursor-default ${
   367→   367→        hovered
   368→   368→          ? 'bg-white/20 backdrop-blur-md border-white/30 shadow-lg shadow-emerald-500/10'
