   131→function getScoreColor(score: number, maxScore: number = 20): string {
   132→  const halfMax = maxScore / 2
   133→  if (score >= halfMax) return '#10b981'
   134→  if (score >= halfMax * 0.8) return '#f59e0b'
   135→  return '#ef4444'
   136→}
   137→
   138→function getTauxColor(taux: number): string {
   139→  if (taux >= 70) return '#10b981'
   140→  if (taux >= 50) return '#f59e0b'
   141→  return '#ef4444'
   142→}

   144→function formatDateFR(dateStr: string): string {
   145→  try {
   146→    const date = new Date(dateStr)
   147→    return date.toLocaleDateString('fr-FR', {
   148→      day: '2-digit',
   149→      month: 'long',
   150→      year: 'numeric',
   151→    })
   152→  } catch {
   153→    return dateStr
   154→  }
   155→}
   156→
   157→function formatMonth(mois: string): string {
   158→  try {
   159→    const [year, month] = mois.split('-')
   160→    const date = new Date(parseInt(year), parseInt(month) - 1)
   161→    return date.toLocaleDateString('fr-FR', { month: 'short', year: '2-digit' })
   162→  } catch {
   163→    return mois
   164→  }
   165→}

   496→// ─── Main Component ───
   497→export function ResponsableDashboard() {
   498→  const user = useAuthStore((s) => s.user)
   499→  const name = user?.name ?? 'Responsable'
   500→
   501→  const [data, setData] = useState<StatsData | null>(null)
   502→  const [loading, setLoading] = useState(true)
   503→  const [error, setError] = useState(false)
   504→
   505→  const fetchStats = useCallback(async () => {
   506→    setLoading(true)
   507→    setError(false)
   508→    try {
   509→      const controller = new AbortController()
   510→      const timeoutId = setTimeout(() => controller.abort(), 15000)
   511→
   512→      const res = await fetch(`/api/stats/responsable`, {
   513→        signal: controller.signal,
   514→      })
   515→      clearTimeout(timeoutId)
   516→
   517→      if (!res.ok) throw new Error('Erreur réseau')
   518→      const json: StatsData = await res.json()
   519→      setData(json)
   520→    } catch (err) {
   521→      console.error('Dashboard fetch error:', err)
   522→      setError(true)
   523→      if (data === null) {
   524→        toast.error('Impossible de charger les statistiques du tableau de bord.', {
   525→          action: { label: 'Réessayer', onClick: () => fetchStats() },
   526→        })
   527→      }
   528→    } finally {
   529→      setLoading(false)
   530→    }
   531→  }, [])
   532→
   533→  useEffect(() => {
   534→    fetchStats()
   535→  }, [fetchStats])
   536→
   537→  if (loading && !data) {
   538→    return <DashboardSkeleton />
   539→  }
   540→
   541→  if (error && !data) {
   542→    return <DashboardError onRetry={fetchStats} />
   543→  }
   544→
   545→  if (!data) {
   546→    return <EmptyDashboard name={name} />
   547→  }
   548→
   549→  const hasNoActivity = data.nbEtudiants === 0 && data.nbEvaluations === 0
   550→
   551→  if (hasNoActivity) {
   552→    return <EmptyDashboard name={name} />
   553→  }
   554→
   555→  return (
   556→    <motion.div
   557→      className="space-y-6"
   558→      variants={containerVariants}
   559→      initial="hidden"
   560→      animate="visible"
   561→    >
   562→      <AnimatePresence>
   563→        <motion.h1 variants={itemVariants} className="text-2xl font-bold tracking-tight md:text-3xl">
   564→          Bonjour, {name} ! Vue stratégique de votre établissement.
   565→        </motion.h1>
   566→      </AnimatePresence>
   567→
   568→      {/* Quick stats bar */}
   569→      <motion.div variants={itemVariants} className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   570→        <Card className="p-4">
   571→          <div className="flex items-center gap-3">
   572→            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-amber-100 dark:bg-amber-900/40">
   573→              <GraduationCap className="h-5 w-5 text-amber-600" />
   574→            </div>
   575→            <div>
   576→              <p className="text-2xl font-bold">{data.nbEtudiants}</p>
   577→              <p className="text-xs text-muted-foreground">Étudiant{data.nbEtudiants !== 1 ? 's' : ''}</p>
   578→            </div>
   579→          </div>
   580→        </Card>
