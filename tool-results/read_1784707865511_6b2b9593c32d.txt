   168→// ─── Slider Row Component ───
   169→
   170→function SliderRow({
   171→  id,
   172→  label,
   173→  description,
   174→  value,
   175→  onValueChange,
   176→  min,
   177→  max,
   178→  step = 1,
   179→  formatValue,
   180→  icon: Icon,
   181→}: {
   182→  id: string
   183→  label: string
   184→  description: string
   185→  value: number
   186→  onValueChange: (value: number) => void
   187→  min: number
   188→  max: number
   189→  step?: number
   190→  formatValue?: (value: number) => string
   191→  icon: React.ComponentType<{ className?: string }>
   192→}) {
   193→  const displayValue = formatValue ? formatValue(value) : String(value)
   194→
   195→  return (
   196→    <div className="rounded-lg border p-4 space-y-3">
   197→      <div className="flex items-center gap-3">
   198→        <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-md bg-info/10 text-info">
   199→          <Icon className="h-4 w-4" />
   200→        </div>
   201→        <div className="flex-1 min-w-0">
   202→          <div className="flex items-center justify-between gap-2">
   203→            <Label htmlFor={id} className="text-sm font-medium">
   204→              {label}
   205→            </Label>
   206→            <Badge variant="secondary" className="bg-info/10 text-info font-mono tabular-nums text-xs">
   207→              {displayValue}
   208→            </Badge>
   209→          </div>
   210→          <p className="text-xs text-muted-foreground mt-0.5">
   211→            {description}
   212→          </p>
   213→        </div>
   214→      </div>
   215→      <Slider
   216→        id={id}
   217→        value={[value]}
   218→        onValueChange={([val]) => onValueChange(val)}
   219→        min={min}
   220→        max={max}
   221→        step={step}
   222→        className="w-full"
   223→      />
   224→      <div className="flex justify-between text-xs text-muted-foreground font-mono tabular-nums">
   225→        <span>{formatValue ? formatValue(min) : min}</span>
   226→        <span>{formatValue ? formatValue(max) : max}</span>
   227→      </div>
   228→    </div>
   229→  )
   230→}

   234→export function SecuritePage() {
   235→  const { user } = useAuthStore()
   236→  const queryClient = useQueryClient()
   237→
   238→  // Data state
   239→  const [etablissements, setEtablissements] = useState<Etablissement[]>([])
   240→  const [selectedEtablissementId, setSelectedEtablissementId] = useState<string>('')
   241→  const [settings, setSettings] = useState<SecuritySettings | null>(null)
   242→  const [allSettings, setAllSettings] = useState<SecuritySettings[]>([])
   243→
   244→  // Loading states
   245→  const [loadingSettings, setLoadingSettings] = useState(false)
   246→  const [saving, setSaving] = useState(false)
   247→
   248→  // ─── Data state (BUGFIX QUERY-MIGRATION-GROUP-A : TanStack Query) ───
   249→  // Le cache survit au démontage → 0 refetch au retour, 0 skeleton, navigation
   250→  // instantanée. Les 2 ressources de mount-time fetching sont migrées vers
   251→  // useQuery. Le loadSettings réactif à selectedEtablissementId reste en
   252→  // useEffect (pattern différent : pas un mount-time fetch).
   253→  const etablissementsQuery = useQuery<{ etablissements: Etablissement[] }>({
   254→    queryKey: ['etablissements'],
   255→    queryFn: async () => {
   256→      const res = await fetch('/api/etablissements')
   257→      if (!res.ok) throw new Error('Failed to fetch etablissements')
   258→      return res.json()
   259→    },
   260→    staleTime: 60 * 1000,
   261→    refetchOnWindowFocus: false,
   262→  })
   263→
   264→  const loadingEtablissements = etablissementsQuery.isLoading
   265→
   266→  // Sync query data to local state (préserve l'API interne basée sur
   267→  // setEtablissements utilisée ailleurs dans le composant).
   268→  useEffect(() => {
   269→    if (etablissementsQuery.data) {
   270→      setEtablissements(etablissementsQuery.data.etablissements || [])
   271→    }
   272→  }, [etablissementsQuery.data])
   273→
   274→  // Toast sur erreur de chargement (équivalent du catch du fetch original).
   275→  useEffect(() => {
   276→    if (etablissementsQuery.error) {
   277→      toast.error('Erreur', { description: 'Impossible de charger les établissements' })
   278→    }
   279→  }, [etablissementsQuery.error])
   280→
   281→  const allSettingsQuery = useQuery<{ securitySettings: SecuritySettings[] }>({
   282→    queryKey: ['security-settings'],
   283→    queryFn: async () => {
   284→      const res = await fetch('/api/security-settings')
   285→      if (!res.ok) throw new Error('Failed to fetch security settings')
   286→      return res.json()
   287→    },
   288→    staleTime: 60 * 1000,
   289→    refetchOnWindowFocus: false,
   290→  })
   291→
   292→  // Sync query data to local state.
   293→  useEffect(() => {
   294→    if (allSettingsQuery.data) {
   295→      setAllSettings(allSettingsQuery.data.securitySettings || [])
   296→    }
   297→  }, [allSettingsQuery.data])
   298→
   299→  // Helper pour invalider le cache après mutation (save).
   300→  const refreshAllSettings = () =>
   301→    queryClient.invalidateQueries({ queryKey: ['security-settings'] })
   302→
   303→  // ─── Compute stats ───
   304→
   305→  const statsProctoring = allSettings.filter((s) => s.proctoringActif).length
   306→  const statsVerifIdentite = allSettings.filter((s) => s.verificationIdentite).length
   307→  // Simulated monthly fraud alerts count (based on alert thresholds)
   308→  const statsAlertesFraude = allSettings.reduce((acc, s) => acc + (s.autoSubmitOnViolation ? Math.floor(s.nbAlertesMax * 0.6) : 0), 0)
   309→
   310→  // ─── Load settings for selected etablissement ───
   311→
   312→  useEffect(() => {
   313→    if (!selectedEtablissementId) {
   314→      setSettings(null)
   315→      return
   316→    }
   317→
   318→    async function loadSettings() {
   319→      try {
   320→        setLoadingSettings(true)
   321→        const res = await fetch(`/api/security-settings/etablissement/${selectedEtablissementId}`)
   322→        if (!res.ok) throw new Error('Failed to fetch')
   323→        const data = await res.json()
   324→        setSettings(data.securitySettings)
   325→      } catch {
   326→        toast.error('Erreur', { description: 'Impossible de charger les paramètres de sécurité' })
   327→        setSettings(null)
   328→      } finally {
   329→        setLoadingSettings(false)
   330→      }
   331→    }
   332→
   333→    loadSettings()
   334→  }, [selectedEtablissementId])
   335→
   336→  // ─── Update a single field ───
   337→
   338→  const updateField = <K extends keyof SecuritySettings>(key: K, value: SecuritySettings[K]) => {
   339→    if (!settings) return
   340→    setSettings({ ...settings, [key]: value })
   341→  }
   342→
   343→  // ─── Save settings ───
   344→
   345→  const handleSave = async () => {
   346→    if (!settings) return
   347→
   348→    try {
   349→      setSaving(true)
   350→      const res = await fetch(`/api/security-settings/${settings.id}`, {
   351→        method: 'PATCH',
   352→        headers: { 'Content-Type': 'application/json' },
   353→        body: JSON.stringify({
   354→          proctoringActif: settings.proctoringActif,
   355→          detectionCopie: settings.detectionCopie,
   356→          detectionOnglet: settings.detectionOnglet,
   357→          detectionFullscreen: settings.detectionFullscreen,
   358→          captureEcran: settings.captureEcran,
   359→          blocageCopie: settings.blocageCopie,
   360→          blocageClicDroit: settings.blocageClicDroit,
   361→          blocageImpression: settings.blocageImpression,
   362→          verificationIdentite: settings.verificationIdentite,
   363→          tempsInactiviteMax: settings.tempsInactiviteMax,
   364→          nbOngletsMax: settings.nbOngletsMax,
   365→          nbAlertesMax: settings.nbAlertesMax,
   366→          autoSubmitOnViolation: settings.autoSubmitOnViolation,
   367→          rapportFraude: settings.rapportFraude,
   368→          seuilSimilarite: settings.seuilSimilarite,
   369→          penaliteFullscreenExit: settings.penaliteFullscreenExit,
   370→          fullscreenObligatoire: settings.fullscreenObligatoire,
   371→          intervalleCaptureEcran: settings.intervalleCaptureEcran,
   372→        }),
   373→      })
   374→
   375→      if (!res.ok) {
   376→        const errData = await res.json()
   377→        throw new Error(errData.error || 'Erreur lors de la sauvegarde')
   378→      }
   379→
   380→      toast.success('Paramètres sauvegardés', {
   381→        description: `Les paramètres de sécurité ont été mis à jour pour ${settings.etablissement?.nom || "l'établissement"}.`,
   382→      })
   383→
   384→      // Refresh overview
   385→      refreshAllSettings()
   386→    } catch (err) {
   387→      toast.error('Erreur de sauvegarde', {
   388→        description: err instanceof Error ? err.message : 'Une erreur est survenue',
   389→      })
   390→    } finally {
   391→      setSaving(false)
   392→    }
   393→  }
   394→
   395→  // ─── Navigate to a specific etablissement from the overview table ───
   396→
   397→  const handleSelectFromTable = (etabId: string) => {
   398→    setSelectedEtablissementId(etabId)

   401→
   402→  return (

   405→        {/* ─── Header ─── */}

   410→              Sécurité des Évaluations

   413→              Configurez les paramètres anti-fraude et de surveillance

   437→                    Établissements avec proctoring

   444→          <Card className="border-l-4 border-l-primary">
