    62→function formatDateTime(date: string | Date | null): string {
    63→  if (!date) return '—'
    64→  const d = typeof date === 'string' ? new Date(date) : date
    65→  return d.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit', second: '2-digit' })
    66→}
    67→
    68→function formatRelativeDate(dateStr: string): string {
    69→  const date = new Date(dateStr)
    70→  const now = new Date()
    71→  const diffMs = now.getTime() - date.getTime()
    72→  const diffMin = Math.floor(diffMs / 60000)
    73→  const diffH = Math.floor(diffMin / 60)
    74→  const diffD = Math.floor(diffH / 24)
    75→  if (diffMin < 1) return "À l'instant"
    76→  if (diffMin < 60) return `Il y a ${diffMin} min`
    77→  if (diffH < 24) return `Il y a ${diffH}h`
    78→  if (diffD < 7) return `Il y a ${diffD}j`
    79→  return date.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit' })
    80→}
    81→
    82→function formatTime(date: string): string {
    83→  return new Date(date).toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit', second: '2-digit' })
    84→}

   148→function useDebounce<T>(value: T, delay = 350): T {
   149→  const [debounced, setDebounced] = useState(value)
   150→  useEffect(() => {
   151→    const t = setTimeout(() => setDebounced(value), delay)
   152→    return () => clearTimeout(t)
   153→  }, [value, delay])
   154→  return debounced
   155→}

   898→// ─── Captures Viewer (R2) ───
   899→function CapturesViewer({ captures, index, onClose, onIndexChange }: {
   900→  captures: SessionCapture[]; index: number; onClose: () => void; onIndexChange: (index: number) => void
   901→}) {
   902→  const current = captures[index]
   903→  if (!current) return null
   904→
   905→  const formatFileSize = (bytes?: number) => {
   906→    if (!bytes) return '—'
   907→    if (bytes < 1024) return `${bytes} o`
   908→    if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} Ko`
   909→    return `${(bytes / (1024 * 1024)).toFixed(1)} Mo`
   910→  }
   911→
   912→  return (
   913→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   914→      <DialogContent className="sm:max-w-4xl">
   915→        <DialogHeader>
   916→          <DialogTitle className="flex items-center justify-between gap-2">
   917→            <span>Capture #{current.captureIndex} — {index + 1} / {captures.length}</span>
   918→            <span className="text-xs font-normal text-muted-foreground">
   919→              {formatDateTime(current.createdAt)}
   920→              {current.fileSize ? ` • ${formatFileSize(current.fileSize)}` : ''}
   921→            </span>
   922→          </DialogTitle>
   923→        </DialogHeader>
   924→        <div className="relative flex items-center justify-center">

   928→              alt={`Capture #${current.captureIndex}`}

   931→          ) : (
