   107→   107→const MONTHS_FR = [
   108→   108→  'janvier', 'février', 'mars', 'avril', 'mai', 'juin',
   109→   109→  'juillet', 'août', 'septembre', 'octobre', 'novembre', 'décembre',

   111→   111→
   112→   112→const DAYS_FR = [
   113→   113→  'dimanche', 'lundi', 'mardi', 'mercredi', 'jeudi', 'vendredi', 'samedi',
   114→   114→]
   115→   115→
   116→   116→function formatDateFR(dateStr: string | Date): string {
   117→   117→  const d = typeof dateStr === 'string' ? new Date(dateStr) : dateStr
   118→   118→  return `${d.getDate()} ${MONTHS_FR[d.getMonth()]} ${d.getFullYear()}`
   119→   119→}
   120→   120→
   121→   121→function formatDateTimeFR(dateStr: string | Date): string {
   122→   122→  const d = typeof dateStr === 'string' ? new Date(dateStr) : dateStr
   123→   123→  const day = DAYS_FR[d.getDay()]
   124→   124→  const hours = d.getHours().toString().padStart(2, '0')
   125→   125→  const minutes = d.getMinutes().toString().padStart(2, '0')
   126→   126→  return `${day} ${d.getDate()} ${MONTHS_FR[d.getMonth()]} ${d.getFullYear()} à ${hours}h${minutes}`
   127→   127→}

   129→   129→function parseJsonSafe<T>(value: string | null | unknown, fallback: T): T {
   130→   130→  if (!value) return fallback
   131→   131→  try {
   132→   132→    return JSON.parse(typeof value === 'string' ? value : JSON.stringify(value)) as T
   133→   133→  } catch {
   134→   134→    return fallback
   135→   135→  }
   136→   136→}

   138→   138→function getScoreBadgeClasses(score: number): string {
   139→   139→  if (score >= 10) {
   140→   140→    return 'bg-emerald-100 text-emerald-800 border-emerald-200 dark:bg-emerald-900/40 dark:text-emerald-300 dark:border-emerald-800'
   141→   141→  }
   142→   142→  if (score >= 8) {
   143→   143→    return 'bg-amber-100 text-amber-800 border-amber-200 dark:bg-amber-900/40 dark:text-amber-300 dark:border-amber-800'
   144→   144→  }
   145→   145→  return 'bg-red-100 text-red-800 border-red-200 dark:bg-red-900/40 dark:text-red-300 dark:border-red-800'
   146→   146→}

   149→   149→  if (score >= 10) return 'text-emerald-700 dark:text-emerald-400'
   150→   150→  if (score >= 8) return 'text-amber-700 dark:text-amber-400'
   151→   151→  return 'text-red-700 dark:text-red-400'
