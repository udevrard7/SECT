   104→   104→function wrapText(doc: jsPDF, text: string, maxWidth: number): string[] {

   112→   112→function getTextHeight(doc: jsPDF, text: string, maxWidth: number, fontSize: number = FONT_BODY): number {
   113→   113→  doc.setFontSize(fontSize)
   114→   114→  const lines = wrapText(doc, text, maxWidth)
   115→   115→  return lines.length * (fontSize * 0.3528 * 1.4) + 2
   116→   116→}

   120→   120→function addHeader(doc: jsPDF, data: EpreuvePDFData, isCorrige: boolean = false): void {
   121→   121→  const y = MARGIN_TOP
   122→   122→
   123→   123→  // Logo (if available)
   124→   124→  let textOffsetX = MARGIN_LEFT
   125→   125→  if (data.etablissement.logo) {
   126→   126→    try {
   127→   127→      const logoData = data.etablissement.logo
   128→   128→      // SVG is not supported by jsPDF - skip it
   129→   129→      if (logoData.includes('image/svg+xml')) {
   130→   130→        console.warn('[PDF] SVG logos are not supported in PDF generation, skipping')
   131→   131→      } else {
   132→   132→        // Detect format from data URL
   133→   133→        let format = 'PNG'
   134→   134→        if (logoData.includes('image/jpeg') || logoData.includes('image/jpg')) {
   135→   135→          format = 'JPEG'
   136→   136→        } else if (logoData.includes('image/webp')) {
   137→   137→          format = 'PNG' // jsPDF may not support WEBP; will try as PNG fallback
   138→   138→        }
   139→   139→        const logoWidth = 18  // mm
   140→   140→        const logoHeight = 18 // mm
   141→   141→        doc.addImage(logoData, format, MARGIN_LEFT, y, logoWidth, logoHeight)
   142→   142→        textOffsetX = MARGIN_LEFT + logoWidth + 4
   143→   143→      }
   144→   144→    } catch (e) {
   145→   145→      // If logo fails to render, just skip it
   146→   146→      console.warn('[PDF] Failed to render logo:', e)
   147→   147→    }
   148→   148→  }
   149→   149→
   150→   150→  // Left side: Etablissement info
   151→   151→  doc.setFontSize(12)
   152→   152→  doc.setFont('helvetica', 'bold')
   153→   153→  doc.text(data.etablissement.nom, textOffsetX, y + 5)
   154→   154→
   155→   155→  doc.setFontSize(FONT_SMALL)
   156→   156→  doc.setFont('helvetica', 'normal')
   157→   157→  let infoY = y + 10
   158→   158→  if (data.etablissement.ville) {
   159→   159→    doc.text(`${data.etablissement.ville}${data.etablissement.pays ? `, ${data.etablissement.pays}` : ''}`, textOffsetX, infoY)
   160→   160→    infoY += 4
   161→   161→  }
   162→   162→
   163→   163→  // Academic year
   164→   164→  doc.text(`Année universitaire : ${getAcademicYear()}`, textOffsetX, infoY)
   165→   165→
   166→   166→  // Right side: UE & Filière info
   167→   167→  const rightX = PAGE_WIDTH - MARGIN_RIGHT
   168→   168→  doc.setFontSize(FONT_SMALL)
   169→   169→  doc.setFont('helvetica', 'normal')
   170→   170→
   171→   171→  let rightY = y + 5
   172→   172→  if (data.uniteEnseignement) {
   173→   173→    doc.setFont('helvetica', 'bold')
   174→   174→    const ueText = `UE : ${data.uniteEnseignement.code} - ${data.uniteEnseignement.nom}`
   175→   175→    const ueLines = wrapText(doc, ueText, 90)
   176→   176→    for (const line of ueLines) {
   177→   177→      doc.text(line, rightX, rightY, { align: 'right' })
   178→   178→      rightY += 4
   179→   179→    }
   180→   180→    doc.setFont('helvetica', 'normal')
   181→   181→  }
   182→   182→  if (data.filiere) {
   183→   183→    doc.text(`Filière : ${data.filiere.nom}${data.filiere.code ? ` (${data.filiere.code})` : ''}`, rightX, rightY, { align: 'right' })
   184→   184→    rightY += 4
   185→   185→  }
   186→   186→
   187→   187→  // Separator line
   188→   188→  const sepY = HEADER_BOTTOM - 4
   189→   189→  doc.setDrawColor(180, 180, 180)
   190→   190→  doc.setLineWidth(0.5)
   191→   191→  doc.line(MARGIN_LEFT, sepY, PAGE_WIDTH - MARGIN_RIGHT, sepY)
   192→   192→
   193→   193→  // Watermark for Corrigé
   194→   194→  if (isCorrige) {
   195→   195→    doc.setFontSize(9)
   196→   196→    doc.setFont('helvetica', 'bold')
   197→   197→    doc.setTextColor(220, 50, 50)
   198→   198→    doc.text('CORRIGÉ TYPE — CONFIDENTIEL', PAGE_WIDTH / 2, sepY - 2, { align: 'center' })
   199→   199→    doc.setTextColor(0, 0, 0)
   200→   200→  }
   201→   201→}

   203→   203→function addFooter(doc: jsPDF, data: EpreuvePDFData, pageNum: number, totalPages: number, isCorrige: boolean = false): void {
   204→   204→  const y = FOOTER_TOP
   205→   205→
   206→   206→  // Separator line

   210→   210→
   211→   211→  doc.setFontSize(FONT_TINY)
   212→   212→  doc.setFont('helvetica', 'normal')
   213→   213→  doc.setTextColor(120, 120, 120)
   214→   214→
   215→   215→  // Left: Confidential
   216→   216→  const label = isCorrige ? 'Corrigé confidentiel' : 'Confidentiel'
   217→   217→  doc.text(label, MARGIN_LEFT, y + 4)
   218→   218→
   219→   219→  // Center: Epreuve title (truncated)
   220→   220→  const maxTitleLen = 60
   221→   221→  const title = data.titre.length > maxTitleLen ? data.titre.slice(0, maxTitleLen) + '…' : data.titre
   222→   222→  doc.text(title, PAGE_WIDTH / 2, y + 4, { align: 'center' })
   223→   223→
   224→   224→  // Right: Page number
   225→   225→  doc.text(`Page ${pageNum} / ${totalPages}`, PAGE_WIDTH - MARGIN_RIGHT, y + 4, { align: 'right' })
   226→   226→
   227→   227→  doc.setTextColor(0, 0, 0)
   228→   228→}

   255→   255→// ─── Question Rendering (Sujet) ───
   256→   256→
   257→   257→/**
   258→   258→ * Estimate the height needed for a question in Sujet mode.
   259→   259→ */
   260→   260→function estimateQuestionHeightSujet(doc: jsPDF, q: PDFQuestion): number {
   261→   261→  let height = 0
   262→   262→
   263→   263→  // Question header line
   264→   264→  height += 8
   265→   265→
   266→   266→  // Question text
   267→   267→  height += getTextHeight(doc, q.enonce, CONTENT_WIDTH - 5, FONT_BODY)
   268→   268→
   269→   269→  // Propositions for QCU/QCM
   270→   270→  if ((q.type === 'QCU' || q.type === 'QCM') && q.propositions && q.propositions.length > 0) {
   271→   271→    height += 3 // gap
   272→   272→    for (const prop of q.propositions) {
   273→   273→      height += getTextHeight(doc, prop.text, CONTENT_WIDTH - 15, FONT_BODY) + 4
   274→   274→    }
   275→   275→  }
   276→   276→
   277→   277→  // Response lines for QRC
   278→   278→  if (q.type === 'QRC') {
   279→   279→    height += 3 + 5 * ANSWER_LINE_SPACING
   280→   280→  }
   281→   281→
   282→   282→  // Response lines for REFLEXION
   283→   283→  if (q.type === 'REFLEXION') {
   284→   284→    height += 3 + 8 * ANSWER_LINE_SPACING
   285→   285→  }
   286→   286→
   287→   287→  // Code section for CODE type
   288→   288→  if (q.type === 'CODE') {
   289→   289→    height += 3
   290→   290→    // Language badge + signature
   291→   291→    height += 8
   292→   292→    // Starter code box
   293→   293→    if (q.codeInitial) {
   294→   294→      const codeLines = q.codeInitial.split('\n').length
   295→   295→      height += Math.max(codeLines * 4 + 10, 40) // 4mm per code line, min 40mm
   296→   296→    } else {
   297→   297→      height += 40 // Empty code area
   298→   298→    }
   299→   299→    // Public tests table
   300→   300→    if (q.testsPublics && q.testsPublics.length > 0) {
   301→   301→      height += 8 + q.testsPublics.length * 8
   302→   302→    }
   303→   303→  }
   304→   304→
   305→   305→  // Bottom gap + separator
   306→   306→  height += QUESTION_GAP + 3
   307→   307→
   308→   308→  return height
   309→   309→}

   311→   311→/**
   312→   312→ * Render a single question in Sujet mode.
   313→   313→ * Returns the Y position after rendering.
   314→   314→ */
   315→   315→function renderQuestionSujet(doc: jsPDF, q: PDFQuestion, index: number, startY: number): number {
   316→   316→  let y = startY
   317→   317→
   318→   318→  // Question header — measure width at the SAME font size before switching
   319→   319→  doc.setFontSize(FONT_HEADING)
   320→   320→  doc.setFont('helvetica', 'bold')
   321→   321→  doc.text(`Question ${index + 1}`, MARGIN_LEFT, y)
   322→   322→  const questionLabelWidth = doc.getTextWidth(`Question ${index + 1}  `) // Measure at size 11 bold
   323→   323→
   324→   324→  // Type and bareme
   325→   325→  doc.setFontSize(FONT_SMALL)
   326→   326→  doc.setFont('helvetica', 'normal')
   327→   327→  const typeLabel = getTypeLabel(q.type)
   328→   328→  doc.text(typeLabel, MARGIN_LEFT + questionLabelWidth, y)
   329→   329→
   330→   330→  // Bareme on the right
   331→   331→  doc.setFont('helvetica', 'bold')
   332→   332→  const baremeText = `${q.bareme} pt${q.bareme > 1 ? 's' : ''}`
   333→   333→  doc.text(baremeText, PAGE_WIDTH - MARGIN_RIGHT, y, { align: 'right' })
   334→   334→
   335→   335→  y += 6
   336→   336→
   337→   337→  // Question text
   338→   338→  doc.setFontSize(FONT_BODY)
   339→   339→  doc.setFont('helvetica', 'normal')
   340→   340→  const lines = wrapText(doc, q.enonce, CONTENT_WIDTH - 5)
   341→   341→  for (const line of lines) {
   342→   342→    doc.text(line, MARGIN_LEFT, y)
   343→   343→    y += LINE_HEIGHT_BODY
   344→   344→  }
   345→   345→
   346→   346→  y += 2
   347→   347→
   348→   348→  // Propositions for QCU/QCM
   349→   349→  if ((q.type === 'QCU' || q.type === 'QCM') && q.propositions && q.propositions.length > 0) {
   350→   350→    for (let pi = 0; pi < q.propositions.length; pi++) {
   351→   351→      const prop = q.propositions[pi]
   352→   352→      const letter = String.fromCharCode(65 + pi)
   353→   353→
   354→   354→      if (q.type === 'QCU') {
   355→   355→        // Draw circle for QCU
   356→   356→        doc.setDrawColor(80, 80, 80)
   357→   357→        doc.setLineWidth(0.4)
   358→   358→        doc.circle(MARGIN_LEFT + 3, y - 1, 2.5)
   359→   359→        doc.setFontSize(FONT_SMALL)
   360→   360→        doc.setFont('helvetica', 'bold')
   361→   361→        doc.text(letter, MARGIN_LEFT + 3, y - 0.5, { align: 'center' })
   362→   362→      } else {
   363→   363→        // Draw square for QCM
   364→   364→        doc.setDrawColor(80, 80, 80)
   365→   365→        doc.setLineWidth(0.4)
   366→   366→        doc.rect(MARGIN_LEFT + 0.5, y - 3.5, 5, 5)
   367→   367→        doc.setFontSize(FONT_SMALL)
   368→   368→        doc.setFont('helvetica', 'bold')
   369→   369→        doc.text(letter, MARGIN_LEFT + 3, y - 0.5, { align: 'center' })
   370→   370→      }
   371→   371→
   372→   372→      // Proposition text
   373→   373→      doc.setFontSize(FONT_BODY)
   374→   374→      doc.setFont('helvetica', 'normal')
   375→   375→      const propLines = wrapText(doc, prop.text, CONTENT_WIDTH - 15)
   376→   376→      for (const line of propLines) {
   377→   377→        doc.text(line, MARGIN_LEFT + 10, y)
   378→   378→        y += LINE_HEIGHT_BODY
   379→   379→      }
   380→   380→      y += 1
   381→   381→    }
   382→   382→  }
   383→   383→
   384→   384→  // Response lines for QRC
   385→   385→  if (q.type === 'QRC') {
   386→   386→    y += 2
   387→   387→    const numLines = 5
   388→   388→    for (let i = 0; i < numLines; i++) {
   389→   389→      doc.setDrawColor(190, 190, 190)
   390→   390→      doc.setLineWidth(0.3)
   391→   391→      doc.line(MARGIN_LEFT, y, PAGE_WIDTH - MARGIN_RIGHT, y)
   392→   392→      y += ANSWER_LINE_SPACING
   393→   393→    }
   394→   394→  }
   395→   395→
   396→   396→  // Response lines for REFLEXION
   397→   397→  if (q.type === 'REFLEXION') {
   398→   398→    y += 2
   399→   399→    const numLines = 8
   400→   400→    for (let i = 0; i < numLines; i++) {
   401→   401→      doc.setDrawColor(190, 190, 190)
   402→   402→      doc.setLineWidth(0.3)
   403→   403→      doc.line(MARGIN_LEFT, y, PAGE_WIDTH - MARGIN_RIGHT, y)
   404→   404→      y += ANSWER_LINE_SPACING
   405→   405→    }
   406→   406→  }
   407→   407→
   408→   408→  // Code section for CODE type
   409→   409→  if (q.type === 'CODE') {
   410→   410→    y += 2
   411→   411→
   412→   412→    // Language + signature info
   413→   413→    doc.setFontSize(FONT_SMALL)
   414→   414→    doc.setFont('helvetica', 'bold')
   415→   415→    doc.setTextColor(100, 60, 160)
   416→   416→    const langLabel = q.langage ? q.langage.toUpperCase() : 'CODE'
   417→   417→    doc.text(`Langage : ${langLabel}`, MARGIN_LEFT, y)
   418→   418→    if (q.fonctionSignature) {
   419→   419→      doc.setFont('helvetica', 'normal')
   420→   420→      doc.setTextColor(80, 80, 80)
   421→   421→      doc.text(`Signature : ${q.fonctionSignature}`, MARGIN_LEFT + 50, y)
   422→   422→    }
   423→   423→    doc.setTextColor(0, 0, 0)
   424→   424→    y += 6
   425→   425→
   426→   426→    // Starter code box
   427→   427→    doc.setFillColor(245, 245, 250)
   428→   428→    doc.setDrawColor(180, 170, 210)
   429→   429→    doc.setLineWidth(0.4)
   430→   430→    const codeText = q.codeInitial || '// Écrire votre code ici'
   431→   431→    const codeLines = codeText.split('\n')
   432→   432→    const codeBoxHeight = Math.max(codeLines.length * 4 + 10, 35)
   433→   433→    doc.roundedRect(MARGIN_LEFT, y - 3, CONTENT_WIDTH, codeBoxHeight, 2, 2, 'FD')
   434→   434→
   435→   435→    // Code text
   436→   436→    doc.setFontSize(9)
   437→   437→    doc.setFont('courier', 'normal')
   438→   438→    doc.setTextColor(40, 40, 60)
   439→   439→    let codeY = y + 2
   440→   440→    for (const line of codeLines.slice(0, 20)) { // Max 20 lines to prevent overflow
   441→   441→      if (codeY > y + codeBoxHeight - 5) break
   442→   442→      doc.text(line.substring(0, 80), MARGIN_LEFT + 4, codeY) // Truncate long lines
   443→   443→      codeY += 4
   444→   444→    }
   445→   445→    doc.setTextColor(0, 0, 0)
   446→   446→    y += codeBoxHeight + 4
   447→   447→
   448→   448→    // Public tests table
   449→   449→    if (q.testsPublics && q.testsPublics.length > 0) {
   450→   450→      doc.setFontSize(FONT_SMALL)
   451→   451→      doc.setFont('helvetica', 'bold')
   452→   452→      doc.text('Tests publics :', MARGIN_LEFT, y)
   453→   453→      y += 5
   454→   454→
   455→   455→      const testHead = [['N°', 'Entrée', 'Sortie attendue']]
   456→   456→      const testBody = q.testsPublics.map((t, i) => [
   457→   457→        `${i + 1}`,
   458→   458→        t.entree.substring(0, 40),

   504→   504→ * Estimate the height needed for a question in Corrigé mode.
   505→   505→ */
   506→   506→function estimateQuestionHeightCorrige(doc: jsPDF, q: PDFQuestion): number {
   507→   507→  let height = estimateQuestionHeightSujet(doc, q)
   508→   508→
   509→   509→  // Additional height for correct answers / model response
   510→   510→  if ((q.type === 'QCU' || q.type === 'QCM') && q.reponseCorrecte) {
   511→   511→    height += LINE_HEIGHT_BODY + 12  // Label line + answer content + spacing
   512→   512→  }

   514→   514→  if ((q.type === 'QRC' || q.type === 'REFLEXION') && q.reponseCorrecte) {
   515→   515→    const text = Array.isArray(q.reponseCorrecte) ? q.reponseCorrecte.join('\n') : q.reponseCorrecte
   516→   516→    height += getTextHeight(doc, text, CONTENT_WIDTH - 15, FONT_BODY) + LINE_HEIGHT_BODY + 8
   517→   517→  }
   518→   518→
   519→   519→  if (q.explication) {
   520→   520→    height += getTextHeight(doc, q.explication, CONTENT_WIDTH - 15, FONT_BODY) + LINE_HEIGHT_BODY + 8
   521→   521→  }
   522→   522→
   523→   523→  return height
   524→   524→}

   526→   526→/**
   527→   527→ * Render a single question in Corrigé mode.
   528→   528→ * Returns the Y position after rendering.
   529→   529→ */
   530→   530→function renderQuestionCorrige(doc: jsPDF, q: PDFQuestion, index: number, startY: number): number {
   531→   531→  let y = startY
   532→   532→
   533→   533→  // First render the question like Sujet
   534→   534→  y = renderQuestionSujet(doc, q, index, y)
   535→   535→
   536→   536→  // Correct answers for QCU/QCM
   537→   537→  if ((q.type === 'QCU' || q.type === 'QCM') && q.propositions && q.reponseCorrecte) {
   538→   538→    const correctIds = Array.isArray(q.reponseCorrecte) ? q.reponseCorrecte : [q.reponseCorrecte]
   539→   539→
   540→   540→    doc.setFontSize(FONT_SMALL)
   541→   541→    doc.setFont('helvetica', 'bold')
   542→   542→    doc.setTextColor(0, 130, 60)
   543→   543→    doc.text('Réponse correcte :', MARGIN_LEFT, y)
   544→   544→    y += LINE_HEIGHT_BODY  // Advance past the label to avoid overlap
   545→   545→
   546→   546→    const correctLabels = q.propositions
   547→   547→      .filter((p) => correctIds.includes(p.id))
   548→   548→      .map((p) => {
   549→   549→        const idx = q.propositions!.indexOf(p)
   550→   550→        return `${String.fromCharCode(65 + idx)}. ${p.text}`
   551→   551→      })
   552→   552→
   553→   553→    doc.setFont('helvetica', 'normal')
   554→   554→    doc.setFontSize(FONT_BODY)
   555→   555→    for (const label of correctLabels) {
   556→   556→      const lines = wrapText(doc, label, CONTENT_WIDTH - 20)
   557→   557→      for (const line of lines) {
   558→   558→        doc.text(line, MARGIN_LEFT + 10, y)
   559→   559→        y += LINE_HEIGHT_BODY
   560→   560→      }
   561→   561→    }
   562→   562→    doc.setTextColor(0, 0, 0)
   563→   563→    y += 3
   564→   564→  }
   565→   565→
   566→   566→  // Model response for QRC / REFLEXION
   567→   567→  if ((q.type === 'QRC' || q.type === 'REFLEXION') && q.reponseCorrecte) {
   568→   568→    const text = Array.isArray(q.reponseCorrecte) ? q.reponseCorrecte.join('\n') : q.reponseCorrecte
   569→   569→    const label = q.type === 'QRC' ? 'Réponse modèle :' : 'Guide de correction :'
   570→   570→
   571→   571→    doc.setFontSize(FONT_SMALL)
   572→   572→    doc.setFont('helvetica', 'bold')
   573→   573→    doc.setTextColor(0, 100, 50)
   574→   574→    doc.text(label, MARGIN_LEFT, y)
   575→   575→    doc.setTextColor(0, 0, 0)
   576→   576→    y += LINE_HEIGHT_BODY  // Advance past the label to avoid overlap
   577→   577→
   578→   578→    doc.setFontSize(FONT_BODY)
   579→   579→    doc.setFont('helvetica', 'normal')
   580→   580→    const lines = wrapText(doc, text, CONTENT_WIDTH - 15)
   581→   581→    const boxHeight = lines.length * LINE_HEIGHT_BODY + 6
   582→   582→
   583→   583→    doc.setFillColor(245, 250, 245)
   584→   584→    doc.setDrawColor(0, 160, 80)
   585→   585→    doc.roundedRect(MARGIN_LEFT, y - 3, CONTENT_WIDTH, boxHeight, 2, 2, 'FD')
   586→   586→
   587→   587→    doc.setTextColor(0, 80, 40)
   588→   588→    for (const line of lines) {
   589→   589→      doc.text(line, MARGIN_LEFT + 5, y)
   590→   590→      y += LINE_HEIGHT_BODY
   591→   591→    }
   592→   592→    doc.setTextColor(0, 0, 0)
   593→   593→    y += 5
   594→   594→  }
   595→   595→
   596→   596→  // Explanation
   597→   597→  if (q.explication) {
   598→   598→    doc.setFontSize(FONT_SMALL)
   599→   599→    doc.setFont('helvetica', 'bold')
   600→   600→    doc.setTextColor(80, 80, 140)
   601→   601→    doc.text('Explication :', MARGIN_LEFT, y)
   602→   602→    doc.setTextColor(0, 0, 0)
   603→   603→    y += LINE_HEIGHT_BODY  // Advance past the label to avoid overlap
   604→   604→
   605→   605→    doc.setFontSize(FONT_BODY)
   606→   606→    doc.setFont('helvetica', 'normal')
   607→   607→    doc.setTextColor(80, 80, 80)
   608→   608→    const lines = wrapText(doc, q.explication, CONTENT_WIDTH - 10)
   609→   609→    for (const line of lines) {
   610→   610→      doc.text(line, MARGIN_LEFT + 5, y)
   611→   611→      y += LINE_HEIGHT_BODY
   612→   612→    }
   613→   613→    doc.setTextColor(0, 0, 0)
   614→   614→    y += 3
   615→   615→  }
   616→   616→
   617→   617→  return y
   618→   618→}

   704→   704→    addFooter(doc, data, p, totalPages, false)
   705→   705→  }
   706→   706→
   707→   707→  return doc
   708→   708→}

   904→   904→        if (hookData.cell.raw === '__CIRCLE__') {
   905→   905→          hookData.cell.raw = ''
   906→   906→          hookData.cell.text = ''  // Clear text content to prevent any rendering
   907→   907→          ;(hookData.cell as any)._isCircle = true
   908→   908→        }
   909→   909→      },
   910→   910→      didDrawCell: (hookData) => {
   911→   911→        // Draw a circle in cells that had __CIRCLE__
   912→   912→        if ((hookData.cell as any)._isCircle) {
   913→   913→          const { x, y: cellY, width, height } = hookData.cell
   914→   914→          const cx = x + width / 2
   915→   915→          const cy = cellY + height / 2
   916→   916→          const radius = Math.min(width, height) / 2 - 1.5
   917→   917→          doc.setDrawColor(60, 60, 60)
   918→   918→          doc.setLineWidth(0.5)
   919→   919→          doc.circle(cx, cy, radius)
   920→   920→        }
   921→   921→      },
   922→   922→    })
   923→   923→
   924→   924→    // Get the Y position after the table
   925→   925→    y = (doc as any).lastAutoTable?.finalY ?? y + 20
   926→   926→    y += 10
   927→   927→  }
   928→   928→
   929→   929→  // Open-ended questions section
   930→   930→  const openQuestions = data.questions.filter((q) => q.type === 'QRC' || q.type === 'REFLEXION')
   931→   931→
   932→   932→  if (openQuestions.length > 0) {
   933→   933→    y = ensureSpace(doc, data, y, 20, false)
   934→   934→
   935→   935→    doc.setFontSize(FONT_HEADING)
   936→   936→    doc.setFont('helvetica', 'bold')
   937→   937→    doc.text('Questions ouvertes', MARGIN_LEFT, y)
   938→   938→    y += 8
   939→   939→
   940→   940→    for (const q of openQuestions) {
   941→   941→      const originalIdx = data.questions.indexOf(q)
   942→   942→      const numLines = q.type === 'QRC' ? 5 : 8
   943→   943→      const neededHeight = numLines * ANSWER_LINE_SPACING + 20
   944→   944→
   945→   945→      y = ensureSpace(doc, data, y, neededHeight, false)
   946→   946→
   947→   947→      doc.setFontSize(FONT_BODY)
   948→   948→      doc.setFont('helvetica', 'bold')
   949→   949→      doc.text(`Question ${originalIdx + 1} (${q.type} - ${q.bareme} pts)`, MARGIN_LEFT, y)
   950→   950→      y += 5
   951→   951→
   952→   952→      doc.setFont('helvetica', 'normal')
   953→   953→      const lines = wrapText(doc, q.enonce, CONTENT_WIDTH - 5)
   954→   954→      for (const line of lines) {
   955→   955→        doc.text(line, MARGIN_LEFT, y)
   956→   956→        y += LINE_HEIGHT_BODY
   957→   957→      }
   958→   958→      y += 3
   959→   959→
   960→   960→      // Response lines
   961→   961→      for (let i = 0; i < numLines; i++) {
   962→   962→        doc.setDrawColor(190, 190, 190)
   963→   963→        doc.setLineWidth(0.3)
   964→   964→        doc.line(MARGIN_LEFT, y, PAGE_WIDTH - MARGIN_RIGHT, y)
   965→   965→        y += ANSWER_LINE_SPACING
   966→   966→      }
   967→   967→      y += 5
   968→   968→    }
   969→   969→  }
   970→   970→
   971→   971→  // Add footers
   972→   972→  const totalPages = doc.getNumberOfPages()
   973→   973→  for (let p = 1; p <= totalPages; p++) {
   974→   974→    doc.setPage(p)
   975→   975→    addFooter(doc, data, p, totalPages, false)
   976→   976→  }
   977→   977→
   978→   978→  return doc
   979→   979→}
