   155→// SUBMIT-JITTER-1 (OPT-7) : amplitude du jitter appliqué à l'auto-submit de
   156→// fin de temps. Quand le chronomètre atteint 0, au lieu de soumettre à l'instant
   157→// (t=0 pour toute la promotion → pic qui sature le backend), on tire un délai
   158→// aléatoire dans [0, SUBMIT_JITTER_MS] pour étaler les soumissions sur cette
   159→// fenêtre. 45s étaie 1000 étudiants sur ~6-7 submits/s, juste sous la capacité
   160→// Render free. Overridable via NEXT_PUBLIC_SUBMIT_JITTER_MS (ex. 60000 en prod).
   161→const SUBMIT_JITTER_MS = (() => {
   162→  const raw = typeof process !== 'undefined' ? process.env.NEXT_PUBLIC_SUBMIT_JITTER_MS : undefined
   163→  const parsed = raw ? parseInt(raw, 10) : NaN
   164→  return Number.isFinite(parsed) && parsed >= 0 && parsed <= 120000 ? parsed : 45000
   165→})()

   170→// ─── Utility: format time as HH:MM:SS ──────────────────────────────────────
   171→
   172→function formatTime(totalSeconds: number): string {
   173→  if (totalSeconds <= 0) return '00:00:00'
   174→  const h = Math.floor(totalSeconds / 3600)
   175→  const m = Math.floor((totalSeconds % 3600) / 60)
   176→  const s = totalSeconds % 60
   177→  return [h, m, s].map((v) => String(v).padStart(2, '0')).join(':')
   178→}

   182→function parseJsonSafe<T>(value: string | null | undefined, fallback: T): T {
   183→  if (!value) return fallback
   184→  try {
   185→    return JSON.parse(value) as T
   186→  } catch {
   187→    return fallback
   188→  }
   189→}

   197→// la passation attendait des strings. React error #31 quand il tentait de
   198→// rendre {prop} (objet) comme enfant de <span>. Ce helper gère les 2 formats.
   199→function getPropText(prop: unknown): string {
   200→  if (typeof prop === 'string') return prop
   201→  if (prop && typeof prop === 'object' && 'text' in prop) {
   202→    return String((prop as { text: unknown }).text)
   203→  }
   204→  return String(prop)
   205→}

   462→              sessionData.reponses.forEach((r: { questionId: string; contenu: string | null }) => {

   471→              const answer = parseCodingAnswer(contenu)

   473→                loadedLanguages[qId] = answer.language
   474→              }
   475→            })

   477→              activeCodeLanguages = loadedLanguages
   478→            }
   479→          }

   481→          // Count fullscreen exits from logEvents to restore penalty count

   484→              const logs = JSON.parse(activeSession.logEvents as string)
   485→              fullscreenExitCount = logs.filter((l: { type: string }) => l.type === 'FULLSCREEN_EXIT').length
   486→            } catch {
   487→              // Ignore parse errors
   488→            }
   489→          }

   491→          // Resume exam directly
   492→          phase = 'in-exam'
   493→        }
   494→      }
   495→
   496→      // OPT-10: save snapshot for future stale-while-revalidate
   497→      if (user?.id) {
   498→        saveExamSnapshot({
   499→          epreuveId,
   500→          userId: user.id,
   501→          epreuve: epreuveInfo,
   502→          questions: questionsData,
   503→          session: activeSession,
   504→          reponses,
   505→          activeCodeLanguages,
   506→          timeRemaining,
   507→          fullscreenExitCount,
   508→          totalAlertCount,
   509→          penalite,
   510→        }).catch(() => {}) // non-blocking
   511→      }

   514→        epreuve: epreuveInfo,
   515→        securityConfig,
   516→        session: activeSession,
   517→        questions: questionsData,

   530→    refetchOnReconnect: false,
   531→  })

   533→  // Sync query.data → local state (préserve le comportement du fetch original
   534→  // qui setEpreuve/setSecurityConfig/etc. au fur et à mesure). Le useEffect ne

   553→    // elles reflètent la dernière action de l'étudiant.
   554→    if (d.session?.id) {

   558→          // présentes dans le cache (plus récentes)

   560→        } else {

   563→      })
   564→    } else {

   568→

   574→        description: err instanceof Error ? err.message : 'Impossible de charger l\'épreuve',
   575→      })

   578→

   580→  // du guard original), sinon isLoading de la query.

   582→
   583→  // ─── Start exam ────────────────────────────────────────────────────────
   584→  const startExam = useCallback(async () => {

   600→        }

   619→

   621→      lastActivityRef.current = Date.now()
   622→

   624→      // E2E-TEST-FIX : bypass en mode test (NEXT_PUBLIC_TEST_MODE=true) pour

   649→    }

   651→
   652→  // ─── Auto-save answers ─────────────────────────────────────────────────
   653→  // BUGFIX (EXAM-OFFLINE-1) : saveAnswers gère maintenant le hors-ligne.
   654→  // Si le fetch échoue (micro-coupure), les réponses sont déjà en IndexedDB
   655→  // (via handleAnswerChange) et un retry est programmé au retour online.
   656→  //
   657→  // OPT-5 : skip l'auto-save si les réponses n'ont pas changé depuis le dernier save.
   658→  // Réduit ~50% des requêtes HTTP pendant l'examen (les étudiants ne modifient
   659→  // pas leurs réponses en continu). Pour 5000 étudiants, on passe de 167 saves/s
   660→  // à ~83 saves/s en moyenne.
   661→  const saveAnswers = useCallback(async () => {
   662→    if (!sessionRef.current || phaseRef.current !== 'in-exam') return
   663→
   664→    try {
   665→      const currentReponses = reponsesRef.current
   666→      const entries = Object.entries(currentReponses)
   667→
   668→      if (entries.length === 0) {
   669→        setLastSaved(new Date())
   670→        return
   671→      }
   672→
   673→      // OPT-5 : comparer les réponses actuelles avec les dernières sauvegardées.
   674→      // Si rien n'a changé, on skip le save (économise 1 requête HTTP + 1 write DB).
   675→      const currentHash = JSON.stringify(entries.sort())
   676→      if (currentHash === lastSavedReponsesHashRef.current) {
   677→        return // Rien n'a changé, skip
   678→      }
   679→
   680→      // Batch save all answers
   681→      const res = await fetch(`/api/sessions/${sessionRef.current.id}`, {
   682→        method: 'PUT',
   683→        headers: { 'Content-Type': 'application/json' },
   684→        body: JSON.stringify({ reponses: currentReponses }),
   685→      })
   686→
   687→      if (res.ok) {
   688→        lastSavedReponsesHashRef.current = currentHash
   689→        setLastSaved(new Date())
   690→      }
   691→    } catch {
   692→      // BUGFIX (EXAM-OFFLINE-1) : réseau coupé — les réponses sont déjà
   693→      // en IndexedDB (sauvegardées à chaque handleAnswerChange). Un retry
   694→      // automatique se déclenchera au retour du réseau (online event).
   695→      // On ne perturbe pas l'étudiant (silent fail).
   696→    }
   697→  }, [])
   698→
   699→  // OPT-9 : version debouncée de saveAnswers.
   700→  // Quand l'étudiant navigue rapidement entre les questions, on coalesce
   701→  // les saves en un seul appel API après 2s d'inactivité.
   702→  // Le flush immédiat reste disponible pour le submit (debouncedSave.flush()).
   703→  const debouncedSaveAnswers = useMemo(
   704→    () => debounce(saveAnswers, 2000),
   705→    [saveAnswers]
   706→  )
   707→
   708→  // Cleanup du debounce au démontage
   709→  useEffect(() => {
   710→    return () => {
   711→      debouncedSaveAnswers.cancel()
   712→    }
   713→  }, [debouncedSaveAnswers])
   714→
   715→  // Annule le jitter si l'étudiant soumet manuellement avant la fin du compte
   716→  // à rebours. Appelé depuis submitExam (via cleanup) et au unmount.
   717→  const cancelPendingAutoSubmit = useCallback(() => {
   718→    if (pendingAutoSubmitTimeoutRef.current) {
   719→      clearTimeout(pendingAutoSubmitTimeoutRef.current)
   720→      pendingAutoSubmitTimeoutRef.current = null
   721→    }
   722→    setPendingAutoSubmitIn(null)
   723→  }, [])
   724→
   725→  // ─── Submit exam ───────────────────────────────────────────────────────
   726→  // SUBMIT-JITTER-1 + SUBMIT-RATELIMIT-1 (OPT-7) : gère le 202 Accepted du
   727→  // backend (file de submits pleine) en réessayant après le Retry-After
   728→  // indiqué. Le submit reste idempotent côté backend (statut session vérifié),
   729→  // donc un retry est sûr. Max SUBMIT_MAX_RETRIES pour rester sous le timeout
   730→  // de session frontend.
   731→  const submitExam = useCallback(async (autoSubmit: boolean = false, reason: AutoSubmitReason = null) => {
   732→    if (!sessionRef.current || isAutoSubmittingRef.current) return
   733→    if (isSubmitting && !autoSubmit) return
   734→
   735→    // SUBMIT-JITTER-1 : si l'étudiant soumet manuellement pendant la fenêtre
   736→    // de jitter, annuler le submit jitteré planifié.
   737→    cancelPendingAutoSubmit()
   738→
   739→    isAutoSubmittingRef.current = true
   740→    setIsSubmitting(true)
   741→
   742→    try {
   743→      // Save answers first
   744→      // OPT-9 : flush immédiat de tous les saves en attente avant le submit
   745→      debouncedSaveAnswers.flush()
   746→      // Small delay to let the flush complete
   747→      await new Promise(resolve => setTimeout(resolve, 100))
   748→
   749→      type SubmitResponse = {
   750→        penalite?: number
   751→        scenario?: 'A' | 'B'
   752→        scenarioMessage?: string
   753→        score?: number
   754→        totalPossible?: number
   755→        autoGradableTotal?: number
   756→        pendingCorrection?: number
   757→      }
   758→
   759→      // SUBMIT-RATELIMIT-1 : on ne peut pas utiliser submitOffline pour le
   760→      // submit final car le hook traite tout 2xx (dont 202) comme un succès.
   761→      // Or 202 = "file pleine, réessayez" → il faut retry avec Retry-After.
   762→      // On fait donc un fetch direct avec boucle de retry, en gardant le
   763→      // fallback offline (outbox) si le réseau lâche complètement.
   764→      const submitUrl = `/api/sessions/${sessionRef.current.id}/submit`
   765→      const submitBody = JSON.stringify({ autoSubmit, reponses: reponsesRef.current })
   766→
   767→      let data: SubmitResponse | undefined
   768→      let lastError: string | undefined
   769→      let attempt = 0
   770→      let queued = false
   771→
   772→      while (attempt < SUBMIT_MAX_RETRIES) {
   773→        attempt += 1
   774→
   775→        // Offline → outbox (une seule fois, puis break)
   776→        if (typeof navigator !== 'undefined' && !navigator.onLine) {
   777→          const result = await submitOffline<SubmitResponse>(
   778→            submitUrl,
   779→            { autoSubmit, reponses: reponsesRef.current },
   780→            { type: 'submit-exam', meta: { examTitle: epreuve?.titre ?? 'Épreuve' } }
   781→          )
   782→          if (!result.synced) {
   783→            // Hook a déjà toastyé. Si l'outbox échoue aussi, on remonte.
   784→            if (result.error) {
   785→              toast.error('Sauvegarde hors ligne impossible', { description: result.error })
   786→            }
   787→            return
   788→          }
   789→          if (result.error) {
   790→            throw new Error(result.error)
   791→          }
   792→          data = result.data
   793→          break
   794→        }
   795→
   796→        // Online → fetch direct pour pouvoir distinguer 202 des vrais succès
   797→        let res: Response
   798→        try {
   799→          res = await fetch(submitUrl, {
   800→            method: 'POST',
   801→            headers: { 'Content-Type': 'application/json' },
   802→            body: submitBody,
   803→            credentials: 'same-origin',
   804→          })
   805→        } catch (err) {
   806→          // Erreur réseau inattendue → fallback offline outbox (une fois)
   807→          const result = await submitOffline<SubmitResponse>(
   808→            submitUrl,
   809→            { autoSubmit, reponses: reponsesRef.current },
   810→            { type: 'submit-exam', meta: { examTitle: epreuve?.titre ?? 'Épreuve' } }
   811→          )
   812→          if (!result.synced) {
   813→            if (result.error) {
   814→              toast.error('Sauvegarde hors ligne impossible', { description: result.error })
   815→            }
   816→            return
   817→          }
   818→          if (result.error) throw new Error(result.error)
   819→          data = result.data
   820→          break
   821→        }
   822→
   823→        // SUBMIT-RATELIMIT-1 : 202 Accepted = file pleine, retry après Retry-After
   824→        if (res.status === 202) {
   825→          const body = await res.json().catch(() => ({} as { retryAfter?: number }))
   826→          const retryAfter = (body as { retryAfter?: number }).retryAfter ?? 3
   827→          queued = true
   828→          // Indicateur visuel pendant l'attente
   829→          if (attempt === 1) {
   830→            toast.info('Soumission en file d\'attente…', {
   831→              description: `Le serveur traite d'autres copies. Nouvelle tentative dans ${retryAfter}s.`,
   832→              duration: retryAfter * 1000 + 500,
   833→            })
   834→          }
   835→          await new Promise((resolve) => setTimeout(resolve, retryAfter * 1000))
   836→          continue
   837→        }
   838→
   839→        // Vrai succès (200) ou erreur HTTP
   840→        if (!res.ok) {
   841→          const err = await res.json().catch(() => ({}))
   842→          throw new Error((err as { error?: string }).error ?? `HTTP ${res.status}`)
   843→        }
   844→
   845→        data = (await res.json().catch(() => undefined)) as SubmitResponse | undefined
   846→        break
   847→      }
   848→
   849→      if (queued && !data) {
   850→        // Épuisé les retries sans succès — l'étudiant garde sa copie en IndexedDB
   851→        // (déjà sauvée) et le SW retryera. On ne perd rien.
   852→        toast.warning('Soumission en cours de traitement', {
   853→          description: 'Le serveur est très sollicité. Votre copie est sauvegardée et sera soumise automatiquement.',
   854→          duration: 6000,
   855→        })
   856→        return
   857→      }
   858→
   859→      if (!data) {
   860→        throw new Error(lastError ?? 'Réponse vide du serveur')
   861→      }
   862→
   863→      if (autoSubmit) {
   864→        setAutoSubmitted(true)
   865→        setAutoSubmitReason(reason)
   866→      }
   867→
   868→      // Update penalty from server response
   869→      if (data?.penalite !== undefined) {
   870→        setPenalite(data.penalite)
   871→      }
   872→
   873→      // Store submit result for scenario-specific display
   874→      if (data?.scenario) {
   875→        setSubmitResult({
   876→          score: data.score ?? 0,
   877→          totalPossible: data.totalPossible ?? 0,
   878→          autoGradedTotal: data.autoGradableTotal ?? data.totalPossible ?? 0,
   879→          pendingCorrection: data.pendingCorrection ?? 0,
   880→          scenario: data.scenario,
   881→          scenarioMessage: data.scenarioMessage ?? '',
   882→        })
   883→      }
   884→
   885→      setPhase('post-exam')
   886→
   887→      // BUGFIX (EXAM-OFFLINE-1) : nettoyer le cache IndexedDB après submit
   888→      // réussi (les réponses sont maintenant sur le serveur).
   889→      if (sessionRef.current) {
   890→        clearCachedAnswers(sessionRef.current.id)
   891→      }
   892→
   893→      // OPT-10: nettoyer le snapshot IndexedDB après submit réussi
   894→      clearExamSnapshot(epreuveId).catch(() => {})
   895→
   896→      // Exit fullscreen
   897→      try {
   898→        if (document.fullscreenElement) {
   899→          await document.exitFullscreen()
   900→        }
   901→      } catch {
   902→        // Ignore
   903→      }
   904→
   905→      // Clean up intervals
   906→      if (autoSaveIntervalRef.current) clearInterval(autoSaveIntervalRef.current)
   907→      if (timerIntervalRef.current) clearInterval(timerIntervalRef.current)
   908→      if (inactivityIntervalRef.current) clearInterval(inactivityIntervalRef.current)
   909→      if (captureIntervalRef.current) clearInterval(captureIntervalRef.current)
   910→    } catch (err) {
   911→      toast.error('Erreur', {
   912→        description: err instanceof Error ? err.message : 'Impossible de soumettre l\'épreuve',
   913→      })
   914→    } finally {
   915→      setIsSubmitting(false)
   916→      isAutoSubmittingRef.current = false
   917→    }
   918→  }, [isSubmitting, debouncedSaveAnswers, submitOffline, epreuve, cancelPendingAutoSubmit])
   919→
   920→  // ─── Log alert (enhanced with penalty) ─────────────────────────────────
   921→  const logAlert = useCallback(async (type: string, details?: string, alertPenalite?: number) => {
   922→    if (!sessionRef.current) return
   923→
   924→    try {
   925→      await fetch(`/api/sessions/${sessionRef.current.id}`, {
   926→        method: 'PUT',
   927→        headers: { 'Content-Type': 'application/json' },
   928→        body: JSON.stringify({
   929→          alerte: {
   930→            type,
   931→            details,
   932→            penalite: alertPenalite || 0,
   933→          },
   934→        }),
   935→      })
   936→
   937→      // Increment client-side alert count
   938→      setTotalAlertCount((prev) => {
   939→        const newCount = prev + 1
   940→        // Check if we need to auto-submit
   941→        const config = securityConfigRef.current
   942→        if (config.autoSubmitOnViolation && newCount >= config.nbAlertesMax && !isAutoSubmittingRef.current) {
   943→          // Trigger auto-submit via violation dialog
   944→          setAutoSubmitReason('violations')
   945→          setShowViolationDialog(true)
   946→        }
   947→        return newCount
   948→      })
   949→
   950→      // Update penalty
   951→      if (alertPenalite && alertPenalite > 0) {
   952→        setPenalite((prev) => prev + alertPenalite)
   953→      }
   954→    } catch {
   955→      // Silent fail
   956→    }
   957→  }, [])
   958→
   959→  // ─── Handle violation dialog confirmation ──────────────────────────────
   960→  const handleViolationSubmit = useCallback(() => {
   961→    setShowViolationDialog(false)
   962→    submitExam(true, 'violations')
   963→  }, [submitExam])
   964→
   965→  // ─── Capture screenshot ───────────────────────────────────────────────
   966→  const captureScreen = useCallback(async () => {
   967→    if (!sessionRef.current || !examContainerRef.current) return
   968→
   969→    try {
   970→      // Dynamic import of html2canvas-pro
   971→      const html2canvas = (await import('html2canvas-pro')).default
   972→      const canvas = await html2canvas(examContainerRef.current, {
   973→        scale: 0.5, // Lower resolution for smaller payload
   974→        useCORS: true,
   975→        logging: false,
   976→        allowTaint: true,
   977→      })
   978→
   979→      const image = canvas.toDataURL('image/jpeg', 0.4) // Compress to JPEG 40% quality
   980→
   981→      // Upload to server
   982→      await fetch(`/api/sessions/${sessionRef.current.id}/capture`, {
   983→        method: 'POST',
   984→        headers: { 'Content-Type': 'application/json' },
   985→        body: JSON.stringify({ image }),
   986→      })
   987→
   988→      // Show brief flash indicator
   989→      setShowCaptureFlash(true)
   990→      setLastCaptureTime(new Date())
   991→      setTimeout(() => setShowCaptureFlash(false), 1500)
   992→    } catch {
   993→      // Silent fail — don't disrupt the exam
   994→    }
   995→  }, [])
   996→
   997→  // ─── SUBMIT-JITTER-1 (OPT-7) : planification du submit jitteré ────────
   998→  // Quand le temps expire, on ne soumet pas immédiatement. On tire un délai
