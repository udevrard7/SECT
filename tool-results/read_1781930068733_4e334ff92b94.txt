    48→    48→/* ─── Particle system ─── */
    49→    49→function Particles() {
    50→    50→  const canvasRef = useRef<HTMLCanvasElement>(null)
    51→    51→
    52→    52→  useEffect(() => {
    53→    53→    const canvas = canvasRef.current
    54→    54→    if (!canvas) return
    55→    55→    const ctx = canvas.getContext('2d')
    56→    56→    if (!ctx) return
    57→    57→
    58→    58→    let animationId: number
    59→    59→    const particles: { x: number; y: number; vx: number; vy: number; size: number; opacity: number; hue: number }[] = []
    60→    60→
    61→    61→    const resize = () => {
    62→    62→      canvas.width = window.innerWidth
    63→    63→      canvas.height = window.innerHeight
    64→    64→    }
    65→    65→    resize()
    66→    66→    window.addEventListener('resize', resize)
    67→    67→
    68→    68→    for (let i = 0; i < 50; i++) {
    69→    69→      particles.push({
    70→    70→        x: Math.random() * canvas.width,
    71→    71→        y: Math.random() * canvas.height,
    72→    72→        vx: (Math.random() - 0.5) * 0.4,
    73→    73→        vy: (Math.random() - 0.5) * 0.4,
    74→    74→        size: Math.random() * 2.5 + 1,
    75→    75→        opacity: Math.random() * 0.25 + 0.08,
    76→    76→        hue: Math.random() > 0.5 ? 160 : 170,
    77→    77→      })
    78→    78→    }
    79→    79→
    80→    80→    const animate = () => {
    81→    81→      ctx.clearRect(0, 0, canvas.width, canvas.height)
    82→    82→
    83→    83→      particles.forEach((p, i) => {
    84→    84→        p.x += p.vx
    85→    85→        p.y += p.vy
    86→    86→        if (p.x < 0 || p.x > canvas.width) p.vx *= -1
    87→    87→        if (p.y < 0 || p.y > canvas.height) p.vy *= -1
    88→    88→
    89→    89→        ctx.beginPath()
    90→    90→        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2)
    91→    91→        ctx.fillStyle = `hsla(${p.hue}, 70%, 45%, ${p.opacity})`
    92→    92→        ctx.fill()
    93→    93→
    94→    94→        for (let j = i + 1; j < particles.length; j++) {
    95→    95→          const dx = particles[j].x - p.x
    96→    96→          const dy = particles[j].y - p.y
    97→    97→          const dist = Math.sqrt(dx * dx + dy * dy)
    98→    98→          if (dist < 120) {
    99→    99→            ctx.beginPath()
   100→   100→            ctx.moveTo(p.x, p.y)
   101→   101→            ctx.lineTo(particles[j].x, particles[j].y)
   102→   102→            ctx.strokeStyle = `hsla(165, 60%, 45%, ${0.06 * (1 - dist / 120)})`
   103→   103→            ctx.lineWidth = 0.5
   104→   104→            ctx.stroke()
   105→   105→          }
   106→   106→        }
   107→   107→      })
   108→   108→
   109→   109→      animationId = requestAnimationFrame(animate)
   110→   110→    }
   111→   111→    animate()
   112→   112→
   113→   113→    return () => {
   114→   114→      cancelAnimationFrame(animationId)
   115→   115→      window.removeEventListener('resize', resize)
   116→   116→    }
   117→   117→  }, [])
   118→   118→
   119→   119→  return <canvas ref={canvasRef} className="absolute inset-0 pointer-events-none" style={{ zIndex: 1 }} />
   120→   120→}

   122→   122→/* ─── Animated counter ─── */
   123→   123→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
   124→   124→  const [count, setCount] = useState(0)
   125→   125→  const ref = useRef(null)
   126→   126→  const isInView = useInView(ref, { once: true })
   127→   127→
   128→   128→  useEffect(() => {
   129→   129→    if (!isInView) return
   130→   130→    const duration = 2000
   131→   131→    const steps = 60
   132→   132→    const increment = target / steps
   133→   133→    let current = 0
   134→   134→    const timer = setInterval(() => {
   135→   135→      current += increment
   136→   136→      if (current >= target) {
   137→   137→        setCount(target)
   138→   138→        clearInterval(timer)
   139→   139→      } else {
   140→   140→        setCount(Math.floor(current))
   141→   141→      }
   142→   142→    }, duration / steps)
   143→   143→    return () => clearInterval(timer)
   144→   144→  }, [isInView, target])
   145→   145→
   146→   146→  return <span ref={ref}>{count.toLocaleString()}{suffix}</span>
   147→   147→}

   150→   150→function FadeInWhenVisible({
   151→   151→  children,
   152→   152→  delay = 0,
   153→   153→  direction = 'up',
   154→   154→  className,
   155→   155→}: {
   156→   156→  children: React.ReactNode
   157→   157→  delay?: number
   158→   158→  direction?: 'up' | 'down' | 'left' | 'right'
   159→   159→  className?: string
   160→   160→}) {
   161→   161→  const ref = useRef(null)
   162→   162→  const isInView = useInView(ref, { once: true, margin: '-80px' })
   163→   163→
   164→   164→  const directionMap = {
   165→   165→    up: { y: 40, x: 0 },
   166→   166→    down: { y: -40, x: 0 },
   167→   167→    left: { x: 40, y: 0 },
   168→   168→    right: { x: -40, y: 0 },
   169→   169→  }
   170→   170→
   171→   171→  return (
   172→   172→    <motion.div
   173→   173→      ref={ref}
   174→   174→      initial={{ opacity: 0, ...directionMap[direction] }}
   175→   175→      animate={isInView ? { opacity: 1, x: 0, y: 0 } : {}}
   176→   176→      transition={{ duration: 0.6, delay, ease: [0.22, 1, 0.36, 1] }}
   177→   177→      className={className}
   178→   178→    >
   179→   179→      {children}
   180→   180→    </motion.div>
   181→   181→  )
   182→   182→}

   222→   222→/* ─── Typing text animation ─── */
   223→   223→function TypingText({ texts, className }: { texts: string[]; className?: string }) {
   224→   224→  const [currentIndex, setCurrentIndex] = useState(0)
   225→   225→  const [displayText, setDisplayText] = useState('')
   226→   226→  const [isDeleting, setIsDeleting] = useState(false)
   227→   227→
   228→   228→  useEffect(() => {
   229→   229→    const currentFullText = texts[currentIndex]
   230→   230→    let timeout: ReturnType<typeof setTimeout>
   231→   231→
   232→   232→    if (!isDeleting) {
   233→   233→      if (displayText.length < currentFullText.length) {
   234→   234→        timeout = setTimeout(() => {
   235→   235→          setDisplayText(currentFullText.slice(0, displayText.length + 1))
   236→   236→        }, 60)
   237→   237→      } else {
   238→   238→        timeout = setTimeout(() => setIsDeleting(true), 2000)
   239→   239→      }
   240→   240→    } else {
   241→   241→      if (displayText.length > 0) {
   242→   242→        timeout = setTimeout(() => {
   243→   243→          setDisplayText(displayText.slice(0, -1))
   244→   244→        }, 30)
   245→   245→      } else {
   246→   246→        timeout = setTimeout(() => {
   247→   247→          setIsDeleting(false)
   248→   248→          setCurrentIndex((prev) => (prev + 1) % texts.length)
   249→   249→        }, 300)
   250→   250→      }
   251→   251→    }
   252→   252→
   253→   253→    return () => clearTimeout(timeout)
   254→   254→  }, [displayText, isDeleting, currentIndex, texts])
   255→   255→
   256→   256→  return (
   257→   257→    <span className={className}>
   258→   258→      {displayText}
   259→   259→      <motion.span
   260→   260→        animate={{ opacity: [1, 0] }}
   261→   261→        transition={{ duration: 0.5, repeat: Infinity, repeatType: 'reverse' }}
   262→   262→        className="inline-block w-[3px] h-[1em] bg-emerald-600 ml-1 align-middle"
   263→   263→      />
   264→   264→    </span>
   265→   265→  )
   266→   266→}
