   346→function LogoUpload({
   347→  logo,
   348→  etablissementId,
   349→  accent,
   350→  onLogoUpdate,
   351→}: {
   352→  logo: string | null
   353→  etablissementId: string
   354→  accent: ReturnType<typeof getAccent>
   355→  onLogoUpdate: (logo: string | null) => void
   356→}) {
   357→  const fileInputRef = useRef<HTMLInputElement>(null)
   358→  const [uploading, setUploading] = useState(false)
   359→  const [dragOver, setDragOver] = useState(false)
   360→
   361→  const handleFileSelect = async (file: File) => {
   362→    const allowedTypes = ['image/png', 'image/jpeg', 'image/webp', 'image/svg+xml']
   363→    if (!allowedTypes.includes(file.type)) {
   364→      toast.error('Format non supporté', { description: 'Utilisez PNG, JPG, WEBP ou SVG.' })
   365→      return
   366→    }
   367→    if (file.size > 2 * 1024 * 1024) {
   368→      toast.error('Fichier trop volumineux', { description: 'La taille maximale est de 2 Mo.' })
   369→      return
   370→    }
   371→
   372→    setUploading(true)
   373→    try {
   374→      const formData = new FormData()
   375→      formData.append('file', file)
   376→      formData.append('etablissementId', etablissementId)
   377→      const res = await fetch('/api/etablissements/upload-logo', {
   378→        method: 'POST',
   379→        body: formData,
   380→      })
   381→      if (!res.ok) {
   382→        const err = await res.json().catch(() => ({}))
   383→        throw new Error(err.error || 'Erreur lors de l\'upload')
   384→      }
   385→      const data = await res.json()
   386→      onLogoUpdate(data.logo || data.etablissement?.logo || null)
   387→      toast.success('Logo mis à jour', { description: 'Le logo a été téléchargé avec succès.' })
   388→    } catch (err) {
   389→      toast.error('Erreur', {
   390→        description: err instanceof Error ? err.message : 'Impossible de télécharger le logo.',
   391→      })
   392→    } finally {
   393→      setUploading(false)
   394→    }
   395→  }
   396→
   397→  const handleDrop = (e: React.DragEvent) => {
   398→    e.preventDefault()
   399→    setDragOver(false)
   400→    const file = e.dataTransfer.files[0]
   401→    if (file) handleFileSelect(file)
   402→  }
   403→
   404→  const handleDelete = async () => {
   405→    setUploading(true)
   406→    try {
   407→      // PARAMETRES-FIX-P3 : endpoint dédié DELETE /api/etablissements/{id}/logo.
   408→      // PARAMETRES-FIX-P3b : le backend utilise désormais ClearLogo (SET logo = NULL)
   409→      // au lieu d'UpdateLogo(...,"") qui était rejeté par la validation.
   410→      const res = await fetch(`/api/etablissements/${etablissementId}/logo`, {
   411→        method: 'DELETE',
   412→      })
   413→      if (!res.ok) {
   414→        const err = await res.json().catch(() => ({}))
   415→        throw new Error(err.error || err.message || `Erreur ${res.status}`)
   416→      }
   417→      onLogoUpdate(null)
   418→      toast.success('Logo supprimé', { description: 'Le logo a été retiré.' })
   419→    } catch (err) {
   420→      toast.error('Erreur', {
   421→        description: err instanceof Error ? err.message : 'Impossible de supprimer le logo.',
   422→      })
   423→    } finally {
   424→      setUploading(false)
   425→    }
   426→  }
   427→
   428→  return (
   429→    <div className="space-y-3">
   430→      <Label className="flex items-center gap-1.5">
   431→        <ImageIcon className="h-3.5 w-3.5 text-muted-foreground" />
   432→        Logo de l&apos;établissement
   433→      </Label>
   434→      <div className="flex items-start gap-4">
   435→        {/* Preview */}
   436→        <div
   437→          className={`relative flex h-20 w-20 shrink-0 items-center justify-center rounded-xl border-2 border-dashed transition-colors cursor-pointer overflow-hidden ${
   438→            dragOver ? `${accent.activeBorder} ${accent.activeBg}` : 'border-muted-foreground/25 hover:border-muted-foreground/50'
   439→          }`}
   440→          onClick={() => fileInputRef.current?.click()}
   441→          onDragOver={(e) => { e.preventDefault(); setDragOver(true) }}
   442→          onDragLeave={() => setDragOver(false)}
   443→          onDrop={handleDrop}
   444→        >
   445→          {logo ? (
   446→            <img src={logo} alt="Logo" className="h-full w-full object-cover rounded-xl" />
   447→          ) : (
   448→            <div className="flex flex-col items-center gap-1 text-muted-foreground/50">
   449→              <Upload className="h-5 w-5" />
   450→              <span className="text-[10px]">Clic/Glisser</span>
   451→            </div>
   452→          )}
   453→          {uploading && (
   454→            <div className="absolute inset-0 flex items-center justify-center bg-background/80 rounded-xl">
   455→              <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   456→            </div>
   457→          )}
   458→        </div>
   459→        <div className="flex-1 space-y-2">
   460→          <p className="text-xs text-muted-foreground">
   461→            Cliquez ou glissez-déposez une image. Formats acceptés : PNG, JPG, WEBP, SVG (max 2 Mo).
   462→          </p>
   463→          <div className="flex gap-2">
   464→            <Button
   465→              variant="outline"
   466→              size="sm"
   467→              onClick={() => fileInputRef.current?.click()}
   468→              disabled={uploading}
   469→            >
   470→              <Upload className="h-3.5 w-3.5 mr-1.5" />
   471→              Parcourir
   472→            </Button>
   473→            {logo && (
   474→              <Button
   475→                variant="outline"
   476→                size="sm"
   477→                onClick={handleDelete}
   478→                disabled={uploading}
   479→                className="text-destructive hover:text-destructive hover:bg-destructive/10"
   480→              >
   481→                <Trash2 className="h-3.5 w-3.5 mr-1.5" />
   482→                Supprimer
   483→              </Button>
   484→            )}
   485→          </div>
   486→        </div>
   487→      </div>
   488→      <input
   489→        ref={fileInputRef}
   490→        type="file"
   491→        accept="image/png,image/jpeg,image/webp,image/svg+xml"
   492→        className="hidden"
   493→        onChange={(e) => {
   494→          const file = e.target.files?.[0]
   495→          if (file) handleFileSelect(file)
   496→          e.target.value = ''
   497→        }}
   498→      />
   499→    </div>
   500→  )
   501→}
