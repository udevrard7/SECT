    63→function formatDateTime(date: string | Date | null): string {
    64→  if (!date) return '—'
    65→  const d = typeof date === 'string' ? new Date(date) : date
    66→  return d.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric', hour: '2-digit', minute: '2-digit', second: '2-digit' })
    67→}
    68→
    69→function formatRelativeDate(dateStr: string): string {
    70→  const date = new Date(dateStr)
    71→  const now = new Date()
    72→  const diffMs = now.getTime() - date.getTime()
    73→  const diffMin = Math.floor(diffMs / 60000)
    74→  const diffH = Math.floor(diffMin / 60)
    75→  const diffD = Math.floor(diffH / 24)
    76→  if (diffMin < 1) return "À l'instant"
    77→  if (diffMin < 60) return `Il y a ${diffMin} min`
    78→  if (diffH < 24) return `Il y a ${diffH}h`
    79→  if (diffD < 7) return `Il y a ${diffD}j`
    80→  return date.toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit' })
    81→}
    82→
    83→function formatTime(date: string): string {
    84→  return new Date(date).toLocaleTimeString('fr-FR', { hour: '2-digit', minute: '2-digit', second: '2-digit' })
    85→}

   149→function useDebounce<T>(value: T, delay = 350): T {
   150→  const [debounced, setDebounced] = useState(value)
   151→  useEffect(() => {
   152→    const t = setTimeout(() => setDebounced(value), delay)
   153→    return () => clearTimeout(t)
   154→  }, [value, delay])
   155→  return debounced
   156→}
