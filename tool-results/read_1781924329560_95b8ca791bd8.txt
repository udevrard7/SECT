   185→// ─── Custom Pie Label ───
   186→
   187→function renderPieLabel({
   188→  cx,
   189→  cy,
   190→  midAngle,
   191→  outerRadius,
   192→  percent,
   193→  name,
   194→  value,
   195→}: {
   196→  cx: number
   197→  cy: number
   198→  midAngle: number
   199→  innerRadius: number
   200→  outerRadius: number
   201→  percent: number
   202→  name: string
   203→  value: number
   204→}) {
   205→  const RADIAN = Math.PI / 180
   206→  const radius = outerRadius + 28
   207→  const x = cx + radius * Math.cos(-midAngle * RADIAN)
   208→  const y = cy + radius * Math.sin(-midAngle * RADIAN)
   209→
   210→  if (percent < 0.03) return null
   211→
   212→  return (
   213→    <text
   214→      x={x}
   215→      y={y}
   216→      textAnchor={x > cx ? 'start' : 'end'}
   217→      dominantBaseline="central"
   218→      className="text-xs fill-muted-foreground"
   219→    >
   220→      {name} ({value}, {`${(percent * 100).toFixed(0)}%`})
   221→    </text>
   222→  )
   223→}

   227→function RevenueTooltip({ active, payload, label }: { active?: boolean; payload?: Array<{ name: string; value: number; color: string }>; label?: string }) {
   228→  if (!active || !payload) return null
   229→  return (
   230→    <div className="rounded-lg border bg-background p-3 shadow-lg">
   231→      <p className="mb-1 text-sm font-medium text-foreground">{label}</p>
   232→      {payload.map((entry, idx) => (
   233→        <p key={idx} className="text-xs" style={{ color: entry.color }}>
   234→          Revenus : {entry.value.toLocaleString('fr-FR')} FCFA
   235→        </p>
   236→      ))}
   237→    </div>
   238→  )
   239→}
